# Base image: Ubuntu 22.04 LTS (Jammy Jellyfish)
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Perform a single, multi-step RUN command to minimize layer count.
# This command will:
# 1. Update the package index and install all necessary build dependencies.
#    - Includes build-essential, git, wget for source retrieval and compilation.
#    - Includes gfortran as required by many HPC applications.
#    - Includes autoconf, automake, libtool, flex, hwloc for OpenMPI build.
#    - Includes ca-certificates for secure network communication (e.g., git, wget).
#    - Includes libfabric-dev for AWS EFA (Elastic Fabric Adapter) support.
#    - Includes libgomp1 as explicitly required for OpenMP support.
# 2. Download, compile, and install OpenMPI version 4.1.2.
#    - Configured with --with-libfabric=/usr to enable high-performance networking on AWS.
#    - Installed into /usr/local, which is in the default PATH.
# 3. Clone, build, and install the amg2023 (LLNL AMG) application.
#    - The latest branch is cloned from the official repository.
#    - The 'amg' executable is built from the test directory.
#    - The executable is copied directly to /usr/local/bin/ to be available in the system PATH.
# 4. Clean up all build artifacts, source code, and package manager caches.
#    - This reduces the final image size significantly.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    git \
    wget \
    make \
    autoconf \
    automake \
    libtool \
    flex \
    hwloc \
    ca-certificates \
    libfabric-dev \
    libgomp1 && \
    # --- Build and Install OpenMPI ---
    # Use /tmp as a temporary build directory
    cd /tmp && \
    OMPI_VERSION="4.1.2" && \
    wget https://download.open-mpi.org/release/open-mpi/v4.1/openmpi-${OMPI_VERSION}.tar.gz && \
    tar -xvf openmpi-${OMPI_VERSION}.tar.gz && \
    cd openmpi-${OMPI_VERSION} && \
    # Configure OpenMPI for a system-wide install, enabling libfabric for AWS EFA support
    ./configure --prefix=/usr/local --with-libfabric=/usr && \
    make -j$(nproc) all && \
    make install && \
    # Update the dynamic linker cache to recognize new OpenMPI libraries
    ldconfig && \
    # --- Build and Install amg2023 ---
    # The application name is amg2023, which corresponds to the LLNL AMG benchmark
    cd /tmp && \
    git clone --depth 1 https://github.com/LLNL/AMG.git && \
    cd AMG/test && \
    make amg && \
    # Copy the final executable to a directory on the system PATH
    cp amg /usr/local/bin/ && \
    # --- Final Cleanup ---
    # Remove all temporary build files and source code
    rm -rf /tmp/* && \
    # Clean apt cache to reduce final image size
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# No CMD or ENTRYPOINT is specified. The container is designed to be run
# by a workload manager (e.g., in Kubernetes) which will provide the
# command and arguments at runtime, such as:
# mpirun -np <NUM_PROCS> amg -n <X> <Y> <Z>

    # Source code retrieval

    ldconfig && \
