# Base Image: Ubuntu 22.04 LTS for a modern build environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, git, wget, ca-certificates, and libfabric for EFA support.
# Also install libgomp1 as required for the application's runtime dependencies.
# Clean up apt cache in the same layer to reduce image size.
RUN apt-get update && apt-get install -y \
    build-essential \
    gfortran \
    git \
    wget \
    make \
    ca-certificates \
    libfabric-dev \
    librdmacm-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Define OpenMPI version and installation path as environment variables for clarity and reusability
ENV OMPI_VERSION=4.1.2
ENV OMPI_PATH=/usr/local/openmpi

# Add OpenMPI to the PATH and LD_LIBRARY_PATH for system-wide access
ENV PATH=$OMPI_PATH/bin:$PATH
ENV LD_LIBRARY_PATH=$OMPI_PATH/lib:$LD_LIBRARY_PATH

# Download, build from source, and install OpenMPI.
# This build is configured with libfabric support for high-performance networking on AWS (EFA).
# --enable-orterun-prefix-by-default is critical for containerized environments.
# Build artifacts are cleaned up in the same layer.
RUN mkdir -p /tmp/ompi-build \
    && cd /tmp/ompi-build \
    && wget https://download.open-mpi.org/release/open-mpi/v4.1/openmpi-${OMPI_VERSION}.tar.gz \
    && tar -xvf openmpi-${OMPI_VERSION}.tar.gz \
    && cd openmpi-${OMPI_VERSION} \
    && ./configure --prefix=${OMPI_PATH} \
                   --with-libfabric=/usr \
                   --enable-orterun-prefix-by-default \
    && make -j$(nproc) all \
    && make install \
    && cd / \
    && rm -rf /tmp/ompi-build

# Clone the latest branch of the amg2023 benchmark from its canonical repository.
# Build the application using the OpenMPI compilers now available on the PATH.
# Copy the final executable to a system PATH directory, as required.
# Clean up the source code to minimize final image size.
RUN mkdir -p /opt/build \
    && cd /opt/build \
    && git clone https://github.com/LLNL/AMG.git \
    && cd AMG/test \
    && make \
    && cp amg /usr/local/bin/amg \
    && cd / \
    && rm -rf /opt/build

# Set a default working directory for when the container starts.
WORKDIR /

# Base Image: Ubuntu 22.04 LTS for a modern build environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, git, wget, ca-certificates, and libfabric for EFA support.
# Also install libgomp1 as required for the application's runtime dependencies.
# Clean up apt cache in the same layer to reduce image size.
RUN apt-get update && apt-get install -y \
    build-essential \
    gfortran \
    git \
    wget \
    make \
    ca-certificates \
    libfabric-dev \
    librdmacm-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Define OpenMPI version and installation path as environment variables for clarity and reusability
ENV OMPI_VERSION=4.1.2
ENV OMPI_PATH=/usr/local/openmpi

# Add OpenMPI to the PATH and LD_LIBRARY_PATH for system-wide access
ENV PATH=$OMPI_PATH/bin:$PATH
ENV LD_LIBRARY_PATH=$OMPI_PATH/lib:$LD_LIBRARY_PATH

# Download, build from source, and install OpenMPI.
# This build is configured with libfabric support for high-performance networking on AWS (EFA).
# --enable-orterun-prefix-by-default is critical for containerized environments.
# Build artifacts are cleaned up in the same layer.
RUN mkdir -p /tmp/ompi-build \
    && cd /tmp/ompi-build \
    && wget https://download.open-mpi.org/release/open-mpi/v4.1/openmpi-${OMPI_VERSION}.tar.gz \
    && tar -xvf openmpi-${OMPI_VERSION}.tar.gz \
    && cd openmpi-${OMPI_VERSION} \
    && ./configure --prefix=${OMPI_PATH} \
                   --with-libfabric=/usr \
                   --enable-orterun-prefix-by-default \
    && make -j$(nproc) all \
    && make install \
    && cd / \
    && rm -rf /tmp/ompi-build

# Clone the latest branch of the amg2023 benchmark from its canonical repository.
# FIX: Build from the root of the AMG directory. The top-level Makefile correctly
# handles building all dependency libraries before building the final executable.
# This resolves the linker errors from the previous build attempt.
RUN mkdir -p /opt/build \
    && cd /opt/build \
    && git clone https://github.com/LLNL/AMG.git \
    && cd AMG \
    && make \
    && cp test/amg /usr/local/bin/ \
    && cd / \
    && rm -rf /opt/build

# Set a default working directory for when the container starts.
WORKDIR /
