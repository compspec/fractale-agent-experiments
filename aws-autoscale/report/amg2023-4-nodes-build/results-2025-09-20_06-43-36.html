# Dockerfile for amg2023 for AWS CPU multi-node environment
# Base Image: Ubuntu 22.04
FROM ubuntu:22.04

# Set environment variables to avoid interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Pin versions for reproducible builds
ENV LIBFABRIC_VERSION=1.15.2
ENV OPENMPI_VERSION=4.1.2

# Install build-time and run-time dependencies
# - build-essential, git, wget, etc., are required for compiling the components
# - ca-certificates is required for secure downloads
# - libgomp1 is a specific runtime dependency for amg2023
# - libudev-dev, libnl-*-dev, rdma-core are dependencies for building libfabric with EFA support
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    git \
    wget \
    ca-certificates \
    autoconf \
    automake \
    libtool \
    make \
    cmake \
    libgomp1 \
    libudev-dev \
    libnl-3-dev \
    libnl-route-3-dev \
    rdma-core && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Build and install libfabric with EFA (Elastic Fabric Adapter) support for AWS
# This is critical for high-performance, low-latency multi-node communication in AWS
RUN mkdir /tmp/build-libfabric && \
    cd /tmp/build-libfabric && \
    wget https://github.com/ofiwg/libfabric/releases/download/v${LIBFABRIC_VERSION}/libfabric-${LIBFABRIC_VERSION}.tar.gz && \
    tar -zxf libfabric-${LIBFABRIC_VERSION}.tar.gz && \
    cd libfabric-${LIBFABRIC_VERSION} && \
    ./configure --prefix=/usr/local --enable-efa && \
    make -j$(nproc) all && \
    make install && \
    cd / && \
    rm -rf /tmp/build-libfabric

# Build and install OpenMPI, configured to use the EFA-enabled libfabric
# --disable-getpwuid is a common optimization for container environments
# ldconfig updates the shared library cache to include the newly installed libraries
RUN mkdir /tmp/build-openmpi && \
    cd /tmp/build-openmpi && \
    wget https://download.open-mpi.org/release/open-mpi/v$(echo $OPENMPI_VERSION | cut -d. -f1-2)/openmpi-${OPENMPI_VERSION}.tar.gz && \
    tar -zxf openmpi-${OPENMPI_VERSION}.tar.gz && \
    cd openmpi-${OPENMPI_VERSION} && \
    ./configure --prefix=/usr/local \
                --with-libfabric=/usr/local \
                --disable-getpwuid && \
    make -j$(nproc) all && \
    make install && \
    ldconfig && \
    cd / && \
    rm -rf /tmp/build-openmpi

# Clone, build, and install the amg2023 application (LLNL/AMG)
# The executable 'amg' is copied to /usr/local/bin to make it available on the system PATH
RUN mkdir /app && \
    cd /app && \
    git clone https://github.com/LLNL/AMG.git && \
    cd AMG && \
    make && \
    cp test/amg /usr/local/bin/amg && \
    cd / && \
    rm -rf /app

# Set a default command to be executed when the container starts
# This will display the MPI version and confirm the installation
CMD ["mpirun", "--version"]

    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# This is critical for high-performance, low-latency multi-node communication in AWS
# FIX: Corrected the libfabric source tarball filename to remove the 'v' prefix, matching the actual release asset name.

    tar -zxf libfabric-${LIBFABRIC_VERSION}.tar.gz && \

    make -j$(nproc) all && \
    make install && \
    cd / && \

# Build and install libfabric with EFA (Elastic Fabric Adapter) support for AWS
# This is critical for high-performance, low-latency multi-node communication in AWS
RUN mkdir /tmp/build-libfabric && \
    cd /tmp/build-libfabric && \
    wget https://github.com/ofiwg/libfabric/releases/download/v${LIBFABRIC_VERSION}/libfabric-${LIBFABRIC_VERSION}.tar.bz2 && \
    tar -jxf libfabric-${LIBFABRIC_VERSION}.tar.bz2 && \
    cd libfabric-${LIBFABRIC_VERSION} && \
    ./configure --prefix=/usr/local --enable-efa && \
    make -j$(nproc) all && \
    make install && \
    cd / && \
    rm -rf /tmp/build-libfabric

# Build and install OpenMPI, configured to use the EFA-enabled libfabric
# --disable-getpwuid is a common optimization for container environments
# ldconfig updates the shared library cache to include the newly installed libraries
RUN mkdir /tmp/build-openmpi && \
    cd /tmp/build-openmpi && \
    wget https://download.open-mpi.org/release/open-mpi/v$(echo $OPENMPI_VERSION | cut -d. -f1-2)/openmpi-${OPENMPI_VERSION}.tar.gz && \
    tar -zxf openmpi-${OPENMPI_VERSION}.tar.gz && \
    cd openmpi-${OPENMPI_VERSION} && \
    ./configure --prefix=/usr/local \
                --with-libfabric=/usr/local \
                --disable-getpwuid && \
    make -j$(nproc) all && \
    make install && \
    ldconfig && \
    cd / && \
    rm -rf /tmp/build-openmpi
