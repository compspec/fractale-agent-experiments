# Base image: Ubuntu 22.04 for a modern, stable environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installation to prevent prompts

# Install essential build tools, git, MPI, and required libraries
# ca-certificates: Required for secure network communications (e.g., git clone)
# openmpi-bin, libopenmpi-dev: OpenMPI runtime and development headers
# libgomp1: Required for GNU OpenMP support, a dependency for amg
# The entire operation is in one RUN layer to reduce image size.
# We do not use 'apt-get purge' as instructed.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    make \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    libgomp1 \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

# Clone the latest version of the AMG benchmark, build it,
# copy the executable to a system path, and clean up the source code.
# This is done in a single RUN command to ensure the source code
# does not remain in an intermediate layer, optimizing the final image size.
RUN git clone https://github.com/LLNL/AMG.git /tmp/amg2023 \
    && cd /tmp/amg2023/test \
    && make \
    && cp amg /usr/local/bin/amg \
    && cd / \
    && rm -rf /tmp/amg2023

# Configure OpenMPI for containerized environments like Kubernetes on AWS CPU instances.
# These settings ensure reliable communication over the container's virtual ethernet interface.
# 1. OMPI_ALLOW_RUN_AS_ROOT: Allow MPI to run as the root user.
# 2. OMPI_MCA_btl_tcp_if_include: Force TCP communication over the primary network interface (typically eth0 in containers).
# 3. OMPI_MCA_oob_tcp_if_include: Do the same for out-of-band communication.
# 4. OMPI_MCA_btl: Exclude high-performance fabrics like InfiniBand (^openib) which are not present.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_oob_tcp_if_include=eth0
ENV OMPI_MCA_btl=^openib

# Set a default working directory for application execution.
WORKDIR /app

# Default command to run the application's help menu.
# This can be overridden at runtime, e.g., with 'mpirun -np 4 amg ...'
CMD ["amg", "--help"]
