FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments
# Allow running as root, which is common in container scenarios
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Use TCP for inter-container communication, disable shared memory mechanisms that may not work across nodes
ENV OMPI_MCA_btl=tcp,self
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Install dependencies, clone source, build, install, and clean up in a single layer
# This avoids creating unnecessary image layers, which is a best practice when not using multi-stage builds.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    # Clone the latest branch of the AMG source code
    git clone https://github.com/LLNL/AMG.git /opt/amg_src && \
    # Navigate to the test directory which contains the Makefile for the benchmark
    cd /opt/amg_src/test && \
    # Build the amg executable using the provided Makefile
    make && \
    # Copy the compiled executable to a system location on the PATH
    cp amg /usr/local/bin/amg && \
    # Clean up the source code to reduce final image size
    cd / && \
    rm -rf /opt/amg_src && \
    # Clean up apt cache to further reduce image size
    rm -rf /var/lib/apt/lists/*

# Set a working directory for when the container starts
WORKDIR /

# Provide a default command. This can be overridden at runtime.
# Running a shell is useful for interactive debugging or launching the application with custom parameters.
CMD ["/bin/bash"]
