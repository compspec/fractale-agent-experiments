# Use a standard Ubuntu base image for broad compatibility and package availability.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent apt-get from prompting for input.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, and required libraries in a single RUN command.
# This approach helps to minimize the number of layers in the Docker image.
# - build-essential, gfortran, cmake: For compiling C/C++ and Fortran code.
# - git: For cloning the application source code from a repository.
# - openmpi-bin, libopenmpi-dev: Open MPI runtime and development files.
# - ca-certificates: For secure HTTPS connections during clone.
# - libgomp1: A specific requirement for GNU OpenMP runtime library.
# We run apt-get update and install in the same command, then clean up the apt cache.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    git \
    cmake \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments, typical in Kubernetes.
# These settings force MPI to use TCP over the standard network interface (e.g., eth0)
# and prevent it from searching for specialized interconnects like InfiniBand or shared-memory
# optimizations that can fail or cause performance issues in a container.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=tcp,self
ENV OMPI_MCA_plm_rsh_agent=ssh

# Create a temporary working directory for the build process.
WORKDIR /build

# Clone the latest branch of the AMG (amg2023) source code.
RUN git clone https://github.com/LLNL/AMG.git

# Change to the test directory where the main application is built.
WORKDIR /build/AMG/test

# Compile the application using the provided Makefile.
RUN make

# Copy the compiled 'amg' executable to /usr/local/bin, which is on the system's PATH.
# This makes the application directly executable from any directory.
RUN cp amg /usr/local/bin/amg

# Clean up the build directory to reduce the final image size.
RUN rm -rf /build

# Set the final working directory for the container.
WORKDIR /root

# The Docker image is now built. The application can be run via mpirun.
# Example usage: mpirun -np <num_procs> amg <args>
