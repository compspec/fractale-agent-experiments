# Use a standard Ubuntu 22.04 base image
FROM ubuntu:22.04

# Set non-interactive frontend for package management to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, git, OpenMPI, required libraries, and certificates
# libgomp1 is required for OpenMP support at runtime
# ca-certificates are necessary for secure connections (e.g., git clone)
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    ca-certificates && \
    # Clean up the apt cache to reduce image size
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for container environments
# Allow running MPI applications as the root user inside the container
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Prevent OpenMPI from using shared memory mechanisms that can cause issues in some container runtimes
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
# Disable help message aggregation which can be problematic in some environments
ENV OMPI_MCA_orte_base_help_aggregate=0
# Use async modex for better scalability in some network environments
ENV OMPI_MCA_pmix_base_async_modex=1

# Create a temporary directory for building the application
WORKDIR /build

# Clone the latest branch of the amg2023 (AMG) repository from GitHub,
# build the executable, copy it to a location on the system PATH,
# and then clean up the build directory to minimize final image size.
RUN git clone https://github.com/LLNL/AMG.git && \
    cd AMG/test && \
    make && \
    cp amg /usr/local/bin/amg && \
    cd / && \
    rm -rf /build

# Set a default entrypoint or command to facilitate running the container
CMD ["/bin/bash"]
