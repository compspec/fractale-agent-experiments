# Base Image: Ubuntu 22.04 LTS for a stable, recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies for building LAMMPS and running MPI applications
# - ca-certificates: For secure network communication (e.g., git clone)
# - build-essential, cmake, git: Core tools for compiling C++ projects
# - libfftw3-dev: FFTW library, a common LAMMPS dependency
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel execution
# - libgomp1: GNU OpenMP library, explicitly requested
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    ca-certificates \
    build-essential \
    cmake \
    git \
    libfftw3-dev \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments
# - Allow running as root, common in containers
# - Enable oversubscribing nodes, useful in Kubernetes/resource-limited environments
# - Disable rsh/ssh as a process launch agent, not applicable in single-container MPI
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_rmaps_base_oversubscribe=1
ENV OMPI_MCA_plm_rsh_agent=/bin/false

# Clone, build, and install LAMMPS from source
# This single RUN command improves layer caching and reduces image size
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D CMAKE_INSTALL_PREFIX=/usr/local \
        -D BUILD_MPI=yes \
        -D BUILD_OMP=yes \
        -D PKG_REAXFF=on \
        -D PKG_MANYBODY=on && \
    make -j"$(nproc)" && \
    # Install the 'lmp' executable to a system location on the PATH
    make install && \
    # Create the application WORKDIR and place example files there
    mkdir -p /app && \
    cp /lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up the source code to reduce final image size
    rm -rf /lammps

# Set the working directory for the container
WORKDIR /app

# Set a default command to run a sample simulation
CMD ["lmp", "-in", "in.hns"]
