# Base image: Ubuntu 22.04 LTS for a stable, well-supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments like Kubernetes
# This allows running MPI processes as the root user inside the container
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build dependencies, MPI, LAMMPS prerequisites, and required certificates
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    libopenmpi-dev \
    openmpi-bin \
    libfftw3-dev \
    libgomp1 && \
    # Clean up apt cache to reduce final image size
    rm -rf /var/lib/apt/lists/*

# Clone the LAMMPS source code from the official repository
# The 'stable' branch is used as it represents the latest stable release, suitable for production workloads
RUN git clone --branch stable https://github.com/lammps/lammps.git /usr/src/lammps

# Build LAMMPS with MPI support and the REAXFF package enabled
# This is a CPU-only build configured for performance (Release type)
RUN mkdir /usr/src/lammps/build && \
    cd /usr/src/lammps/build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=yes && \
    # Compile using all available processor cores for a faster build
    make -j$(nproc) && \
    # Copy the main LAMMPS executable 'lmp' to a system location on the PATH
    # This makes the executable available globally without needing to modify PATH or use symlinks
    cp lmp /usr/local/bin/

# Set a working directory for application data and examples
WORKDIR /app

# Copy the specific ReaxFF example files for HNS into the working directory
# These files can be used for testing or as a starting point for simulations
RUN cp /usr/src/lammps/examples/reaxff/HNS/* .

# Clean up the downloaded source code to reduce the final image size
RUN rm -rf /usr/src/lammps

# Set a default command to provide an interactive shell
# In a Kubernetes deployment, this will typically be overridden by the pod's command/args
CMD ["/bin/bash"]
