# Use a standard Ubuntu base image suitable for CPU builds
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build-essential, git, cmake, MPI, and other LAMMPS dependencies
# Also installs ca-certificates and libgomp1 as required
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest branch of the LAMMPS source code, build it, and clean up
# This is done in a single RUN layer to optimize image size
RUN git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps-src && \
    cd /opt/lammps-src && \
    mkdir build && \
    cd build && \
    # Configure the build with the required REAXFF package and MPI support
    cmake ../cmake \
      -D PKG_REAXFF=on \
      -D BUILD_MPI=on \
      -D CMAKE_BUILD_TYPE=Release && \
    # Build LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the compiled executable to a system location on the PATH
    cp lmp /usr/local/bin/ && \
    # Create the final working directory
    mkdir -p /app && \
    # Copy the required example files into the working directory
    cp /opt/lammps-src/examples/reaxff/HNS/* /app/ && \
    # Clean up the source and build directories to minimize final image size
    rm -rf /opt/lammps-src

# Set the working directory for the container
WORKDIR /app

# The container is now ready. The 'lmp' executable is on the PATH.
# To run an MPI job, use a command like:
# mpirun -np <num_procs> --allow-run-as-root --mca btl_tcp_if_exclude lo,docker0 lmp -in <your_input_script>
# Setting a default command to an interactive shell for ease of use.
CMD ["/bin/bash"]
