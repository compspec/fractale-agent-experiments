# Base Image: Ubuntu 22.04 LTS for a stable, common base
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies, build LAMMPS with REAXFF, and clean up in a single layer
# This approach minimizes image size in a single-stage build.
RUN apt-get update && \
    # Install build-time and run-time dependencies
    # ca-certificates: for secure network communication (e.g., git clone)
    # git, build-essential, cmake: for compiling the source code
    # openmpi-bin, libopenmpi-dev: for MPI support
    # libgomp1: required OpenMP runtime library
    apt-get install -y --no-install-recommends \
        ca-certificates \
        git \
        build-essential \
        cmake \
        openmpi-bin \
        libopenmpi-dev \
        libgomp1 && \
    #
    # Clone the latest stable branch of the LAMMPS repository
    git clone --depth 1 -b stable https://github.com/lammps/lammps.git /opt/lammps && \
    cd /opt/lammps && \
    #
    # Configure and build LAMMPS using CMake
    # -D PKG_REAXFF=on: Enables the ReaxFF package required for the simulation
    mkdir build && \
    cd build && \
    cmake \
        -D PKG_REAXFF=on \
        ../cmake && \
    #
    # Compile using all available processor cores
    make -j$(nproc) && \
    #
    # Copy the main LAMMPS executable to a location on the system PATH
    cp lmp /usr/local/bin/ && \
    #
    # Set up the working directory and copy example files
    mkdir -p /app && \
    cp /opt/lammps/examples/reaxff/HNS/* /app/ && \
    #
    # Clean up build dependencies and source code to reduce final image size
    cd / && \
    rm -rf /opt/lammps && \
    apt-get purge -y git build-essential cmake libopenmpi-dev && \
    apt-get autoremove -y && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set the default working directory for the container
WORKDIR /app

# For MPI applications in containers, it's often necessary to specify network interfaces at runtime.
# Example for Open MPI with mpirun:
# mpirun -np 4 --mca btl_tcp_if_exclude lo,docker0 lmp -in in.hns
# The ENTRYPOINT is set to the LAMMPS executable so users can pass arguments directly.
ENTRYPOINT ["lmp"]

# Default command to run if no other command is provided
CMD ["--help"]
