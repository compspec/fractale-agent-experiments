# Use a standard Ubuntu base image for broad compatibility
FROM ubuntu:22.04

# To run MPI as root in a container, this environment variable is often needed.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install all necessary dependencies in a single layer to optimize image size.
# This includes build tools, git, cmake, OpenMPI for parallel execution,
# FFTW library which is a common LAMMPS dependency, libgomp1 for OpenMP support,
# and ca-certificates for secure network access.
RUN apt-get update && \
    DEBIAN_FRONTEND=noninteractive apt-get install -y --no-install-recommends \
    ca-certificates \
    git \
    cmake \
    build-essential \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# This single RUN command performs all steps to download, build, and install LAMMPS.
# Chaining these commands together and cleaning up in the same layer significantly
# reduces the final Docker image size.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps_src && \
    mkdir -p /lammps_src/build && \
    cd /lammps_src/build && \
    # Configure the LAMMPS build using CMake.
    # -D BUILD_MPI=yes: Enables MPI support.
    # -D PKG_REAXFF=yes: Enables the ReaxFF package as requested.
    # -D PKG_KSPACE=yes: Includes FFT-based solvers, common for many simulations.
    # -D CMAKE_INSTALL_PREFIX=/usr/local: Ensures binaries are installed to a standard PATH location.
    cmake ../cmake \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=yes \
      -D PKG_KSPACE=yes \
      -D CMAKE_INSTALL_PREFIX=/usr/local && \
    # Build LAMMPS using all available processor cores.
    make -j$(nproc) && \
    # Install the 'lmp' executable to /usr/local/bin, making it available on the system PATH.
    make install && \
    # Create the final working directory.
    mkdir /app && \
    # Copy the specified example files into the working directory.
    cp /lammps_src/examples/reaxff/HNS/* /app/ && \
    # Clean up the source code and build artifacts to minimize the final image size.
    cd / && \
    rm -rf /lammps_src

# Set the final working directory for the container.
WORKDIR /app

# The 'lmp' executable is now on the PATH.
# When running in a containerized environment (like Kubernetes), you may need to add
# specific flags to mpirun for compatibility, e.g.:
# mpirun --mca btl_vader_single_copy_mechanism none -np <cores> lmp -in <input_script>
# Setting a default command to the lammps executable.
CMD ["lmp"]
