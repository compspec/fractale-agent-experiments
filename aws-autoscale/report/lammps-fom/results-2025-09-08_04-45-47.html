# Base Image: Ubuntu 22.04 LTS for a modern, stable environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies for LAMMPS, MPI, and certificates
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    fftw3-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest branch of the LAMMPS source code
RUN git clone https://github.com/lammps/lammps.git /opt/lammps

# Configure, build, and install LAMMPS with the REAXFF package enabled
# This is done in a single RUN layer to optimize image size
RUN cd /opt/lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on && \
    make -j"$(nproc)" && \
    # Copy the main LAMMPS executable to a system location on the PATH
    cp src/lmp /usr/local/bin/

# Set the working directory for the container
WORKDIR /app

# Copy the example files for the ReaxFF HNS simulation into the working directory
RUN cp /opt/lammps/examples/reaxff/HNS/* /app/

# Clean up the build source to reduce the final image size
RUN rm -rf /opt/lammps

# Configure OpenMPI for containerized environments like Kubernetes
# This prevents MPI from trying to use loopback or docker interfaces
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Set the default command to run the HNS example simulation
# The user can override this to run their own input scripts
ENTRYPOINT ["mpirun", "--oversubscribe"]
CMD ["lmp", "-in", "in.hns"]
