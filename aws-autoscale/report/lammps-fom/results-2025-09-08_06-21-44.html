# Base Image: Ubuntu 22.04 LTS for a stable, recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies for building LAMMPS and running it
# Includes build tools, Git, OpenMPI for parallel execution,
# ca-certificates for secure connections, and libgomp1 for OpenMP support.
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    gfortran \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    ca-certificates \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments like Kubernetes.
# - Allow oversubscribing resources, common in K8s.
# - Use TCP for communication, disabling InfiniBand which is typically not available.
# - Set a specific BTL (Byte Transfer Layer) to avoid issues in some container runtimes.
ENV OMPI_MCA_rmaps_base_mapping_policy=oversubscribe
ENV OMPI_MCA_btl=tcp,self,vader
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Define the working directory for the application examples
WORKDIR /app

# Build LAMMPS from source
# This single RUN command clones, configures, builds, and installs LAMMPS,
# then copies the example files and cleans up to reduce layer size.
RUN set -e; \
    # Create a temporary directory for the build process
    BUILD_DIR=$(mktemp -d); \
    cd "${BUILD_DIR}"; \
    \
    # Clone the latest stable branch of the LAMMPS repository
    git clone --depth 1 https://github.com/lammps/lammps.git; \
    cd lammps; \
    \
    # Create a build directory and configure the build with CMake
    # We enable the REAXFF package for ReaxFF potential and OpenMP for shared-memory parallelism
    mkdir build && cd build; \
    cmake ../cmake \
          -D CMAKE_BUILD_TYPE=Release \
          -D BUILD_OMP=on \
          -D PKG_REAXFF=on; \
    \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc); \
    \
    # Copy the main LAMMPS executable to a system-wide location to be on the PATH
    cp lmp /usr/local/bin/lmp; \
    \
    # Copy the specific example files requested into the final WORKDIR
    cp "${BUILD_DIR}/lammps/examples/reaxff/HNS/"* /app/; \
    \
    # Clean up the build directory to keep the final image smaller
    cd /; \
    rm -rf "${BUILD_DIR}";

# Set the entrypoint to the LAMMPS executable.
# Users can override this with commands like `mpirun -np 4 lmp -in in.hns`
ENTRYPOINT ["lmp"]

# Default command, prints LAMMPS help info if no other command is provided
CMD ["-h"]
