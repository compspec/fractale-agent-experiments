# Specifies the base image for the build. Ubuntu 22.04 is a stable and widely-used base.
FROM ubuntu:22.04

# Install build dependencies, MPI, LAMMPS prerequisites, and certificates.
# A single RUN layer is used to reduce image size by minimizing layers.
# DEBIAN_FRONTEND=noninteractive prevents interactive prompts during installation.
# --no-install-recommends avoids installing unnecessary packages.
# The apt cache is cleaned at the end to further reduce image size.
RUN apt-get update && \
    DEBIAN_FRONTEND=noninteractive apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments.
# This allows MPI processes to run as the root user, a common requirement in containers.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# This single RUN layer handles the entire process of downloading, building, and installing LAMMPS.
# It also prepares the final working directory and cleans up to minimize image size.
RUN export LAMMPS_SRC_DIR=/opt/lammps-src && \
    # Clone the latest branch of the LAMMPS source code from GitHub.
    # --depth 1 creates a shallow clone, which is faster and smaller than a full history clone.
    git clone --depth 1 https://github.com/lammps/lammps.git ${LAMMPS_SRC_DIR} && \
    \
    # Create a dedicated directory for the build process.
    mkdir ${LAMMPS_SRC_DIR}/build && \
    cd ${LAMMPS_SRC_DIR}/build && \
    \
    # Configure the LAMMPS build using CMake.
    # -D BUILD_MPI=yes: Enables MPI support, crucial for parallel HPC applications.
    # -D PKG_REAXFF=on: Enables the ReaxFF package as required.
    cmake \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=on \
      ../cmake && \
    \
    # Compile the LAMMPS source code using all available processor cores for a faster build.
    make -j$(nproc) && \
    \
    # Copy the compiled LAMMPS executable 'lmp' to a directory in the system's PATH.
    # This makes the executable directly available from any location within the container.
    cp lmp /usr/local/bin/lmp && \
    \
    # Create the final working directory for running simulations.
    mkdir /data && \
    \
    # Copy the required example files for the ReaxFF HNS simulation into the working directory.
    cp ${LAMMPS_SRC_DIR}/examples/reaxff/HNS/* /data/ && \
    \
    # Clean up the source code directory to significantly reduce the final image size.
    rm -rf ${LAMMPS_SRC_DIR}

# Set the working directory to /data.
# This is where the example files are located and where simulations will be run by default.
WORKDIR /data

# Sets a default command to start an interactive bash shell when the container is run.
# A typical LAMMPS command would be: mpirun -np <cores> lmp -in <input_script>
CMD ["/bin/bash"]
