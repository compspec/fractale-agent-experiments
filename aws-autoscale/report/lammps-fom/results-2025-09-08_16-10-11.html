# Base Image: Ubuntu 22.04 LTS for a stable, modern environment.
FROM ubuntu:22.04

# This prevents debconf from asking interactive questions during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary dependencies in a single layer to optimize image size.
# This includes build tools, version control, MPI, required libraries for LAMMPS,
# certificates for secure connections, and libgomp1 as requested.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    ca-certificates \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for running in containerized environments like Kubernetes.
# These settings allow running MPI applications as the root user and ensure
# that TCP is used for communication between pods, which is standard in K8s.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_rmaps_base_oversubscribe=1

# Clone the latest branch of the LAMMPS source code from the official repository.
# Build LAMMPS from source using CMake, enabling the required REAXFF, MPI, and OMP packages.
# Finally, copy the main LAMMPS executable 'lmp' to a system-wide location on the PATH.
RUN git clone https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D BUILD_OMP=yes \
      -D PKG_REAXFF=yes && \
    make -j$(nproc) && \
    # The 'install' command copies the executable and sets correct permissions.
    install -m 755 lmp /usr/local/bin/

# Set the final working directory for the container.
WORKDIR /app

# As requested, copy the specific example files from the cloned source
# into the final working directory.
RUN cp /lammps/examples/reaxff/HNS/* .

# Set a default command to provide an interactive shell.
CMD ["/bin/bash"]
