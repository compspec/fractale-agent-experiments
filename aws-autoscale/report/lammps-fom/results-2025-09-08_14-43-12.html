# Base image: Ubuntu 22.04 LTS for a stable, well-supported foundation
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments like Kubernetes
# - Disable OpenIB BTL to avoid warnings/errors in environments without InfiniBand.
# - Force TCP communication over the primary container network interface (eth0).
# - Address potential shared memory issues within containers.
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Install dependencies, clone, build, and clean up in a single layer to optimize image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    wget \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    gfortran && \
    #
    # Clone the latest branch of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps-src && \
    cd /opt/lammps-src && \
    #
    # Configure the build with CMake, enabling the REAXFF package and MPI support
    mkdir build && cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on && \
    #
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) && \
    #
    # Copy the main LAMMPS executable to a system-wide location on the PATH
    cp lmp /usr/local/bin/lmp && \
    #
    # Create the application working directory
    mkdir -p /app && \
    #
    # Copy the required example files into the working directory
    cp /opt/lammps-src/examples/reaxff/HNS/* /app/ && \
    #
    # Clean up to reduce final image size
    # Remove source code and build artifacts
    rm -rf /opt/lammps-src && \
    #
    # Uninstall build-time dependencies
    apt-get purge -y --auto-remove \
        build-essential \
        cmake \
        git \
        wget \
        libopenmpi-dev \
        libfftw3-dev \
        gfortran && \
    #
    # Clear apt cache
    rm -rf /var/lib/apt/lists/*

# Set the default working directory for the container
WORKDIR /app
