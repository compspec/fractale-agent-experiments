# Base Image: Ubuntu 22.04 LTS for a modern, stable environment
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments.
# These settings are crucial for running MPI applications inside containers,
# especially in environments like Kubernetes.
# Allow running MPI processes as the root user.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Disable searching for high-speed interconnects like InfiniBand which are not present.
ENV OMPI_MCA_btl="^openib"
# Use the OB1 PML (Point-to-Point Messaging Layer), a reliable default.
ENV OMPI_MCA_pml="ob1"
# This setting can prevent hangs related to shared memory transport (BTL vader) in some container runtimes.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Install dependencies, clone, build LAMMPS, and clean up in a single RUN command
# This minimizes the number of layers in the final Docker image.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    gfortran \
    git \
    ca-certificates \
    libopenmpi-dev \
    openmpi-bin \
    libfftw3-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/* \
    # Clone the latest stable version of the LAMMPS source code
    && git clone --depth 1 -b stable https://github.com/lammps/lammps.git /opt/lammps-src \
    && cd /opt/lammps-src \
    && mkdir build && cd build \
    # Configure the LAMMPS build with CMake
    # Enable MPI, REAXFF, and other common packages for general-purpose simulations
    && cmake ../cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=yes \
      -D PKG_MOLECULE=yes \
      -D PKG_KSPACE=yes \
    # Compile LAMMPS using all available processor cores
    && make -j$(nproc) \
    # Copy the compiled LAMMPS executable to a standard location in the system PATH
    && cp ./lmp /usr/local/bin/lmp \
    # Create the final working directory
    && mkdir -p /data \
    # Copy the HNS reaxff example files into the working directory
    && cp /opt/lammps-src/examples/reaxff/HNS/* /data/ \
    # Clean up the source code to reduce the final image size
    && rm -rf /opt/lammps-src

# Set the final working directory for the container
WORKDIR /data

# This MiniCluster manifest is corrected to prevent Out-of-Memory (OOM) errors
# by selecting an AWS instance type with more memory.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  name: lammps-reaxff-hns-experiment
  namespace: default
spec:
  # The job will run within a single pod.
  size: 1

  # The number of tasks (e.g., MPI ranks) is set to 64 for a large-scale run.
  tasks: 64

  # Defines the application container.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
      # The command is configured for the 64-task simulation (8x8 grid).
      command: lmp -v x 8 -v y 8 -v z 8 -in ./in.reaxff.hns -nocite
      # This container is a worker, not the launcher.
      launcher: false

  # Custom flux container configuration.
  flux:
    container:
      # Use a flux-view image compatible with the ARM architecture of the Graviton instance.
      image: ghcr.io/converged-computing/flux-view-rocky:arm-9

  # Pod-level specifications.
  pod:
    # A nodeSelector is used to schedule the pod onto a memory-optimized instance.
    # The m7g.16xlarge instance provides 256 GiB of memory for the 64 CPUs,
    # addressing the OOM issues seen on instances with less RAM.
    nodeSelector:
      node.kubernetes.io/instance-type: m7g.16xlarge

  # Logging settings for the Flux Operator.
  logging:
    strict: true

  # When set to false, the MiniCluster resources are not deleted after job completion,
  # allowing for post-mortem analysis.
  cleanup: false
