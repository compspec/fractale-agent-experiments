# Base Image: Ubuntu 22.04 LTS for a stable, recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, FFTW, certificates, and other required libraries
# - build-essential: Compiler toolchain (gcc, g++, make)
# - gfortran: Fortran compiler, required by some LAMMPS packages
# - cmake: Build system generator for LAMMPS
# - git: For cloning the source code from the repository
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel execution
# - libfftw3-dev: FFTW3 library for KSPACE package (common dependency for performance)
# - ca-certificates: For secure network communications (e.g., git clone)
# - libgomp1: GNU OpenMP library, often a runtime dependency
# Clean up apt cache to reduce image size
RUN apt-get update && apt-get install -y \
    build-essential \
    gfortran \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    ca-certificates \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Set a working directory for the build process
WORKDIR /opt

# Clone the latest stable version of LAMMPS, build it, and clean up
# This is done in a single RUN command to reduce image layers and size
RUN git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake
    # - PKG_REAXFF=yes: Enables the ReaxFF package as requested
    # - BUILD_MPI=yes: Enables MPI for parallel execution
    # - PKG_KSPACE=yes: A common package that uses FFTW, good for general use
    cmake ../cmake \
        -D PKG_REAXFF=yes \
        -D BUILD_MPI=yes \
        -D PKG_KSPACE=yes \
        -D BUILD_OMP=yes && \
    # Build LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the main LAMMPS executable to a directory on the system's PATH
    cp lmp /usr/local/bin/lmp && \
    # Go back to the root of the cloned repo to copy examples
    cd /opt/lammps && \
    # Create the final application directory and copy example files into it
    mkdir -p /app && \
    cp examples/reaxff/HNS/* /app/ && \
    # Clean up the source code to minimize final image size
    cd / && \
    rm -rf /opt/lammps

# Set the final working directory for the container
WORKDIR /app

# Set a default command to provide a shell, allowing users to run commands interactively
# or override with their own mpirun command.
CMD ["/bin/bash"]
