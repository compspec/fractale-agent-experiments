# Dockerfile for lammps-reax on AWS CPU
# Base Image: Ubuntu 22.04 LTS for a modern, stable environment
FROM ubuntu:22.04

# Set non-interactive frontend for package management
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies for building LAMMPS
# - build-essential, cmake, git: For compiling the source code
# - openmpi-bin, libopenmpi-dev: For MPI support, crucial for HPC
# - libfftw3-dev: FFTW library, a common dependency for LAMMPS KSPACE package
# - ca-certificates: For secure network communications (e.g., git clone)
# - libgomp1: For OpenMP support, as requested
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    ca-certificates \
    libgomp1 && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest version of the LAMMPS source code
RUN git clone https://github.com/lammps/lammps.git /tmp/lammps-src

# Configure, build, and install LAMMPS
# We create a build directory as is standard practice with CMake
# -D BUILD_MPI=on: Enable MPI support
# -D PKG_KSPACE=on: Enable the KSPACE package for FFT-based long-range electrostatics
# -D PKG_REAXFF=on: Enable the ReaxFF package, the core requirement
# The executable 'lmp' is then copied to a location on the system PATH
RUN mkdir /tmp/lammps-src/build && \
    cd /tmp/lammps-src/build && \
    cmake \
        -D BUILD_MPI=on \
        -D PKG_KSPACE=on \
        -D PKG_REAXFF=on \
        ../cmake && \
    make -j$(nproc) && \
    cp /tmp/lammps-src/build/lmp /usr/local/bin/

# Set the working directory for running simulations
WORKDIR /app

# Copy the specific ReaxFF example files into the working directory
RUN cp -r /tmp/lammps-src/examples/reaxff/HNS/* .

# Clean up the build directory to reduce image size
RUN rm -rf /tmp/lammps-src

# The container is now ready. The command to run simulations will be provided
# at runtime, typically via a Kubernetes Job specification.
# Example: mpirun -np 4 lmp -in in.hns
