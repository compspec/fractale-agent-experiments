# Use a stable, common base image suitable for HPC applications
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build essentials, dependencies for LAMMPS, MPI, and required certificates in a single layer
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    # Clean up apt cache to reduce image size
    && rm -rf /var/lib/apt/lists/*

# Set a temporary working directory for the source code and build process
WORKDIR /tmp/build

# Clone the latest branch of LAMMPS source code from GitHub.
# --depth 1 creates a shallow clone, which is faster and smaller.
RUN git clone --depth 1 https://github.com/lammps/lammps.git .

# Build LAMMPS from source using CMake
# This process is combined into a single RUN command to optimize layer usage
RUN mkdir cmake-build && cd cmake-build && \
    # Configure the build for a release version with MPI and the REAXFF package
    cmake ../cmake \
    -D CMAKE_BUILD_TYPE=Release \
    -D BUILD_MPI=yes \
    -D PKG_REAXFF=yes && \
    # Compile the code using all available processor cores
    make -j$(nproc) && \
    # Copy the main LAMMPS executable 'lmp' to a directory on the system's PATH
    cp lmp /usr/local/bin/

# Configure OpenMPI for containerized environments
# These settings improve compatibility when running as root and on container networks
RUN echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "orte_base_help_aggregate = 0" >> /etc/openmpi/openmpi-mca-params.conf

# Create the final application directory and set it as the WORKDIR
WORKDIR /app

# Copy the example input files for the ReaxFF HNS benchmark from the source tree
RUN cp /tmp/build/examples/reaxff/HNS/* .

# Clean up the build directory to minimize the final image size
RUN rm -rf /tmp/build
