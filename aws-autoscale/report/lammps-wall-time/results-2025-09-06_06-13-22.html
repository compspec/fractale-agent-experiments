# Dockerfile for lammps-reax
# Target Environment: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for broad compatibility and support.
FROM ubuntu:22.04

# Set non-interactive frontend for package management
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments like Kubernetes.
# This allows running as root and forces TCP communication, which is common
# in container orchestrators where shared memory or high-speed interconnects
# are not available between pods.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib,sm
ENV OMPI_MCA_orte_base_help_aggregate=0

# Install build dependencies, runtime dependencies, and OpenMPI
# This is done in a single RUN layer to reduce image size.
# build-essential, cmake, git are for compiling LAMMPS.
# openmpi-bin, libopenmpi-dev are for MPI support.
# ca-certificates is for secure network communication (e.g., git clone).
# libgomp1 is a required runtime library for OpenMP.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest stable version of the LAMMPS source code
RUN git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps

# Set the working directory for the build process
WORKDIR /opt/lammps/build

# Configure and build LAMMPS using CMake
# -D PKG_REAXFF=on: Enables the REAXFF package as requested for 'lammps-reax'.
# -D BUILD_MPI=yes: Explicitly enables MPI support.
# make -j$(nproc): Compiles the code using all available CPU cores.
RUN cmake -D PKG_REAXFF=on -D BUILD_MPI=yes ../cmake && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable to a location on the system's PATH
# This makes 'lmp' directly available to users and scripts.
RUN cp /opt/lammps/build/lmp /usr/local/bin/

# Set the final working directory for the container
WORKDIR /app

# Copy the example files for the HNS simulation into the working directory
RUN cp /opt/lammps/examples/reaxff/HNS/* .

# Clean up build dependencies and source code to reduce the final image size.
# This is a crucial step for a production-ready image.
RUN apt-get purge -y --auto-remove \
    build-essential \
    cmake \
    git \
    wget \
    libopenmpi-dev && \
    rm -rf /opt/lammps && \
    rm -rf /var/lib/apt/lists/*

# Set a default command to start an interactive shell
CMD ["/bin/bash"]
