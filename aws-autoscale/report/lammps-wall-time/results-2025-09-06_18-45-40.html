# Base Image: Ubuntu 22.04 LTS for a stable and widely supported environment.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installations.
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary dependencies in a single layer to optimize image size.
# This includes:
# - build-essential, cmake, git: Core tools for compiling from source.
# - ca-certificates: For secure git clone and package downloads.
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel execution in a containerized environment.
# - libfftw3-dev: A common and highly recommended dependency for LAMMPS performance (FFTs).
# - libgomp1: GNU OpenMP library, as requested.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 && \
    # Clean up apt cache to keep the final image lean.
    rm -rf /var/lib/apt/lists/*

# Set the primary working directory for the application.
WORKDIR /app

# This single RUN command encapsulates the entire build process to minimize image layers.
# It clones the source, builds the executable, copies it to the PATH,
# stages example files, and cleans up all build artifacts.
RUN \
    # Create a temporary directory for the build to isolate it from the final WORKDIR.
    mkdir -p /tmp/build-area && \
    cd /tmp/build-area && \
    \
    # Clone the latest branch of the LAMMPS source code.
    # --depth 1 creates a shallow clone, significantly reducing download size.
    git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    \
    # Create a dedicated build directory, which is the standard CMake practice.
    mkdir build && \
    cd build && \
    \
    # Configure the build using CMake.
    # - BUILD_MPI=on: Enables MPI support, critical for HPC applications.
    # - PKG_REAXFF=on: Enables the ReaxFF package as required for 'lammps-reax'.
    # - Other PKG flags enable common and useful packages for general molecular dynamics.
    cmake ../cmake \
        -D BUILD_MPI=on \
        -D PKG_REAXFF=on \
        -D PKG_KSPACE=on \
        -D PKG_MOLECULE=on \
        -D PKG_RIGID=on && \
    \
    # Compile LAMMPS using all available processor cores for a faster build.
    make -j$(nproc) && \
    \
    # Copy the compiled 'lmp' executable to a system location on the PATH.
    cp ./lmp /usr/local/bin/lmp && \
    \
    # Copy the specified example input files to the final working directory.
    cp ../examples/reaxff/HNS/* /app/ && \
    \
    # Return to the root directory and remove the entire build area to minimize final image size.
    cd / && \
    rm -rf /tmp/build-area

# Set the default command to a shell. This allows users to easily execute
# LAMMPS with their desired mpirun arguments.
# Example usage: docker run <image_name> mpirun -np 4 lmp -in in.hns
CMD ["/bin/bash"]
