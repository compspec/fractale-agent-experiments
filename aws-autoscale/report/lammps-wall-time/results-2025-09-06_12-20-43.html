# Use a common base image like Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build-time dependencies, runtime dependencies, MPI, and certificates
# This includes build-essential, cmake, git for building,
# and openmpi, fftw, libgomp1 for running LAMMPS with ReaxFF.
# ca-certificates is included for secure network communications.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    gfortran \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    ca-certificates && \
    # Clean up apt cache to reduce image size
    rm -rf /var/lib/apt/lists/*

# Define a temporary build directory
WORKDIR /build

# Clone the latest development branch of LAMMPS directly into the container
# This is done in a RUN step to comply with the "NO COPY/ADD" instruction.
RUN git clone --depth 1 --branch develop https://github.com/lammps/lammps.git .

# Build LAMMPS from source using CMake
# Enable the required REAXFF package and MPI support.
# Since this is a single-stage build, the build artifacts are created here.
RUN mkdir build-lammps && \
    cd build-lammps && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=on \
        -D PKG_REAXFF=on && \
    make -j$(nproc) && \
    # Per instructions, copy the main 'lmp' executable to a system PATH location.
    # This avoids using 'make install' and satisfies the requirement to not symlink.
    cp ./lmp /usr/local/bin/lmp

# Set the final working directory for running the application
WORKDIR /app

# Copy the specified example files from the cloned source into the WORKDIR
# This is also done via a RUN command with 'cp' to meet constraints.
RUN cp /build/examples/reaxff/HNS/* .

# Clean up the build directory to reduce final image size
RUN rm -rf /build

# Configure OpenMPI for containerized environments like Kubernetes
# These settings prevent OpenMPI from trying to use shared memory mechanisms (vader)
# or high-performance interconnects (openib) that are not typically available,
# ensuring it falls back to TCP/IP for communication between pods.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl="^openib"
ENV OMPI_MCA_pml="ob1"

# Provide a default command to allow for interactive use of the container
CMD ["/bin/bash"]
