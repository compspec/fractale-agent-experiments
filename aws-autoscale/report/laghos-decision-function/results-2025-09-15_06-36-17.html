# Base image: Ubuntu 22.04
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments
# Disables InfiniBand/RoCE, which is typically not available in standard cloud CPU instances.
# Helps prevent hangs or errors when MPI tries to initialize hardware that isn't present.
# Sets process binding/mapping policies for better performance and compatibility with schedulers like Kubernetes.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_rmaps_base_mapping_policy=core \
    OMPI_MCA_hwloc_base_binding_policy=none \
    OMPI_MCA_btl=^openib,sm,self \
    OMPI_MCA_pml=ob1

# Install build dependencies, build all components, and clean up in a single layer
# This reduces the final image size and follows best practices.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gcc \
    gfortran \
    make \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/* \
    && mkdir /build \
    && cd /build \
    #
    # --- Build METIS 4.0.3 ---
    #
    && TPLS_DIR=/opt/tpls \
    && METIS_VER=4.0.3 \
    && METIS_URL="https://github.com/mfem/tpls/raw/gh-pages/metis-${METIS_VER}.tar.gz" \
    && echo "Building METIS ${METIS_VER}..." \
    && wget -O metis.tar.gz ${METIS_URL} \
    && tar -xzf metis.tar.gz \
    && cd metis-${METIS_VER} \
    && make cc=mpicc OPTFLAGS='-Wno-error=implicit-function-declaration' \
    && METIS_DIR=${TPLS_DIR}/metis-${METIS_VER} \
    && mkdir -p ${METIS_DIR}/lib ${METIS_DIR}/include \
    && cp libmetis.a ${METIS_DIR}/lib/ \
    && cp Lib/metis.h ${METIS_DIR}/include/ \
    && cd /build \
    #
    # --- Build HYPRE 2.28.0 ---
    #
    && HYPRE_VER=2.28.0 \
    && HYPRE_URL="https://github.com/hypre-space/hypre/archive/refs/tags/v${HYPRE_VER}.tar.gz" \
    && echo "Building HYPRE ${HYPRE_VER}..." \
    && wget -O hypre.tar.gz ${HYPRE_URL} \
    && tar -xzf hypre.tar.gz \
    && cd hypre-${HYPRE_VER}/src \
    && HYPRE_DIR=${TPLS_DIR}/hypre-${HYPRE_VER} \
    && ./configure --prefix=${HYPRE_DIR} --disable-fortran --enable-bigint CXX=mpicxx CC=mpicc \
    && make -j$(nproc) \
    && make install \
    && cd /build \
    #
    # --- Build MFEM 4.7 ---
    #
    && MFEM_VER=4.7 \
    && echo "Building MFEM ${MFEM_VER}..." \
    && git clone https://github.com/mfem/mfem.git \
    && cd mfem \
    && git checkout v${MFEM_VER} \
    && make parallel -j$(nproc) \
        MFEM_USE_MPI=YES \
        MFEM_USE_METIS=YES \
        METIS_DIR=${METIS_DIR} \
        HYPRE_DIR=${HYPRE_DIR} \
    && cd /build \
    #
    # --- Build Laghos ---
    #
    && echo "Building Laghos..." \
    && git clone https://github.com/CEED/Laghos.git \
    && cd Laghos \
    && make -j$(nproc) MFEM_DIR=/build/mfem \
    #
    # --- Finalize Image ---
    #
    # Copy main executable to a location on the system PATH
    && echo "Copying executables and data files..." \
    && cp laghos /usr/local/bin/ \
    # Create the final application directory and copy data files
    && mkdir /app \
    && cp ./data/* /app/ \
    #
    # --- Cleanup ---
    #
    && echo "Cleaning up build artifacts..." \
    && cd / \
    && rm -rf /build

# Set the working directory for the container
WORKDIR /app

# Set the default command to run the application
CMD ["laghos"]
