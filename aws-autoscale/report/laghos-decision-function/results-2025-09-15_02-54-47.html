# Dockerfile for Laghos HPC Application
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for a stable, widely-used foundation.
FROM ubuntu:22.04

# Set environment variables for the build and for MPI execution in containers.
# - DEBIAN_FRONTEND: Prevents interactive prompts during package installation.
# - OMPI_MCA...: Essential MPI tuning for containerized environments.
#   - btl_tcp_if_exclude: Prevents MPI from using loopback or docker-internal network interfaces.
#   - btl_vader_single_copy_mechanism: Avoids a mechanism that can cause issues in some shared memory environments.
#   - OMPI_ALLOW_RUN_AS_ROOT: Necessary as this container runs as the root user.
ENV DEBIAN_FRONTEND=noninteractive \
    OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl_tcp_if_exclude=lo,docker0 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build-essential packages, compilers, version control, MPI, and certificates.
# This is done in a single RUN command to reduce image layers.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    make \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# This single RUN command builds all dependencies and the Laghos application from source.
# This approach minimizes the number of Docker image layers.
RUN mkdir -p /build && cd /build && \
    \
    # --- Build METIS 4.0.3 ---
    # METIS is a graph partitioning library, required by MFEM.
    # The OPTFLAGS variable is set to work around a compiler error with modern GCC versions.
    wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS="-Wno-error=implicit-function-declaration" && \
    # METIS 4.x does not have a 'make install', so we copy the library and header manually to a system location.
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/ && \
    cd /build && \
    \
    # --- Build HYPRE 2.28.0 ---
    # HYPRE is a library of high-performance preconditioners and solvers, required by MFEM.
    # We use --enable-bigint for 64-bit integer support and disable Fortran as it's not needed by Laghos.
    wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --disable-fortran --enable-bigint && \
    make -j$(nproc) && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build MFEM 4.7 ---
    # MFEM is the finite element methods library that Laghos is built upon.
    # We build the parallel (MPI) version and link it against our custom HYPRE and METIS builds.
    git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git && \
    cd mfem && \
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES HYPRE_DIR=/usr/local && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build Laghos ---
    # Finally, we build the Laghos application itself.
    git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    # We explicitly point to the installed MFEM config and use the MPI C++ compiler wrapper.
    make -j$(nproc) MFEM_DIR=/usr/local/share/mfem CXX=mpicxx && \
    # Copy the final executable to a system location on the PATH for easy access.
    cp laghos /usr/local/bin/ && \
    \
    # --- Finalize Image ---
    # Create the final application directory and copy the required data files into it.
    mkdir -p /app && \
    cp -R data/* /app/ && \
    # Clean up the build directory to reduce the final image size.
    rm -rf /build

# Set the final working directory for the container.
WORKDIR /app

# Set a default command to run when the container starts, showing the application's help message.
CMD ["laghos", "--help"]

# Dockerfile for Laghos HPC Application
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for a stable, widely-used foundation.
FROM ubuntu:22.04

# Set environment variables for the build and for MPI execution in containers.
# - DEBIAN_FRONTEND: Prevents interactive prompts during package installation.
# - OMPI_MCA...: Essential MPI tuning for containerized environments.
#   - btl_tcp_if_exclude: Prevents MPI from using loopback or docker-internal network interfaces.
#   - btl_vader_single_copy_mechanism: Avoids a mechanism that can cause issues in some shared memory environments.
#   - OMPI_ALLOW_RUN_AS_ROOT: Necessary as this container runs as the root user.
ENV DEBIAN_FRONTEND=noninteractive \
    OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl_tcp_if_exclude=lo,docker0 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build-essential packages, compilers, version control, MPI, and certificates.
# This is done in a single RUN command to reduce image layers.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    make \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# This single RUN command builds all dependencies and the Laghos application from source.
# This approach minimizes the number of Docker image layers.
RUN mkdir -p /build && cd /build && \
    \
    # --- Build METIS 4.0.3 ---
    # METIS is a graph partitioning library, required by MFEM.
    # The OPTFLAGS variable is set to work around a compiler error with modern GCC versions.
    wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS="-Wno-error=implicit-function-declaration" && \
    # METIS 4.x does not have a 'make install', so we copy the library and header manually to a system location.
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/ && \
    cd /build && \
    \
    # --- Build HYPRE 2.28.0 ---
    # HYPRE is a library of high-performance preconditioners and solvers, required by MFEM.
    # We use --enable-bigint for 64-bit integer support and disable Fortran as it's not needed by Laghos.
    # FIX: Add --prefix=/usr/local to ensure files are installed where MFEM's build expects them.
    wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --prefix=/usr/local --disable-fortran --enable-bigint && \
    make -j$(nproc) && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build MFEM 4.7 ---
    # MFEM is the finite element methods library that Laghos is built upon.
    # We build the parallel (MPI) version and link it against our custom HYPRE and METIS builds.
    git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git && \
    cd mfem && \
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES HYPRE_DIR=/usr/local && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build Laghos ---
    # Finally, we build the Laghos application itself.
    git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    # We explicitly point to the installed MFEM config and use the MPI C++ compiler wrapper.
    make -j$(nproc) MFEM_DIR=/usr/local/share/mfem CXX=mpicxx && \
    # Copy the final executable to a system location on the PATH for easy access.
    cp laghos /usr/local/bin/ && \
    \
    # --- Finalize Image ---
    # Create the final application directory and copy the required data files into it.
    mkdir -p /app && \
    cp -R data/* /app/ && \
    # Clean up the build directory to reduce the final image size.
    rm -rf /build

# Set the final working directory for the container.
WORKDIR /app

# Set a default command to run when the container starts, showing the application's help message.
CMD ["laghos", "--help"]

# Dockerfile for Laghos HPC Application
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for a stable, widely-used foundation.
FROM ubuntu:22.04

# Set environment variables for the build and for MPI execution in containers.
# - DEBIAN_FRONTEND: Prevents interactive prompts during package installation.
# - OMPI_MCA...: Essential MPI tuning for containerized environments.
#   - btl_tcp_if_exclude: Prevents MPI from using loopback or docker-internal network interfaces.
#   - btl_vader_single_copy_mechanism: Avoids a mechanism that can cause issues in some shared memory environments.
#   - OMPI_ALLOW_RUN_AS_ROOT: Necessary as this container runs as the root user.
ENV DEBIAN_FRONTEND=noninteractive \
    OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl_tcp_if_exclude=lo,docker0 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build-essential packages, compilers, version control, MPI, and certificates.
# This is done in a single RUN command to reduce image layers.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    make \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# This single RUN command builds all dependencies and the Laghos application from source.
# This approach minimizes the number of Docker image layers.
RUN mkdir -p /build && cd /build && \
    \
    # --- Build METIS 4.0.3 ---
    # METIS is a graph partitioning library, required by MFEM.
    # The OPTFLAGS variable is set to work around a compiler error with modern GCC versions.
    wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS="-Wno-error=implicit-function-declaration" && \
    # METIS 4.x does not have a 'make install', so we copy the library and header manually to a system location.
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/ && \
    cd /build && \
    \
    # --- Build HYPRE 2.28.0 ---
    # HYPRE is a library of high-performance preconditioners and solvers, required by MFEM.
    # We use --enable-bigint for 64-bit integer support and disable Fortran as it's not needed by Laghos.
    wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --prefix=/usr/local --disable-fortran --enable-bigint && \
    make -j$(nproc) && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build MFEM 4.7 ---
    # MFEM is the finite element methods library that Laghos is built upon.
    # We build the parallel (MPI) version and link it against our custom HYPRE and METIS builds.
    # FIX: Use PREFIX (uppercase) for the install directory variable, which is what the MFEM Makefile expects.
    git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git && \
    cd mfem && \
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES HYPRE_DIR=/usr/local && \
    make install PREFIX=/usr/local && \
    ldconfig && \
    cd /build && \
    \
    # --- Build Laghos ---
    # Finally, we build the Laghos application itself.
    git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    # We explicitly point to the installed MFEM config and use the MPI C++ compiler wrapper.
    make -j$(nproc) MFEM_DIR=/usr/local/share/mfem CXX=mpicxx && \
    # Copy the final executable to a system location on the PATH for easy access.
    cp laghos /usr/local/bin/ && \
    \
    # --- Finalize Image ---
    # Create the final application directory and copy the required data files into it.
    mkdir -p /app && \
    cp -R data/* /app/ && \
    # Clean up the build directory to reduce the final image size.
    rm -rf /build

# Set the final working directory for the container.
WORKDIR /app

# Set a default command to run when the container starts, showing the application's help message.
CMD ["laghos", "--help"]

# Dockerfile for Laghos HPC Application
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for a stable, widely-used foundation.
FROM ubuntu:22.04

# Set environment variables for the build and for MPI execution in containers.
# - DEBIAN_FRONTEND: Prevents interactive prompts during package installation.
# - OMPI_MCA...: Essential MPI tuning for containerized environments.
#   - btl_tcp_if_exclude: Prevents MPI from using loopback or docker-internal network interfaces.
#   - btl_vader_single_copy_mechanism: Avoids a mechanism that can cause issues in some shared memory environments.
#   - OMPI_ALLOW_RUN_AS_ROOT: Necessary as this container runs as the root user.
ENV DEBIAN_FRONTEND=noninteractive \
    OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl_tcp_if_exclude=lo,docker0 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build-essential packages, compilers, version control, MPI, and certificates.
# This is done in a single RUN command to reduce image layers.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    make \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# This single RUN command builds all dependencies and the Laghos application from source.
# This approach minimizes the number of Docker image layers.
RUN mkdir -p /build && cd /build && \
    \
    # --- Build METIS 4.0.3 ---
    # METIS is a graph partitioning library, required by MFEM.
    # The OPTFLAGS variable is set to work around a compiler error with modern GCC versions.
    wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS="-Wno-error=implicit-function-declaration" && \
    # METIS 4.x does not have a 'make install', so we copy the library and header manually to a system location.
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/ && \
    cd /build && \
    \
    # --- Build HYPRE 2.28.0 ---
    # HYPRE is a library of high-performance preconditioners and solvers, required by MFEM.
    # We use --enable-bigint for 64-bit integer support and disable Fortran as it's not needed by Laghos.
    wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --prefix=/usr/local --disable-fortran --enable-bigint && \
    make -j$(nproc) && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build MFEM 4.7 ---
    # MFEM is the finite element methods library that Laghos is built upon.
    # We build the parallel (MPI) version and link it against our custom HYPRE and METIS builds.
    # FIX: Combine 'parallel' and 'install' targets in a single 'make' command. This ensures that
    # build-time variables (like MFEM_USE_MPI) are applied to both build and install steps,
    # preventing an incomplete installation of the parallel library.
    git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git && \
    cd mfem && \
    make parallel install -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES HYPRE_DIR=/usr/local PREFIX=/usr/local && \
    ldconfig && \
    cd /build && \
    \
    # --- Build Laghos ---
    # Finally, we build the Laghos application itself.
    git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    # We explicitly point to the installed MFEM config and use the MPI C++ compiler wrapper.
    make -j$(nproc) MFEM_DIR=/usr/local/share/mfem CXX=mpicxx && \
    # Copy the final executable to a system location on the PATH for easy access.
    cp laghos /usr/local/bin/ && \
    \
    # --- Finalize Image ---
    # Create the final application directory and copy the required data files into it.
    mkdir -p /app && \
    cp -R data/* /app/ && \
    # Clean up the build directory to reduce the final image size.
    rm -rf /build

# Set the final working directory for the container.
WORKDIR /app

# Set a default command to run when the container starts, showing the application's help message.
CMD ["laghos", "--help"]

# Set environment variables for the build and for MPI execution in containers.
# - DEBIAN_FRONTEND: Prevents interactive prompts during package installation.
# - OMPI_MCA...: Essential MPI tuning for containerized environments.
#   - btl_tcp_if_exclude: Prevents MPI from using loopback or docker-internal network interfaces.
#   - btl_vader_single_copy_mechanism: Avoids a mechanism that can cause issues in some shared memory environments.
#   - OMPI_ALLOW_RUN_AS_ROOT: Necessary as this container runs as the root user.

    OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl_tcp_if_exclude=lo,docker0 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build-essential packages, compilers, version control, MPI, and certificates.
# This is done in a single RUN command to reduce image layers.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    make \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# This single RUN command builds all dependencies and the Laghos application from source.
# This approach minimizes the number of Docker image layers.
RUN mkdir -p /build && cd /build && \
    \
    # --- Build METIS 4.0.3 ---
    # METIS is a graph partitioning library, required by MFEM.
    # The OPTFLAGS variable is set to work around a compiler error with modern GCC versions.
    wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS="-Wno-error=implicit-function-declaration" && \
    # METIS 4.x does not have a 'make install', so we copy the library and header manually to a system location.
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/ && \
    cd /build && \
    \
    # --- Build HYPRE 2.28.0 ---
    # HYPRE is a library of high-performance preconditioners and solvers, required by MFEM.
    # We use --enable-bigint for 64-bit integer support and disable Fortran as it's not needed by Laghos.
    wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --prefix=/usr/local --disable-fortran --enable-bigint && \
    make -j$(nproc) && \
    make install && \
    ldconfig && \
    cd /build && \
    \
    # --- Build MFEM 4.7 ---
    # MFEM is the finite element methods library that Laghos is built upon.
    # We build the parallel (MPI) version and link it against our custom HYPRE and METIS builds.
    git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git && \
    cd mfem && \
    # FIX: Build MFEM in-place without installing system-wide. The Laghos makefile is designed to find
    # the MFEM library and config files within this build directory structure.
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES HYPRE_DIR=/usr/local && \
    cd /build && \
    \
    # --- Build Laghos ---
    # Finally, we build the Laghos application itself.
    git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    # FIX: Point MFEM_DIR directly to the MFEM build directory instead of a system install path.
    make -j$(nproc) MFEM_DIR=/build/mfem CXX=mpicxx && \
    # Copy the final executable to a system location on the PATH for easy access.
    cp laghos /usr/local/bin/ && \
    \
    # --- Finalize Image ---
    # Create the final application directory and copy the required data files into it.
    mkdir -p /app && \
    cp -R data/* /app/ && \
    # Clean up the build directory to reduce the final image size.
    rm -rf /build
