# Base image: Ubuntu 22.04 LTS for a stable, common environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, and certificates
# Using a single RUN statement for this layer to optimize image size
# ca-certificates is included for secure network connections (e.g., git, wget)
RUN apt-get update && apt-get install -y \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments
# This prevents MPI from trying to use loopback or docker-internal network interfaces
RUN mkdir -p /etc/openmpi && \
    echo "btl_tcp_if_exclude = lo,docker0" > /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi/openmpi-mca-params.conf

# Set a working directory for building all components
WORKDIR /build

# Build and install METIS 4.0.3 from source
# METIS does not have a standard 'make install' target, so we manually copy the library and header.
# The specified OPTFLAGS is used to prevent build failures with modern compilers.
RUN wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzvf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS=-Wno-error=implicit-function-declaration && \
    cp libmetis.a /usr/local/lib/ && \
    cp metis.h /usr/local/include/

# Build and install HYPRE 2.28.0 from source
# Required flags --disable-fortran and --enable-bigint are included as per instructions.
# Installation prefix is set to a standard system location.
RUN wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzvf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --disable-fortran --enable-bigint --prefix=/usr/local && \
    make -j$(nproc) && \
    make install

# Build and install MFEM 4.7 from source
# We build the parallel version which depends on MPI, HYPRE, and METIS.
# The build system will find the previously installed dependencies in /usr/local.
RUN git clone https://github.com/mfem/mfem.git && \
    cd mfem && \
    git checkout v4.7 && \
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES && \
    make install prefix=/usr/local

# Build the Laghos application from source
# The build links against the previously installed MFEM library.
RUN git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    make -j$(nproc) MFEM_DIR=/usr/local

# Copy the final laghos executable to a system path
# This makes the application directly available to any user or script in the container.
# ldconfig is run to refresh the shared library cache.
RUN cp /build/Laghos/laghos /usr/local/bin/ && \
    ldconfig

# Set the final working directory for the application
WORKDIR /app

# Copy the application data files into the final working directory
# and clean up the build directory to reduce final image size.
RUN cp /build/Laghos/data/* . && \
    rm -rf /build

# Base image: Ubuntu 22.04 LTS for a stable, common environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, and certificates
# Using a single RUN statement for this layer to optimize image size
# ca-certificates is included for secure network connections (e.g., git, wget)
RUN apt-get update && apt-get install -y \
    build-essential \
    g++ \
    gfortran \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments
# This prevents MPI from trying to use loopback or docker-internal network interfaces
RUN mkdir -p /etc/openmpi && \
    echo "btl_tcp_if_exclude = lo,docker0" > /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi/openmpi-mca-params.conf

# Set a working directory for building all components
WORKDIR /build

# Build and install METIS 4.0.3 from source
# METIS does not have a standard 'make install' target, so we manually copy the library and header.
# The specified OPTFLAGS is used to prevent build failures with modern compilers.
RUN wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz && \
    tar -xzvf metis-4.0.3.tar.gz && \
    cd metis-4.0.3 && \
    make OPTFLAGS=-Wno-error=implicit-function-declaration && \
    cp libmetis.a /usr/local/lib/ && \
    cp Lib/metis.h /usr/local/include/

# Build and install HYPRE 2.28.0 from source
# Required flags --disable-fortran and --enable-bigint are included as per instructions.
# Installation prefix is set to a standard system location.
RUN wget --no-check-certificate https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz -O hypre-2.28.0.tar.gz && \
    tar -xzvf hypre-2.28.0.tar.gz && \
    cd hypre-2.28.0/src && \
    ./configure --disable-fortran --enable-bigint --prefix=/usr/local && \
    make -j$(nproc) && \
    make install

# Build and install MFEM 4.7 from source
# We build the parallel version which depends on MPI, HYPRE, and METIS.
# The build system will find the previously installed dependencies in /usr/local.
RUN git clone https://github.com/mfem/mfem.git && \
    cd mfem && \
    git checkout v4.7 && \
    make parallel -j$(nproc) MFEM_USE_MPI=YES MFEM_USE_METIS=YES && \
    make install prefix=/usr/local

# Build the Laghos application from source
# The build links against the previously built MFEM library in /build/mfem,
# which is required by its makefile to find config files.
RUN git clone https://github.com/CEED/Laghos.git && \
    cd Laghos && \
    make -j$(nproc) MFEM_DIR=/build/mfem

# Copy the final laghos executable to a system path
# This makes the application directly available to any user or script in the container.
# ldconfig is run to refresh the shared library cache.
RUN cp /build/Laghos/laghos /usr/local/bin/ && \
    ldconfig

# Set the final working directory for the application
WORKDIR /app

# Copy the application data files into the final working directory
# and clean up the build directory to reduce final image size.
RUN cp /build/Laghos/data/* . && \
    rm -rf /build
