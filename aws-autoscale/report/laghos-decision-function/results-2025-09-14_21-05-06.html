# Base image for the build
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments, allowing root execution
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install dependencies, build applications, and clean up in a single layer
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    make \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    && mkdir -p /build && cd /build \
    #
    # --- Build and install METIS 4.0.3 ---
    #
    && wget -O metis-4.0.3.tar.gz https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz \
    && tar -xvzf metis-4.0.3.tar.gz \
    && cd metis-4.0.3 \
    && make lib OPTFLAGS='-Wno-error=implicit-function-declaration' \
    && cp libmetis.a /usr/local/lib/ \
    && cp Lib/metis.h /usr/local/include/ \
    && cd .. \
    #
    # --- Build and install HYPRE 2.28.0 ---
    #
    && wget -O hypre-2.28.0.tar.gz https://github.com/hypre-space/hypre/archive/refs/tags/v2.28.0.tar.gz \
    && tar -xvzf hypre-2.28.0.tar.gz \
    && cd hypre-2.28.0/src \
    && ./configure --prefix=/usr/local --disable-fortran --enable-bigint \
    && make -j$(nproc) \
    && make install \
    && cd ../.. \
    #
    # --- Build and install MFEM 4.7 ---
    #
    && git clone --depth 1 --branch v4.7 https://github.com/mfem/mfem.git \
    && cd mfem \
    && make parallel -j$(nproc) MFEM_USE_METIS_4=YES HYPRE_DIR=/usr/local MFEM_USE_MPI=YES CXX=mpic++ \
    && make install prefix=/usr/local \
    && cd .. \
    #
    # --- Build Laghos and install executable ---
    #
    && git clone https://github.com/CEED/Laghos.git \
    && cd Laghos \
    && make -j$(nproc) MFEM_DIR=/usr/local \
    && cp laghos /usr/local/bin/ \
    #
    # --- Set up final working directory and copy data files ---
    #
    && mkdir -p /app \
    && cp -r data/* /app/ \
    && cd / \
    #
    # --- Clean up build directory and apt cache ---
    #
    && rm -rf /build \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

# Set the final working directory
WORKDIR /app
