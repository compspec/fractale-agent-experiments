  name: amg-aws-arm-test

  size: 1
  # Minimum size for the MiniCluster.
  minSize: 1
  # Maximum size for the MiniCluster, allowing scaling up to 4 nodes.
  maxSize: 4

  tasks: 64

  # Pod spec additions to target specific AWS instances for the experiment.
  pod:

      node.kubernetes.io/instance-type: m7g.16xlarge

  # Logging is set to quiet mode as per the update.
  logging:
    quiet: true

  # Per instructions, cleanup of the job is disabled.
  cleanup: false

  # Flux Operator settings for ARM architecture, as required for AWS ARM instances.
  flux:
    arch: "arm"
    container:
      # This specific Flux container image is required for ARM64 architecture.
      image: ghcr.io/converged-computing/flux-view-rocky:arm-9

  # Defines the application container to be run by Flux.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:amg2023
      # The main container is not the launcher; Flux will launch the job. This MUST be false.
      launcher: false
      # Command updated with a reduced problem size to prevent startup failure.
      command: amg -problem 2 -n 32 32 32 -P 4 4 4

      # Environment variables to tune MPI and networking for EFA.
      environment:
        OMP_NUM_THREADS: "1"
        OMPI_MCA_btl_vader_single_copy_mechanism: cma
        OMPI_MCA_pml: cm
        OMPI_MCA_mtl: ofi
        FI_PROVIDER: efa

      # Resource requests and limits for AWS Elastic Fabric Adapter (EFA).
      resources:
        limits:
          vpc.amazonaws.com/efa: "1"
        requests:
          vpc.amazonaws.com/efa: "1"
