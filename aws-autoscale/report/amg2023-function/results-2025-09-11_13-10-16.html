FROM ubuntu:22.04

# Install essential build tools, git, Fortran compiler, OpenMPI, and other dependencies
# - build-essential: Includes make, gcc, g++, etc.
# - git: For cloning the source code repository.
# - gfortran: Required by many scientific applications.
# - openmpi-bin, libopenmpi-dev: OpenMPI runtime and development libraries for parallel processing.
# - ca-certificates: For secure network communications (e.g., git clone over https).
# - libgomp1: GNU OpenMP runtime library, a specific requirement.
# We clean up the apt cache in the same layer to reduce image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    git \
    gfortran \
    libgomp1 \
    libopenmpi-dev \
    make \
    openmpi-bin && \
    rm -rf /var/lib/apt/lists/*

# Set common OpenMPI environment variables for container compatibility
# These settings allow mpirun to execute as root, a common pattern in single-application containers.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Create a temporary directory for the build process
WORKDIR /tmp/build

# Clone the latest version of the AMG (amg2023) source code, build it,
# copy the executable to a system path, and clean up the source directory.
# This entire process is done in a single RUN command to minimize layer count.

    cd test && \
    make amg && \
    cp amg /usr/local/bin/amg && \
    rm -rf /tmp/build

# Set the default working directory for when the container starts.
# This is a good practice for where to mount volumes or place input files.
WORKDIR /data

# Set the default command for the container.
# This will execute the amg benchmark when the container is run without arguments.
CMD ["amg"]
