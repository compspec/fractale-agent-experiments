dockerfile
# Base Image: Ubuntu 22.04 LTS for a stable, well-supported environment
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, compilers, MPI, git, and required libraries
# ca-certificates is included for secure network operations (e.g., git clone)
# libgomp1 is included as specifically requested for OpenMP support
RUN apt-get update && apt-get install -y \
    build-essential \
    gfortran \
    make \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments like Kubernetes
# This is crucial for allowing MPI processes to run correctly as the root user.
# Disabling the unused component warning cleans up the output.
RUN echo "btl_base_warn_component_unused = 0" >> /etc/openmpi-mca-params.conf

# Set environment variables to explicitly allow Open MPI to run with root privileges,
# a common requirement in containerized HPC jobs.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Set a temporary working directory for the build process
WORKDIR /build

# Clone the latest branch of the AMG repository (referred to as amg2023).
# The build process is two-staged within the source tree:
# 1. Run `make` in the root to build the required 'parcsr' libraries first.
# 2. Run `make` in the 'test' directory to build the final 'amg' executable.
# Finally, copy the executable to a system path and clean up the build directory.
RUN git clone https://github.com/LLNL/AMG.git amg2023 \
    && cd amg2023 \
    && make \
    && cd test \
    && make \
    && cp amg /usr/local/bin/ \
    && cd / \
    && rm -rf /build

# Set the default entrypoint to an interactive shell for ease of use and debugging
CMD ["/bin/bash"]

# This MiniCluster manifest defines a single-node Flux job to run the AMG benchmark.
# It is configured for a small-scale test run and includes runtime fixes for known
# library dependencies based on previous experimental runs.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  # The name of our MiniCluster instance.
  name: amg2023-minicluster-cpu
  # The manifest will be applied to the 'default' namespace as none is specified.
spec:
  # The total number of pods in the MiniCluster (the job size).
  # We are running on a single node as required.
  size: 1

  # The total number of tasks to run, which corresponds to the number of MPI ranks.
  # This is set to 4, fulfilling the minimum requirement for the test.
  tasks: 4

  # Keep pods and logs after the job finishes for debugging purposes.
  cleanup: false

  # Logging settings for the Flux Operator.
  logging:
    # Strict mode ensures that any error will cause the job to fail.
    strict: true

  # The application container that will run the benchmark.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:amg2023
      # The launcher flag MUST be false for the application container.
      # The Flux Operator injects the 'flux run' command.
      launcher: false

      # Pre-execution commands to install dependencies not present in the base container.
      commands:
        # Install the libgomp1 shared library to prevent runtime errors.
        pre: "apt-get update && apt-get install -y --no-install-recommends libgomp1"

      # The command to execute. It runs AMG problem 2 with a small global
      # problem size (-n) and a process topology (-P) whose product (2*2*1)
      # matches the number of tasks.
      command: "amg -problem 2 -n 4 4 4 -P 2 2 1"

      # Environment variables for runtime configuration.
      environment:
        # Explicitly set the library path to ensure the linker finds the newly installed library.
        LD_LIBRARY_PATH: "/usr/lib/x86_64-linux-gnu"
        # Set one thread per MPI rank to prevent oversubscription of CPU cores.
        OMP_NUM_THREADS: "1"
