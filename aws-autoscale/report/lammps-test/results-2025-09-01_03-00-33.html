# Dockerfile for lammps-reax on CPU
# Base Image: Ubuntu 22.04 LTS for a stable, well-supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Define reusable environment variables for directories
ENV LAMMPS_SRC_DIR=/usr/src/lammps
ENV LAMMPS_INSTALL_DIR=/opt/lammps

# Configure OpenMPI for containerized environments like Kubernetes
# This prevents errors related to shared memory (vader) and high-speed interconnects (openib)
# and allows oversubscribing processes to cores, a common container pattern.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_rmaps_base_oversubscribe=1

# Install dependencies, clone, build, and install LAMMPS in a single RUN layer
# This approach helps to keep the Docker image size smaller.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates && \
    # Clone the latest branch of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git ${LAMMPS_SRC_DIR} && \
    # Create build directory and change into it
    mkdir -p ${LAMMPS_SRC_DIR}/build && \
    cd ${LAMMPS_SRC_DIR}/build && \
    # Configure the LAMMPS build with CMake
    # Enable the REAXFF package as required for this application
    cmake ../cmake \
        -D PKG_REAXFF=on \
        -D CMAKE_INSTALL_PREFIX=${LAMMPS_INSTALL_DIR} && \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Install the compiled binaries and libraries
    make install && \
    # Set the working directory for the final image
    mkdir -p /data && \
    # Copy the required example files into the working directory
    cp ${LAMMPS_SRC_DIR}/examples/reaxff/HNS/* /data/ && \
    # Clean up to minimize final image size by removing the source code
    rm -rf ${LAMMPS_SRC_DIR} && \
    # Clean up the apt cache to further reduce image size
    rm -rf /var/lib/apt/lists/*

# Add the LAMMPS installation directory to the system PATH
# This makes the 'lmp' executable available globally.
ENV PATH=${LAMMPS_INSTALL_DIR}/bin:${PATH}

# Set the default working directory for the container
WORKDIR /data

# Default command to start an interactive shell.
# The container can be run with `docker run -it <image> lmp -in <input_script>`
CMD ["/bin/bash"]
