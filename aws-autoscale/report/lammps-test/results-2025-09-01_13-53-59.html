Assets for build Agent
dockerfile
# Use a standard Ubuntu base image suitable for CPU workloads
FROM ubuntu:22.04

# Set non-interactive frontend for package managers to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments, allowing root execution.
# This is critical for compatibility with orchestrators like Kubernetes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# This helps prevent hangs in some container/cgroup environments by disabling a shared memory optimization
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# One RUN layer to install dependencies, clone, build, install, and clean up.
# This optimizes layer caching and reduces final image size in a single-stage build.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        ca-certificates \
        git \
        cmake \
        build-essential \
        openmpi-bin \
        libopenmpi-dev \
        python3 && \
    # Clone the latest version of LAMMPS source code. --depth 1 creates a shallow clone.
    git clone --depth 1 https://github.com/lammps/lammps.git /lammps_src && \
    # Create build directory and configure with CMake, enabling the required REAXFF package and MPI support.
    mkdir /lammps_src/build && \
    cd /lammps_src/build && \
    cmake ../cmake \
      -D PKG_REAXFF=on \
      -D BUILD_MPI=on && \
    # Build LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Install the built executables and libraries to a standard system path (/usr/local)
    make install && \
    # Create the application working directory
    mkdir /app && \
    # As per instructions, copy the example files into the working directory
    cp /lammps_src/examples/reaxff/HNS/* /app/ && \
    # Clean up to reduce image size: remove source code and package manager cache
    cd / && \
    rm -rf /lammps_src && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Add the standard installation path for compiled software to the system PATH.
# This ensures 'lmp' and other executables are discoverable without full paths.
ENV PATH="/usr/local/bin:${PATH}"

# Set the final working directory for the container.
WORKDIR /app

# Define a default command. This is useful for testing the image and shows the LAMMPS help message.
CMD ["lmp", "-h"]
