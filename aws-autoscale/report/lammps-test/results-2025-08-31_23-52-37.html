# Use a standard Ubuntu base image suitable for CPU workloads
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build-essential tools, CMake for building, Git for source control,
# OpenMPI for parallel execution, and ca-certificates for network security.
# Clean up apt cache in the same layer to reduce image size.
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments by excluding Docker's virtual interfaces.
# This is a critical step to ensure stable MPI communication inside containers.
RUN echo "btl_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf

# Set the primary working directory for cloning the source code
WORKDIR /opt

# Clone the latest version of the LAMMPS source code from the official repository
RUN git clone --depth 1 https://github.com/lammps/lammps.git

# Set the working directory to the cloned repository
WORKDIR /opt/lammps

# Create a build directory and build LAMMPS using CMake.
# Enable the REAXFF package and its common dependencies (KSPACE, MOLECULE).
# Build in Release mode for performance and use all available cores for a faster build.
RUN mkdir build && \
    cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=yes \
        -D PKG_KSPACE=yes \
        -D PKG_MOLECULE=yes && \
    make -j$(nproc)

# Add the LAMMPS executable directory to the system's PATH environment variable.
# This makes the 'lmp' executable available system-wide.
ENV PATH="/opt/lammps/build:${PATH}"

# Set the final WORKDIR for running simulations.
WORKDIR /opt/workspace

# Copy the specific example input files from the LAMMPS source tree into the workspace.
RUN cp /opt/lammps/examples/reaxff/HNS/* .

# MiniCluster manifest for a LAMMPS ReaxFF experiment on a single Kubernetes node.
# This manifest is designed for the Flux Operator (v1alpha2) and runs on AWS CPU instances.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  # A descriptive name for the MiniCluster instance.
  name: lammps-reaxff-hns
  # Deploying to the 'default' namespace as requested.
  namespace: default
spec:
  # Size specifies the number of pods in the MiniCluster.
  # Setting to 1 ensures the entire job runs on a single node.
  size: 1

  # Tasks specifies the total number of MPI ranks for the job.
  # The lmp command arguments (-v x 2 -v y 2 -v z 2) suggest a 2x2x2
  # process grid, which corresponds to 8 tasks.
  tasks: 8

  # Instructs the operator not to delete the job pods and services after completion.
  # This is useful for debugging and inspecting results.
  cleanup: false

  # Logging settings for the Flux broker.
  logging:
    # Strict mode will fail the job if the broker has an issue.
    strict: true

  # Defines the application container to be run under Flux.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
      # The command to be executed by the job. The Flux Operator will
      # automatically wrap this with 'flux run -n <tasks> ...'.
      command: lmp -v x 2 -v y 2 -v z 2 -in ./in.reaxff.hns -nocite
      # This container is part of the MiniCluster job itself, not a separate launcher pod.
      launcher: false
      # No resource requests or limits are set, allowing the pod to be 'Burstable'
      # and use all available node resources if needed.
