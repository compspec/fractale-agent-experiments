# Base Image: Use a recent, stable base like Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package management to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, and certificates
# - build-essential, cmake: For compiling C++ applications
# - git: To clone the LAMMPS source code repository
# - openmpi-bin, libopenmpi-dev: OpenMPI runtime and development libraries for parallel execution
# - ca-certificates: To allow secure connections (e.g., for git clone)
# We clean up the apt cache in the same RUN layer to reduce the final image size.
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for container environments like Kubernetes.
# These settings prevent OpenMPI from using shared memory mechanisms that
# are not available across pods and tell it to ignore local loopback and
# docker-specific network interfaces, improving stability in clustered runs.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_pml=ob1

# Define the location for the LAMMPS source code
ENV LAMMPS_SRC_DIR=/opt/lammps

# Clone the latest stable branch of the LAMMPS repository.
# A shallow clone (--depth 1) is used to reduce download size.
RUN git clone --depth 1 https://github.com/lammps/lammps.git ${LAMMPS_SRC_DIR}

# Build and install LAMMPS from source.
# 1. Create a build directory to keep the source tree clean.
# 2. Run cmake to configure the build. We enable MPI (BUILD_MPI) and the
#    required ReaxFF package (PKG_REAXFF). KSPACE is also enabled as it is common.
# 3. Compile using all available processor cores (`make -j$(nproc)`).
# 4. Install the 'lmp' executable to `/usr/local/bin`, which is on the system PATH.
RUN cd ${LAMMPS_SRC_DIR} && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
      -D CMAKE_INSTALL_PREFIX=/usr/local \
      -D BUILD_MPI=on \
      -D PKG_REAXFF=on \
      -D PKG_KSPACE=on \
    && make -j$(nproc) \
    && make install

# Set the default working directory for container execution.
# This is where simulations will be run from.
WORKDIR /data

# Copy the required example input files from the cloned source into the WORKDIR.
# This prepares the container to run the 'HNS' ReaxFF example immediately.
RUN cp ${LAMMPS_SRC_DIR}/examples/reaxff/HNS/* /data/

# The 'lmp' executable is now on the system PATH, ready for use.
# Example command to run a simulation:
# mpirun -np <cores> lmp -in in.hns
