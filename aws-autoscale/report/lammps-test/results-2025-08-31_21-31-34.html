# Base Image: Use Ubuntu 22.04 LTS for a modern, stable, and widely supported base.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, and other required packages.
# - ca-certificates: For secure network communications (e.g., git clone).
# - build-essential, cmake, git: Core tools for compiling C++ applications from source.
# - openmpi-bin, libopenmpi-dev: OpenMPI runtime and development headers for parallel execution.
# - libfftw3-dev: A common dependency for LAMMPS used for Fast Fourier Transforms.
# Clean up apt cache in the same layer to reduce image size.
RUN apt-get update && \
    apt-get install -y \
    ca-certificates \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments (e.g., Kubernetes).
# These settings help prevent common errors when running as root or in resource-constrained environments.
# - rmaps_base_oversubscribe: Allows MPI to run more processes than available physical cores,
#   which is essential in Kubernetes where CPU allocation is managed by the scheduler, not MPI.
# - btl_vader_single_copy_mechanism: Avoids potential shared-memory issues within containers.
RUN echo "rmaps_base_oversubscribe = 1" >> /etc/openmpi-mca-params.conf && \
    echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi-mca-params.conf

# Clone the latest stable release of LAMMPS from the official GitHub repository.
# --depth 1 creates a shallow clone, significantly reducing the repository size.
RUN git clone --depth 1 -b stable https://github.com/lammps/lammps.git /lammps

# Set the working directory to the cloned source code for the build process.
WORKDIR /lammps

# Build and install LAMMPS using CMake.
# - A separate build directory is created to keep the source tree clean.
# - D BUILD_MPI=ON: Enables MPI support.
# - D PKG_REAXFF=ON: Compiles the ReaxFF package required for this application.
# - The application is installed to /usr/local, a standard location for system-wide binaries.
# - The build directory is removed after installation to minimize image size.
RUN mkdir build && \
    cd build && \
    cmake ../cmake \
    -D CMAKE_INSTALL_PREFIX=/usr/local \
    -D BUILD_MPI=ON \
    -D PKG_REAXFF=ON \
    && \
    make -j$(nproc) && \
    make install && \
    cd / && \
    rm -rf /lammps/build

# Add the installation directory to the system's PATH.
# This ensures the 'lmp' executable is discoverable without specifying its full path.
ENV PATH="/usr/local/bin:${PATH}"

# Set the final working directory to the specified LAMMPS example directory.
# This allows users to immediately run the HNS ReaxFF example.
WORKDIR /lammps/examples/reaxff/HNS

# The container is now prepared.
# An example command to run a simulation would be provided at runtime, e.g.,
# CMD ["mpirun", "-np", "4", "lmp", "-in", "in.hns"]
