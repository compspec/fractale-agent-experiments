dockerfile
# Dockerfile for lammps-reax for CPU execution in a Kubernetes environment
# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies, build LAMMPS with ReaxFF, and clean up in a single layer
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    python3 \
    python3-dev \
    libopenmpi-dev \
    openmpi-bin \
    libfftw3-dev && \
    # Create a temporary directory for the build
    mkdir -p /tmp/build && \
    cd /tmp/build && \
    # Clone the latest version of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake
    # Enable REAXFF, KSPACE (for FFTW), and MPI support
    cmake ../cmake \
        -D CMAKE_INSTALL_PREFIX=/usr/local \
        -D BUILD_MPI=on \
        -D PKG_REAXFF=on \
        -D PKG_KSPACE=on && \
    # Build and install LAMMPS using all available processor cores
    make -j$(nproc) && \
    make install && \
    # Create the final working directory and copy example files
    mkdir -p /app && \
    cp /tmp/build/lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up build dependencies and source code to reduce image size
    cd / && \
    rm -rf /tmp/build && \
    apt-get purge -y --auto-remove build-essential cmake git python3-dev libopenmpi-dev libfftw3-dev && \
    rm -rf /var/lib/apt/lists/*

# Add LAMMPS executables to the system PATH
ENV PATH=/usr/local/bin:$PATH

# Configure OpenMPI for containerized environments
# Exclude loopback and docker network interfaces to prevent hangs
# Disable shared memory for single-node jobs to avoid potential issues in some container runtimes
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_pml=ob1
ENV OMPI_MCA_btl=^openib

# Set the working directory for the container
WORKDIR /app

# This MiniCluster manifest is designed to run a LAMMPS ReaxFF simulation
# using the Flux Operator on a single Kubernetes node.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  # Name of the MiniCluster instance
  name: lammps-reax-minicluster
  # Deploys to the default namespace as requested
  namespace: default
spec:
  # The total number of pods in the MiniCluster. Set to 1 to run on a single node.
  size: 1

  # The total number of MPI tasks to run.
  # The lmp command uses a 2x2x2 decomposition, which corresponds to 8 tasks.
  tasks: 8

  # Keep the pods and associated resources after the job completes for inspection.
  cleanup: false

  # Enable strict logging for better debugging and production readiness.
  logging:
    strict: true

  # Defines the application container to be run under Flux.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
      # The command to be executed by Flux.
      # The Flux Operator will automatically prepend 'flux run -n <tasks> ...' to this command.
      command: lmp -v x 2 -v y 2 -v z 2 -in ./in.reaxff.hns -nocite

      # Per instructions, this container is a worker, not the launcher.
      # The Flux Operator will create a separate sidecar container to run the Flux broker.
      launcher: false
