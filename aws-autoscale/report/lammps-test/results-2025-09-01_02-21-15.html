dockerfile
# Use a standard Ubuntu base image suitable for AWS CPU instances
FROM ubuntu:22.04

# Prevent interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for container environments
# This allows running as root, enables oversubscription (common in Kubernetes),
# and disables a shared memory mechanism (BTL vader) that can cause issues.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_rmaps_base_oversubscribe=1

# Set the working directory for the application runs
WORKDIR /data

# Install dependencies, clone LAMMPS source, build with REAXFF,
# install the executable, copy example files, and clean up the source code.
# This is done in a single RUN layer to optimize image size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    libfftw3-dev \
    ca-certificates \
    python3 \
    && rm -rf /var/lib/apt/lists/* \
    # Clone the 'develop' branch which contains the latest code updates
    && git clone --depth 1 -b develop https://github.com/lammps/lammps.git /tmp/lammps-src \
    && cd /tmp/lammps-src \
    && mkdir build && cd build \
    # Configure the build with CMake, enabling MPI, OpenMP, and the REAXFF package
    && cmake ../cmake \
        -D CMAKE_INSTALL_PREFIX=/usr/local \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=on \
        -D BUILD_OMP=on \
        -D PKG_REAXFF=on \
    # Compile and install LAMMPS
    && make -j$(nproc) \
    && make install \
    # Copy example files into the WORKDIR before deleting the source
    && cp /tmp/lammps-src/examples/reaxff/HNS/* /data/ \
    # Clean up the source code to reduce final image size
    && rm -rf /tmp/lammps-src

# The LAMMPS executable `lmp` is now installed in /usr/local/bin, which is on the PATH.
# The container is ready to run LAMMPS simulations from the /data directory.
CMD ["/bin/bash"]

# 2. When running on an ARM node, the `flux.container.image` must be updated to an ARM-compatible image.
# 3. Set `backoffLimit` and `pod.restartPolicy` to ensure the job fails cleanly for easier debugging.
apiVersion: flux-framework.org/v1alpha2

  name: lammps-reax-minicluster
  namespace: default

  # Set backoffLimit so the job fails after one attempt, similar to a Kubernetes Job.
  # The field 'maxRestarts' is not valid in the v1alpha2 CRD.
  backoffLimit: 1

  - command: lmp -v x 2 -v y 2 -v z 2 -in ./in.reaxff.hns -nocite
    image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
