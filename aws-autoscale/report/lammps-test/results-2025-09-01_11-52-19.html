# Use a stable, widely-used base image for compatibility. Ubuntu 22.04 is a good choice.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI libraries, git, and certificates in a single layer.
# --no-install-recommends reduces image size.
# Cleaning apt cache at the end of the same RUN command reduces layer size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Clone the LAMMPS source code. The 'stable' branch is used for production-readiness.
# Using --depth 1 for a shallow clone to save space and time.
RUN git clone --depth 1 --branch stable https://github.com/lammps/lammps.git /opt/lammps

# Set the working directory to a dedicated build folder.
WORKDIR /opt/lammps/build

# Configure the LAMMPS build using CMake.
# - PKG_REAXFF=on: Enables the ReaxFF package as required by the 'lammps-reax' application name.
# - BUILD_MPI=yes: Compiles LAMMPS with MPI support, crucial for HPC workloads.
RUN cmake \
    -D PKG_REAXFF=on \
    -D BUILD_MPI=yes \
    ../cmake

# Compile the LAMMPS executable using all available CPU cores to speed up the build.
RUN make -j$(nproc)

# Add the LAMMPS build directory containing the 'lmp' executable to the system's PATH.
# This ensures 'lmp' can be called directly from any location.
ENV PATH /opt/lammps/build:$PATH

# Configure OpenMPI environment variables for better compatibility within containerized environments like Kubernetes/Docker.
# This allows MPI to run as root and disables a shared-memory mechanism that can cause issues.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Create and set a new working directory for running simulations.
WORKDIR /app

# As per instructions, copy all files from the specified ReaxFF example directory into the WORKDIR.
# This stages the required input files for a potential HNS benchmark run.
RUN cp /opt/lammps/examples/reaxff/HNS/* .

# Set a default command. This will execute if the container is run without a command,
# providing basic information about the LAMMPS executable.
CMD ["lmp", "-h"]
