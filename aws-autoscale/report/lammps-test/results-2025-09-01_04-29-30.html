# Use Ubuntu 22.04 as the base image for a stable and modern environment.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary build and runtime dependencies in a single layer.
# This includes:
# - build-essential, cmake, git: For compiling LAMMPS from source.
# - ca-certificates: For secure network communication (e.g., git clone).
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel execution.
# - libfftw3-dev: FFTW library, a common LAMMPS dependency.
# - openssh-client, openssh-server: For MPI communication across container boundaries.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    openssh-client \
    openssh-server && \
    # Create the sshd run directory required by the SSH server.
    mkdir -p /var/run/sshd && \
    # Clean up apt caches to reduce image size.
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure passwordless SSH for MPI process management.
# This is crucial for mpirun to launch processes on other nodes (pods) in a cluster.
# It also allows for seamless multi-process execution within a single container.
RUN ssh-keygen -t rsa -f /root/.ssh/id_rsa -N '' && \
    cat /root/.ssh/id_rsa.pub >> /root/.ssh/authorized_keys && \
    chmod 0600 /root/.ssh/authorized_keys && \
    # Disable strict host key checking, essential for ephemeral container environments
    # where host keys are not consistent.
    echo "    StrictHostKeyChecking no" >> /etc/ssh/ssh_config && \
    echo "    UserKnownHostsFile /dev/null" >> /etc/ssh/ssh_config

# Configure OpenMPI for containerized/cloud environments.
# These settings force MPI to use the TCP protocol over the primary network interface (eth0),
# which is necessary for communication between Kubernetes pods, instead of relying on
# shared memory (shmem) or specialized interconnects not available in this environment.
RUN echo "btl_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl = tcp,self" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf

# Clone the latest stable version of the LAMMPS source code.
# --depth 1 creates a shallow clone, significantly reducing download size and time.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps-src

# Build and install LAMMPS from source.
# The entire process is done in a single RUN command to optimize Docker layer caching.
RUN cd /opt/lammps-src && \
    mkdir build && \
    cd build && \
    # Configure the build using CMake.
    # - PKG_REAXFF=on: Enables the ReaxFF package as requested.
    # - BUILD_MPI=on: Enables MPI support for parallel runs.
    # - CMAKE_BUILD_TYPE=Release: Creates a performance-optimized build.
    # - CMAKE_INSTALL_PREFIX: Specifies a clean installation directory.
    cmake ../cmake \
        -D PKG_REAXFF=on \
        -D BUILD_MPI=on \
        -D CMAKE_BUILD_TYPE=Release \
        -D CMAKE_INSTALL_PREFIX=/usr/local/lammps && \
    # Compile using all available processor cores.
    make -j$(nproc) && \
    # Install the compiled application.
    make install

# Set the primary working directory for the container.
WORKDIR /app

# Copy all example files from the specified LAMMPS directory into the WORKDIR.
RUN cp /opt/lammps-src/examples/reaxff/HNS/* /app/

# Clean up the source code to minimize the final image size.
RUN rm -rf /opt/lammps-src

# Add the LAMMPS binary directory to the system's PATH environment variable.
# This makes the 'lmp' executable directly available without needing the full path.
ENV PATH="/usr/local/lammps/bin:${PATH}"

# Set a default command to provide an interactive shell.
# This allows users to easily execute commands or can be overridden
# in the Kubernetes pod spec with a specific lmp command.
CMD ["/bin/bash"]
