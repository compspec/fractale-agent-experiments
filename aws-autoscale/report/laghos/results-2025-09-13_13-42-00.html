# Base image: Ubuntu 22.04 LTS for a stable and widely supported environment
FROM ubuntu:22.04

# Set environment variable to skip interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments (e.g., Kubernetes)
# Allow running as root, which is common in containers
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Disable shared memory and InfiniBand, forcing TCP communication for inter-pod networking
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_pml=ob1

# A single RUN command to install dependencies, build libraries and the application
# This minimizes the number of layers in the final Docker image
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    make \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    # Clean up apt cache to reduce image size
    && rm -rf /var/lib/apt/lists/* \
    # Create a temporary directory for building dependencies
    && mkdir -p /tmp/build \
    && cd /tmp/build \
    #
    # === Build and install METIS 4.0.3 ===
    #
    && wget --no-check-certificate https://github.com/mfem/tpls/raw/gh-pages/metis-4.0.3.tar.gz -O metis-4.0.3.tar.gz \
    && tar -xzf metis-4.0.3.tar.gz \
    && cd metis-4.0.3 \
    # This flag suppresses a compiler error for an implicit function declaration in the old METIS code.
    && make OPTFLAGS="-Wno-error=implicit-function-declaration" \
    # METIS 4.0.3 lacks a 'make install' target; manually copy the library and header.
    && cp libmetis.a /usr/local/lib/ \
    && cp Lib/metis.h /usr/local/include/ \
    && cd /tmp/build \
    #
    # === Build and install Hypre 2.28.0 ===
    #
    && wget --no-check-certificate https://github.com/hypre-space/hypre/archive/v2.28.0.tar.gz -O hypre-v2.28.0.tar.gz \
    && tar -xzf hypre-v2.28.0.tar.gz \
    && cd hypre-2.28.0/src \
    # Explicitly set the install prefix so linker can find libHYPRE.
    && ./configure --prefix=/usr/local --disable-fortran --enable-bigint \
    && make -j$(nproc) \
    && make install \
    # The 'make install' for this version of Hypre is incomplete and does not copy all required
    # headers. We copy them manually to ensure they are found by MFEM.
    && cp HYPRE_config.h /usr/local/include/ \
    && cp hypre/include/*.h /usr/local/include/ \
    # Update the shared library cache.
    && ldconfig \
    && cd /tmp/build \
    #
    # === Build and install MFEM 4.7 ===
    #
    && git clone https://github.com/mfem/mfem.git \
    && cd mfem \
    && git checkout v4.7 \
    # Explicitly specify METIS and Hypre locations to ensure the build system finds them.
    && make parallel MFEM_USE_METIS=YES METIS_DIR=/usr/local MFEM_USE_HYPRE=YES HYPRE_DIR=/usr/local -j$(nproc) \
    && make install \
    #
    # === Clone and build the Laghos application ===
    #
    && git clone https://github.com/CEED/Laghos.git /opt/laghos \
    && cd /opt/laghos \
    # Point directly to the MFEM build directory, which is more robust
    # than relying on the installed configuration files to find the library.
    && MFEM_DIR=/tmp/build/mfem make -j$(nproc) \
    # Copy the final executable to a system location on the PATH
    && cp laghos /usr/local/bin/laghos \
    #
    # === Clean up the temporary build directory to reduce final image size ===
    #
    && rm -rf /tmp/build

# Set the working directory to the Laghos application directory.
# This ensures that the application can find its required './data' folder at runtime.
WORKDIR /opt/laghos

# Define the default command to run when the container starts.
CMD ["laghos"]
