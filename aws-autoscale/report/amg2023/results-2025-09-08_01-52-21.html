# Base Image: Ubuntu 22.04 LTS for a stable and widely supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, git for cloning the repo, and Open MPI for parallel execution.
# ca-certificates is crucial for secure network communications (e.g., git clone).
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    git \
    make \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments like Kubernetes.
# This allows MPI to run as root and specifies the network interface for communication,
# which is essential for multi-node jobs in a container orchestrator.
RUN echo "btl_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_base_warn_component_unused = 0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "rmaps_base_mapping_policy = core" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "hwloc_base_binding_policy = none" >> /etc/openmpi/openmpi-mca-params.conf

# Set a working directory for the build process
WORKDIR /build

# Clone the amg2023 (AMG) source code from the official LLNL repository.
# git will clone the default branch, which is the latest stable version.
RUN git clone https://github.com/LLNL/AMG.git .

# Compile the application. The Makefile is located in the 'test' subdirectory.
RUN make -C test

# Copy the compiled executable to a system-wide location on the PATH.
# This makes the 'amg' command directly available to the user and applications.
RUN cp /build/test/amg /usr/local/bin/amg

# Clean up the build directory to reduce the final image size.
RUN rm -rf /build

# Set a default working directory for when the container starts.
WORKDIR /data

# Define the default command to run when the container starts.
# This entrypoint simply shows the path to the amg executable.
ENTRYPOINT ["which", "amg"]
