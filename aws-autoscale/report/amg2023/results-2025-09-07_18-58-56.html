# Base image: Ubuntu 22.04 LTS for a stable and recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, and required certificates
# - build-essential: Compiler toolchain (gcc, g++, make)
# - git: For cloning the application source code
# - ca-certificates: For secure network communications (e.g., git clone)
# - openmpi-bin & libopenmpi-dev: OpenMPI runtime and development libraries
# Clean up apt cache to reduce image size
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    git \
    openmpi-bin \
    libopenmpi-dev && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes
# - Allow running as root, which is common in containers.
# - Force TCP communication over the primary network interface (eth0).
# - Disable InfiniBand/other high-speed interconnects not present in standard cloud VMs.
# - Allow oversubscription of CPU cores.
RUN echo "btl_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl = tcp,self,vader" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "orte_base_help_aggregate = 0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "rmaps_base_oversubscribe = 1" >> /etc/openmpi/openmpi-mca-params.conf

# Set a working directory for the build process
WORKDIR /opt/build

# Clone the latest branch of the AMG (Algebraic Multi-Grid) benchmark source code
# The canonical repository is LLNL/AMG. We'll clone it into a directory named 'amg2023'.
RUN git clone https://github.com/LLNL/AMG.git amg2023

# Change directory to the test subdirectory where the Makefile is located
WORKDIR /opt/build/amg2023/test

# Compile the AMG application using the default Makefile which uses 'mpicc'
RUN make

# Copy the compiled executable to a system location on the PATH for easy access
RUN cp amg /usr/local/bin/amg

# Clean up the build directory to remove source code and intermediate object files,
# reducing the final image size.
RUN rm -rf /opt/build

# Set the default working directory for the final image
WORKDIR /
