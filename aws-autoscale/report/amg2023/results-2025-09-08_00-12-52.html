# Stage 1: Build the amg2023 application
# Using Ubuntu 22.04 as the base image for a modern, stable environment.
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies, Git, CA certificates, and Open MPI
# - build-essential: Provides C/C++ compilers (gcc, g++) and make.
# - git: Required to clone the source code from the repository.
# - ca-certificates: Required for secure HTTPS connections (e.g., for git).
# - openmpi-bin & libopenmpi-dev: Provide the MPI implementation (mpicc, mpirun) and libraries.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev && \
    # Clean up apt cache to reduce image size
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments like Kubernetes.
# These settings prevent common issues by:
# - Disabling the shared-memory (vader) and InfiniBand (openib) Byte Transfer Layers (BTLs),
#   which may not be available or can cause hangs in virtualized networks.
# - Forcing TCP communication over the primary network interface (typically eth0 in pods).
# - Disabling help message aggregation to prevent potential hangs with mpirun.
ENV OMPI_MCA_btl=^vader,openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_orte_base_help_aggregate=0

# Set a temporary working directory for the build process.
WORKDIR /build

# Clone the latest version of the AMG benchmark source code, build it,
# install the executable to the system PATH, and clean up.
RUN git clone https://github.com/LLNL/AMG.git && \
    cd AMG/test && \
    # Compile the application using the 'mpicc' wrapper provided by Open MPI.
    make && \
    # Copy the final executable to a directory on the default system PATH.
    # This makes 'amg' directly available to the user.
    cp amg /usr/local/bin/amg && \
    # Clean up the build directory to minimize the final image size.
    cd / && \
    rm -rf /build

# Set the default working directory for the container.
WORKDIR /root

# Define the default command. The user will typically override this with
# 'mpirun' to launch the parallel application.
# Example: mpirun -np 4 amg -n 100 100 100
CMD ["amg", "--help"]
