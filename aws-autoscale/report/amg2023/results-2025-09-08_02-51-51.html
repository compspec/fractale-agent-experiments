# Base image: Ubuntu 22.04 LTS for a stable and widely supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to avoid prompts

# Install build dependencies, MPI, git, and certificates
# - build-essential: Provides C/C++ compilers (gcc, g++) and make
# - git: For cloning the source code repository
# - ca-certificates: Required for secure network communications (e.g., git clone over https)
# - openmpi-bin & libopenmpi-dev: OpenMPI runtime and development libraries for parallel computing
RUN apt-get update && \
    apt-get install -y \
    build-essential \
    ca-certificates \
    git \
    make \
    openmpi-bin \
    libopenmpi-dev && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized/cloud environments (like Kubernetes)
# This forces MPI to use the standard TCP network interface (eth0) inside the container,
# preventing it from trying to find specialized HPC interconnects (like InfiniBand) which don't exist.
RUN echo "btl_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_include = eth0" >> /etc/openmpi/openmpi-mca-params.conf

# Clone, build, and install the amg2023 application in a single layer to optimize image size
# The application executable 'amg' is moved to /usr/local/bin to be in the system's PATH.
# The source code is removed after the build to keep the final image lean.
RUN git clone https://github.com/LLNL/AMG.git /tmp/amg2023_build && \
    cd /tmp/amg2023_build/test && \
    make && \
    mv amg /usr/local/bin/amg && \
    cd / && \
    rm -rf /tmp/amg2023_build

# Set a default working directory for the container
WORKDIR /data

# Define a default command to be executed when the container starts.
# Running 'amg' without arguments typically shows usage information.
CMD ["amg"]
