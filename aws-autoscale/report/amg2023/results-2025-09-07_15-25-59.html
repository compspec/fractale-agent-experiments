# Dockerfile for amg2023
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS for a stable, recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installation to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies, Git, certificates, and OpenMPI
# - build-essential, gfortran, make: For compiling the C/Fortran application
# - git: To clone the source code repository
# - ca-certificates: For secure network communications (e.g., git clone)
# - openmpi-bin, libopenmpi-dev: The MPI implementation and development headers
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    make \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev && \
    # Clean up apt cache to reduce image size
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes
# These settings ensure robust performance by forcing TCP communication over container networks
# and preventing common issues with process binding and shared memory.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=tcp,self
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_rmaps_base_oversubscribe=1
ENV OMPI_MCA_hwloc_base_binding_policy=none

# Create a temporary directory for the build process
WORKDIR /build

# Clone the latest branch of the amg2023 source code, build it,
# install the executable to a system PATH, and clean up the source code.
# This entire process is in a single RUN layer to optimize image size,
# as a multi-stage build was explicitly forbidden.
RUN git clone https://github.com/llnl/amg.git && \
    cd amg/test && \
    make && \
    # Per instructions, copy the executable to a system location on the PATH
    cp amg /usr/local/bin/amg && \
    # Clean up the build directory to keep the final image clean
    cd / && \
    rm -rf /build

# Set the final working directory for the container
WORKDIR /

# Default command to run the application's help message,
# which also serves as a quick verification that the build was successful
# and the executable is in the PATH.
CMD ["amg", "--help"]
