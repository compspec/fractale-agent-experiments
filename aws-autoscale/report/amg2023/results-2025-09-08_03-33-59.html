# Base Image: Ubuntu 22.04 LTS for a stable, well-supported foundation
FROM ubuntu:22.04

# Set non-interactive frontend for package management to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, clone, build, install the application, and clean up in a single layer
# This minimizes image size and keeps the image clean.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        build-essential \
        ca-certificates \
        git \
        openmpi-bin \
        libopenmpi-dev && \
    # Clone the latest branch of the AMG (amg2023) repository
    git clone https://github.com/LLNL/AMG.git /opt/amg-src && \
    # Navigate into the source directory to build
    cd /opt/amg-src/test && \
    # Compile the application using the provided Makefile
    make amg && \
    # Copy the compiled executable to a system location on the PATH
    cp amg /usr/local/bin/amg && \
    # Clean up the build environment to reduce final image size
    cd / && \
    rm -rf /opt/amg-src && \
    apt-get purge -y build-essential git libopenmpi-dev && \
    apt-get autoremove -y && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments and to allow running as root
# This is critical for compatibility with orchestrators like Kubernetes
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Disable shared memory mechanisms that can cause issues in containers
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
# Exclude InfiniBand, as this is a CPU-only environment
ENV OMPI_MCA_btl="^openib"
# Exclude loopback and docker bridge network interfaces from MPI communication
ENV OMPI_MCA_btl_tcp_if_exclude="lo,docker0"

# Set a working directory for application data
WORKDIR /data

# Default command to verify MPI installation. This can be overridden at runtime.
# Example: docker run <image> mpirun -np 4 amg -n 100 100 100
CMD ["mpirun", "--version"]
