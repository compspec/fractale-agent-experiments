# Base image with Flux, OpenMPI, and EFA support for ARM on AWS
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package management to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, including git, cmake, and essential certificates.
# Clean up apt cache to reduce image size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    make \
    && rm -rf /var/lib/apt/lists/*

# Set environment variables to use the MPI compiler wrappers provided by the base image.
# This ensures that Kripke is compiled with MPI support.
ENV CC=mpicc
ENV CXX=mpicxx

# Define the source directory for clarity
ARG APP_SRC_DIR=/opt/kripke

# Clone the Kripke repository, build it from source, install the executable, and clean up.
# --depth 1 is used for a shallow clone to save space and time.
# Kripke is configured with MPI enabled for multi-node execution.
# The build is parallelized using make -j.
# The final executable is installed to /usr/local/bin to be available in the system's PATH.
# The source code and build directories are removed to create a lean final image.
RUN git clone --depth 1 --branch main https://github.com/LLNL/Kripke.git ${APP_SRC_DIR} \
    && cd ${APP_SRC_DIR} \
    && mkdir build \
    && cd build \
    && cmake .. \
    -DENABLE_MPI=ON \
    -DCMAKE_BUILD_TYPE=Release \
    && make -j$(nproc) \
    && install -m 755 bin/kripke /usr/local/bin/kripke \
    && cd / \
    && rm -rf ${APP_SRC_DIR}

# Set the default command to show kripke help message
CMD ["kripke", "--help"]

# Corrected MiniCluster manifest for Kripke based on failure analysis.
# The previous attempt failed due to memory exhaustion on the worker nodes.
# This version scales the job to 4 nodes and 256 tasks to distribute the
# memory load more effectively across a larger number of MPI ranks.
