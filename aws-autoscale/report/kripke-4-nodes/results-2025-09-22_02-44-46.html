# Base image: Ubuntu 22.04 (arm64) with Flux, OpenMPI, and AWS EFA support pre-installed.
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package management to avoid prompts
ARG DEBIAN_FRONTEND=noninteractive

# Set environment variables to point CMake to the MPI C and C++ compilers.
# This is crucial for building MPI-enabled applications.
ENV CC=mpicc
ENV CXX=mpicxx

# Update package lists and install necessary build tools.
# - build-essential: A meta-package that installs the GNU C/C++ compilers and related tools.
# - cmake: Required for Kripke's build system.
# - git: Required to clone the source code repository.
# - ca-certificates: Required for secure network communications (e.g., git clone).
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the Kripke source code from the main branch into a temporary directory
RUN git clone --depth 1 https://github.com/LLNL/Kripke.git /opt/kripke-src

# Set the working directory to the source code for the build process
WORKDIR /opt/kripke-src

# Create a separate build directory for an out-of-source build
RUN mkdir build && cd build && \
    # Configure the build with CMake
    # - CMAKE_BUILD_TYPE=Release: Optimizes the build for performance.
    # - ENABLE_MPI=ON: Explicitly enables MPI support, which is required.
    cmake -DCMAKE_BUILD_TYPE=Release \
          -DENABLE_MPI=ON \
          .. && \
    # Compile the application using all available CPU cores for a faster build
    make -j$(nproc) && \
    # Copy the compiled executable to a standard location in the system's PATH
    # This makes the 'kripke' command available system-wide without needing to specify the full path.
    install bin/kripke /usr/local/bin/ && \
    # Return to the root directory
    cd / && \
    # Clean up the source code to reduce the final image size
    rm -rf /opt/kripke-src

# Set the default working directory for when the container starts
WORKDIR /
