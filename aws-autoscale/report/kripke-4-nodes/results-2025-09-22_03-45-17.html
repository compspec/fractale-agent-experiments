# Use a base image with Flux, OpenMPI, and EFA support for ARM on AWS
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Install build dependencies, git, cmake, and essential certificates
# We combine these into a single layer to optimize image size.
RUN apt-get update && \
    DEBIAN_FRONTEND=noninteractive apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git && \
    rm -rf /var/lib/apt/lists/*

# Clone, build, and install the Kripke application
# This entire process is done in a single RUN command to reduce image layers.
# The source code is cloned into a temporary directory, built, and then removed.
RUN git clone --depth 1 -b main https://github.com/LLNL/Kripke.git /tmp/kripke && \
    cd /tmp/kripke && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake, specifying the MPI compilers and enabling MPI support
    cmake -DCMAKE_BUILD_TYPE=Release \
          -DCMAKE_CXX_COMPILER=mpicxx \
          -DCMAKE_C_COMPILER=mpicc \
          -DENABLE_MPI=ON \
          .. && \
    # Build the application in parallel using all available cores
    make -j$(nproc) && \
    # Copy the executable to a system location on the PATH
    cp bin/kripke /usr/local/bin/kripke && \
    # Clean up the temporary source and build directories
    cd / && \
    rm -rf /tmp/kripke

# Set the default command to the kripke executable to provide usage information
CMD ["kripke", "--help"]
