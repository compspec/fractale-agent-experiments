dockerfile
# Use Ubuntu 22.04 LTS as the base image for its modern toolchain and long-term support.
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation of packages.
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies required for building and running Kripke.
# Manually add the official Kitware APT repository for a modern version of CMake (>=3.23)
# as the default Ubuntu version is too old for Kripke's submodules. This method is more
# reliable in CI/CD environments than using add-apt-repository.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    git \
    gnupg \
    wget \
    libopenmpi-dev \
    openmpi-bin \
    && wget -qO- https://apt.kitware.com/keys/kitware-archive-latest.asc | gpg --dearmor - | tee /etc/apt/trusted.gpg.d/kitware.gpg >/dev/null \
    && echo 'deb https://apt.kitware.com/ubuntu/ jammy main' | tee /etc/apt/sources.list.d/kitware.list >/dev/null \
    && apt-get update \
    && apt-get install -y --no-install-recommends cmake \
    && rm -rf /var/lib/apt/lists/*

# Set a working directory for the source code and build artifacts.
WORKDIR /build

# Clone the Kripke source code from the official LLNL GitHub repository.
# Use --recursive to ensure all necessary git submodules (like BLT and RAJA) are also cloned.
RUN git clone --recursive https://github.com/LLNL/Kripke.git .

# Create a build directory, then configure and compile the application in a single layer.
# This improves layer caching and reduces image size.
RUN mkdir builddir && cd builddir && \
    # Configure the build with CMake.
    # - CMAKE_BUILD_TYPE=Release: Enables compiler optimizations for production performance.
    # - ENABLE_MPI=ON: Explicitly enables MPI support as required.
    cmake \
    -D CMAKE_BUILD_TYPE=Release \
    -D ENABLE_MPI=ON \
    .. && \
    # Compile the application using all available CPU cores to speed up the build.
    make -j$(nproc)

# Install the compiled Kripke executable to a standard system-wide location.
# The build system places the executable in the 'bin' subdirectory.
# This ensures the 'kripke' command is available on the system's PATH.
RUN install /build/builddir/bin/kripke /usr/local/bin/kripke

# Clean up the build directory and source code to reduce the final image size.
RUN rm -rf /build

# Set the default working directory for when the container starts.
WORKDIR /

# Kripke is now installed and on the PATH.
# To run in a containerized environment like Kubernetes, use mpirun.
# Example command:
# mpirun --allow-run-as-root -np <num_processes> kripke <kripke_arguments>
# The '--allow-run-as-root' flag is essential for running MPI as the root user inside a container.
# Depending on the Kubernetes CNI, you may also need to specify the network interface, e.g.,
# --mca btl_tcp_if_include eth0
