# Use a standard Ubuntu base image suitable for building C++ applications.
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation of packages.
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies, and required certificates.
# This includes git for cloning the source, cmake for building, and OpenMPI for parallel execution.
# ca-certificates is crucial for secure connections (e.g., git clone over https).
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin && \
    # Clean up apt cache to reduce image size.
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings improve compatibility and performance by:
# - Allowing execution as the root user (common in containers).
# - Disabling shared memory mechanisms that can be problematic in some container runtimes.
# - Excluding loopback and docker bridge network interfaces from MPI communication.
# - Preventing hangs on error aggregation.
RUN echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "orte_base_help_aggregate = 0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "rmaps_base_mapping_policy = oversized" >> /etc/openmpi/openmpi-mca-params.conf

# Clone, build, and install the Kripke application from its main branch.
# The entire build process happens within the container.
RUN git clone https://github.com/LLNL/Kripke.git /opt/kripke && \
    cd /opt/kripke && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake, ensuring MPI support is enabled.
    cmake .. -DENABLE_MPI=ON && \
    # Compile the source code using all available processor cores.
    make -j$(nproc) && \
    # Copy the compiled executable to a standard system location on the PATH.
    # This makes the application directly runnable without specifying its full path.
    cp bin/kripke.exe /usr/local/bin/kripke && \
    # Clean up the source code directory to minimize the final image size.
    cd / && \
    rm -rf /opt/kripke

# Set the default working directory for the container.
# This is where the container will start, and where users can mount volumes.
WORKDIR /data

# Set the entrypoint to the kripke executable.
# This allows the container to be run as if it were the kripke command itself.
ENTRYPOINT ["kripke"]

# Provide a default command. If no arguments are provided to `docker run`,
# this will execute `kripke --help`, which is a safe and informative default action.
CMD ["--help"]
