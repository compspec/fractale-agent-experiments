dockerfile
# Dockerfile for Kripke HPC Application
# Target: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Prevent interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments.
# These settings help prevent common issues when running MPI in containers
# by allowing oversubscription and tuning the network/shared-memory layers.
ENV OMPI_MCA_rmaps_base_oversubscribe=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl="^openib"
ENV OMPI_MCA_pml="^ucx"

# Combine installation, build, and cleanup into a single RUN layer to minimize image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin && \
    #
    # Clone the Kripke repository. Note: The default branch is 'master', not 'main'.
    # The --recurse-submodules flag is essential to fetch the BLT build system dependency.
    #
    git clone --depth 1 --branch master --recurse-submodules https://github.com/LLNL/Kripke.git /opt/kripke_src && \
    #
    # Configure and build Kripke with MPI support
    #
    mkdir -p /opt/kripke_src/build && \
    cd /opt/kripke_src/build && \
    cmake \
      -DENABLE_MPI=ON \
      -DCMAKE_BUILD_TYPE=Release \
      .. && \
    make -j$(nproc) && \
    #
    # Copy the final executable to a directory on the system PATH.
    # The build produces 'kripke.exe', which is renamed to 'kripke' for standard path access.
    #
    cp /opt/kripke_src/build/bin/kripke.exe /usr/local/bin/kripke && \
    #
    # Clean up build-time dependencies and source files to reduce final image size
    #
    apt-get purge -y --auto-remove \
      build-essential \
      cmake \
      git \
      libopenmpi-dev && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/* /opt/kripke_src

# Set a working directory for the container for user data
WORKDIR /data

# Set the default command to display Kripke's help message.
# This serves as a simple check that the executable is on the PATH and works.
CMD ["kripke", "--help"]

#
# MiniCluster manifest to run Kripke on an AWS CPU instance.
#
# This manifest is intended as a baseline debugging step. Given a history of
# repeated failures caused by incorrect command-line arguments (--nthreads, --nest, --layout),
# this version runs Kripke with no parameters to verify the fundamental
# execution environment before re-introducing complexity.
#

  # Name of the MiniCluster job.
  name: kripke-aws-cpu
  # The job will be deployed to the 'default' namespace.
  namespace: default
spec:
  # The number of pods in the MiniCluster. Set to 1 to run on a single node.
  size: 1
  # The total number of MPI tasks. Kripke with no arguments defaults to a single task.
  tasks: 1

  # Logging configuration for the Flux Operator.
  logging:
    strict: true

  # If set to false, the pods and services are not deleted when the job finishes.
  cleanup: false

  # Pod-level specifications.
  pod:
    # Schedule the pod on a specific AWS instance type.
    nodeSelector:
      node.kubernetes.io/instance-type: "c6in.12xlarge"

  # Define the main application container.
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:kripke
      # FIX: Removed all command-line arguments to establish a baseline run.
      command: kripke
      # This container is not a dedicated launcher. The main job runs here.
      launcher: false
