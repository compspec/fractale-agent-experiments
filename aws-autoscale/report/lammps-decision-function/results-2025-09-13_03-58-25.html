# Base Image: Ubuntu 22.04 LTS for a stable, widely-supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments
# Allow running as root, which is common in containers
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Prevent OpenMPI from using shared memory transport (vader) for single-copy mechanism,
# which can cause issues in some container runtimes or virtualized environments.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
# Exclude loopback and docker network interfaces from TCP communication to ensure MPI uses the correct network for internode traffic.
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_oob_tcp_if_exclude=lo,docker0

# Install dependencies, clone, build LAMMPS, and clean up in a single layer to optimize image size
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/* \
    # Create a temporary directory for the build
    && mkdir -p /build/lammps-reax \
    && cd /build/lammps-reax \
    # Clone the latest version of the LAMMPS source code
    && git clone --depth 1 https://github.com/lammps/lammps.git . \
    # Create a build directory and enter it
    && mkdir build \
    && cd build \
    # Configure the build with CMake
    # Enable MPI, REAXFF package, and KSPACE for FFT support
    && cmake ../cmake \
        -D BUILD_MPI=on \
        -D PKG_REAXFF=on \
        -D PKG_KSPACE=on \
    # Compile LAMMPS using all available processor cores
    && make -j$(nproc) \
    # Copy the compiled LAMMPS executable to a system-wide location on the PATH
    && cp lmp /usr/local/bin/lmp \
    # Create the final working directory
    && mkdir -p /app \
    # Copy the required example files into the working directory
    && cp -r /build/lammps-reax/examples/reaxff/HNS/* /app/ \
    # Clean up the build directory to reduce image size
    && rm -rf /build

# Set the final working directory for the container
WORKDIR /app

# (Optional) A default command can be specified here, e.g., CMD ["lmp", "-in", "in.hns"]
# For this request, no CMD is set, allowing it to be defined at runtime by Kubernetes.
