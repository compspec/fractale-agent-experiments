# Base image: Ubuntu 22.04 LTS for a stable and widely supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments
# Allow running as root, which is common in containers
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Prevent issues with shared memory in some container runtimes (e.g., Docker)
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Install build dependencies, MPI, git, certificates, and other required libraries
# This is done in a single RUN command to reduce image layers
# We clean up apt cache in the same layer to reduce final image size
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    gfortran \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Clone, build, and install LAMMPS in a single RUN command to optimize layer usage
# This includes enabling the required REAXFF package and building with MPI support
# After the build, the executable is copied to the system PATH
# The required example files are moved to the final WORKDIR
# Finally, the build directory is cleaned up to keep the image size minimal
RUN git clone https://github.com/lammps/lammps.git /tmp/lammps \
    && cd /tmp/lammps \
    && mkdir build && cd build \
    && cmake ../cmake \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on \
    && make -j$(nproc) \
    && cp ./lmp /usr/local/bin/ \
    && mkdir -p /app \
    && cp /tmp/lammps/examples/reaxff/HNS/* /app/ \
    && rm -rf /tmp/lammps

# Set the final working directory for the container
WORKDIR /app

# Provide a default command to start an interactive shell
# Users can override this to run their LAMMPS simulations, e.g., `docker run <image> mpirun -np 4 lmp -in in.hns`
CMD ["/bin/bash"]
