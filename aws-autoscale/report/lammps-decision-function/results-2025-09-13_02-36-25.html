# Base image: Ubuntu 22.04 LTS for a modern, stable foundation
FROM ubuntu:22.04

# Set non-interactive frontend for package management to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, required libraries, and certificates
# - build-essential: Compiler toolchain (gcc, g++, make)
# - cmake: Required for building LAMMPS
# - git: For cloning the LAMMPS source code
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel execution
# - ca-certificates: For secure network connections (e.g., git clone)
# - wget: General purpose downloader
# - libfftw3-dev: FFTW library, a common dependency for LAMMPS packages
# - libgomp1: GNU OpenMP library, required for multi-threading
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    wget \
    libfftw3-dev \
    libgomp1 && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Clone the stable branch of LAMMPS source code
# We use /opt as a standard location for optional software
RUN git clone --depth 1 -b stable https://github.com/lammps/lammps.git /opt/lammps

# Configure, build, and install LAMMPS
# We create a separate build directory as is best practice for CMake
# - BUILD_MPI=yes: Enable MPI support
# - PKG_REAXFF=on: Enable the ReaxFF package as requested
# - PKG_MANYBODY=on: A common dependency for many potentials, including ReaxFF
# - PKG_KSPACE=on: For long-range electrostatics (e.g., PPPM)
# - PKG_MOLECULE=on: Required for molecular systems
RUN cd /opt/lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on \
        -D PKG_MANYBODY=on \
        -D PKG_KSPACE=on \
        -D PKG_MOLECULE=on && \
    make -j$(nproc)

# Copy the LAMMPS executable to a system location on the PATH
# This makes 'lmp' directly available to any user or script
RUN cp /opt/lammps/build/lmp /usr/local/bin/

# Create a working directory for running simulations
WORKDIR /data

# Copy the example input files for the specified ReaxFF HNS simulation
# into the working directory so the container is ready to run an example job.
RUN cp /opt/lammps/examples/reaxff/HNS/* .

# Clean up the build directory to reduce image size
RUN rm -rf /opt/lammps

# Set the default command to show LAMMPS version and help
# This confirms the installation and provides usage info
CMD ["lmp"]
