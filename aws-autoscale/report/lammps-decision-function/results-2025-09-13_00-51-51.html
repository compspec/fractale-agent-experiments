# Dockerfile for lammps-reax
# Target Environment: AWS CPU instance in Kubernetes
# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments, preferring TCP communication.
# This avoids issues with shared memory (BTL sm) and high-speed interconnects (like openib)
# that are typically not available in standard cloud container environments.
ENV OMPI_MCA_pml=ucx \
    OMPI_MCA_btl=^openib \
    UCX_TLS=tcp

# Install build dependencies, MPI, required libraries, and certificates
# This single RUN command helps to reduce the number of layers in the final image.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    gfortran \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

# Set a working directory for the build process
WORKDIR /build

# Clone, build, and install LAMMPS from source
# We clone the latest branch as requested.
# The REAXFF package is enabled via CMake flags.
# The resulting 'lmp' executable is copied to /usr/local/bin to be on the system PATH.
# Finally, the required example files are copied to the runtime working directory.
RUN git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    cmake -D PKG_REAXFF=on \
          -D BUILD_MPI=yes \
          ../cmake && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/ && \
    # Create the final runtime directory and copy example files into it
    mkdir -p /app && \
    cp /build/lammps/examples/reaxff/HNS/* /app/

# Set the final working directory where the application will run
WORKDIR /app

# Provide a default command to facilitate interactive use
CMD ["/bin/bash"]
