# Use a standard Ubuntu base image
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, certificates, and libgomp1
# - build-essential, cmake: For compiling C++ applications.
# - git: To clone the LAMMPS source code.
# - openmpi-bin, libopenmpi-dev: For MPI parallel execution.
# - ca-certificates: For secure network connections (e.g., git clone).
# - libgomp1: GNU OpenMP library, a specific requirement.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set the primary working directory for the application
WORKDIR /app

# Clone the latest version of the LAMMPS source code from the official repository
RUN git clone --depth 1 https://github.com/lammps/lammps.git

# Build LAMMPS using CMake
# An out-of-source build is performed in the 'build' directory.
# The 'REAXFF' package is explicitly enabled as required for this build.
# The build is parallelized using all available processor cores to speed up compilation.
RUN cd lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake -D PKG_REAXFF=on && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable to a system-wide location in the PATH
# This makes the 'lmp' command available globally without needing to specify its full path.
RUN cp /app/lammps/build/lmp /usr/local/bin/

# Copy the example files for the HNS (hexanitrostilbene) simulation using ReaxFF
# These files are placed in the WORKDIR to be readily accessible.
RUN cp /app/lammps/examples/reaxff/HNS/* .

# Configure OpenMPI for container environments.
# These settings prevent MPI from using loopback or Docker-internal network interfaces,
# which is a common requirement for running MPI applications in containers like Docker or Kubernetes.
RUN echo "btl_vader_single_copy_mechanism = none" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "btl_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf && \
    echo "oob_tcp_if_exclude = lo,docker0" >> /etc/openmpi/openmpi-mca-params.conf

# Set a default command to execute when the container starts.
# Running 'lmp -h' will display the LAMMPS help message, confirming the executable is on the PATH.
CMD ["lmp", "-h"]
