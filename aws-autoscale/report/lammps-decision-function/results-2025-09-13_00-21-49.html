# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, and required libraries for LAMMPS
# This includes ca-certificates as requested and libgomp1 for OpenMP support.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    fftw3-dev \
    libgomp1 && \
    # Clean up apt cache to reduce image size
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI environment variables for better performance and compatibility in containerized environments like Kubernetes.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_rmaps_base_mapping_policy=slot
ENV OMPI_MCA_orte_tag_output=1

# Clone the latest version of the LAMMPS source code, build it with the REAXFF package,
# install the executable, copy example files, and clean up the source code.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    mkdir build && \
    cd build && \
    # Configure the build using CMake. Enable MPI and the REAXFF and KSPACE packages.
    cmake ../cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=on \
      -D PKG_KSPACE=on && \
    # Compile LAMMPS using all available processor cores.
    make -j$(nproc) && \
    # Copy the compiled LAMMPS executable to a system-wide location on the PATH.
    cp lmp /usr/local/bin/lmp && \
    # Create the working directory for the simulation files.
    mkdir -p /app/data && \
    # Copy the specific example files into the working directory.
    cp /lammps/examples/reaxff/HNS/* /app/data/ && \
    # Remove the source code directory to minimize final image size.
    cd / && \
    rm -rf /lammps

# Set the application's working directory.
WORKDIR /app/data

# Set the default command to execute the LAMMPS binary.
# This allows users to run the container with LAMMPS command-line arguments.
# e.g., docker run <image> -in in.hns
ENTRYPOINT ["lmp"]
