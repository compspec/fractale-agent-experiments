# Use a standard Ubuntu base image
FROM ubuntu:22.04

# Set non-interactive frontend for package installation to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, FFTW, required libraries, and certificates
# This single layer helps optimize image size by combining package installation.
# libgomp1 is explicitly included as requested for OpenMP support.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments like Kubernetes.
# This helps prevent hangs or errors related to process binding and shared memory.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_rmaps_base_oversubscribe=1

# Clone the latest branch of the LAMMPS source code from the official repository
# The --depth 1 flag creates a shallow clone, reducing download size and time.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps

# Build LAMMPS using CMake
# We create an out-of-source build directory, which is standard practice.
# PKG_REAXFF is enabled for the reax potential.
# PKG_KSPACE is enabled for FFTW support, common in many simulations.
# PKG_OPENMP is enabled to utilize multi-threading on CPU cores.
RUN cd /lammps && \
    mkdir build && \
    cd build && \
    cmake \
      -D PKG_REAXFF=on \
      -D PKG_KSPACE=on \
      -D PKG_OPENMP=on \
      ../cmake && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable 'lmp' to a system-wide location on the PATH
# This makes the executable available globally without needing to specify its full path.
RUN cp /lammps/build/lmp /usr/local/bin/

# Set up the working directory for running simulations
WORKDIR /app

# Copy the specific ReaxFF HNS example files into the working directory
# These files will be present when the container starts.
RUN cp /lammps/examples/reaxff/HNS/* /app/

# Clean up the cloned source code directory to reduce the final image size
RUN rm -rf /lammps

# Set the default command to the LAMMPS executable.
# Users can override this or append arguments, e.g., 'docker run <image> -in in.hns'
CMD ["lmp"]
