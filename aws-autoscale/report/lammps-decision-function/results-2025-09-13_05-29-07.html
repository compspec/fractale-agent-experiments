# Base image: Ubuntu 22.04 LTS for a stable, well-supported environment
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, git, MPI, and required libraries
# ca-certificates is included for secure network operations (e.g., git clone)
# libgomp1 is an explicit requirement for OpenMP support
# No-install-recommends is used to keep the image size smaller
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    gfortran \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments, especially Kubernetes
# This allows running as root, oversubscribing CPUs, and forces TCP communication over the primary network interface (eth0)
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_rmaps_base_oversubscribe=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_pml=ob1
ENV OMPI_MCA_btl=self,tcp

# Create a source directory
WORKDIR /opt/src

# Clone the latest stable branch of LAMMPS, build, and install it
# We enable the required REAXFF package and the KSPACE package for FFT support
RUN git clone -b stable https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && cd build && \
    cmake ../cmake \
      -D CMAKE_INSTALL_PREFIX=/opt/lammps \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=yes \
      -D PKG_KSPACE=yes \
      -D FFTW3_LIBRARIES=/usr/lib/x86_64-linux-gnu/libfftw3.so \
      -D FFTW3_INCLUDES=/usr/include && \
    make -j$(nproc) && \
    make install

# Copy the main LAMMPS executable to a standard location in the system PATH
# This makes 'lmp' directly accessible from any directory
RUN cp /opt/lammps/bin/lmp /usr/local/bin/lmp

# Create and set the final working directory for running the application
WORKDIR /opt/app

# Copy the specific ReaxFF example files into the working directory as requested
RUN cp /opt/src/lammps/examples/reaxff/HNS/* /opt/app/

# Clean up the build source to reduce the final image size
RUN rm -rf /opt/src

# Set a default command to display LAMMPS help information
# This serves as a quick self-test that the executable is in the PATH and works
CMD ["lmp", "-h"]
