# Use a standard Ubuntu 22.04 as the base image
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to avoid prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments like Kubernetes
# Allow oversubscription of processes to cores, common in container scheduling
ENV OMPI_MCA_rmaps_base_oversubscribe=1
# Use a shared memory mechanism compatible with container namespaces
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
# Exclude the OpenIB BTL to avoid warnings when InfiniBand is not present
ENV OMPI_MCA_btl="^openib"

# Combine installation, build, and cleanup into a single RUN layer to minimize image size
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    libgomp1 \
    ca-certificates \
    && \
    # Clone the latest branch of the LAMMPS repository into /lammps
    git clone https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    # Create and move into a build directory
    mkdir build && cd build && \
    # Configure the LAMMPS build using CMake
    # Enable the required REAXFF package and other common packages
    cmake \
      -D PKG_REAXFF=on \
      -D PKG_KSPACE=on \
      -D PKG_MOLECULE=on \
      -D PKG_RIGID=on \
      ../cmake && \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the LAMMPS executable to a standard system location to make it available on the PATH
    cp lmp /usr/local/bin/ && \
    # Create the application working directory
    mkdir -p /app && \
    # Copy the specified example files from the cloned source into the working directory
    cp /lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up the build environment to reduce the final image size
    # Remove the cloned source code directory
    cd / && \
    rm -rf /lammps && \
    # Clean up the apt package cache
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set the default working directory for the container
WORKDIR /app

# Set a default command to start an interactive shell
CMD ["/bin/bash"]
