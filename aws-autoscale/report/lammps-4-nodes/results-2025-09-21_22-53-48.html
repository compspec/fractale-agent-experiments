# Use a base image with Flux, OpenMPI, and EFA pre-installed for ARM on AWS
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set a default working directory for the application
WORKDIR /app

# Install dependencies, clone, configure, and build LAMMPS
# This is a single RUN layer to optimize image size and build time.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates && \
    # Clone the latest stable branch of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps && \
    # Create a build directory
    cd /opt/lammps && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake
    # - CMAKE_BUILD_TYPE=Release: Optimize for performance
    # - CMAKE_C_COMPILER/CXX_COMPILER: Use MPI wrappers for compilation
    # - BUILD_MPI=on: Explicitly enable MPI support
    # - PKG_REAXFF=on: Enable the REAXFF package required for the simulation
    cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D CMAKE_C_COMPILER=mpicc \
      -D CMAKE_CXX_COMPILER=mpicxx \
      -D BUILD_MPI=on \
      -D PKG_REAXFF=on \
      ../cmake && \
    # Compile the code using all available processor cores
    make -j$(nproc) && \
    # Copy the main LAMMPS executable to a location in the system's PATH
    cp lmp /usr/local/bin/lmp && \
    # Copy the required example files into the final working directory
    cp /opt/lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up to reduce image size
    # Remove the source code directory as it is no longer needed
    cd / && \
    rm -rf /opt/lammps && \
    # Clean apt cache
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Verify that the 'lmp' executable is in the system PATH
RUN which lmp

# Set a default command to provide a shell for interactive use
CMD ["/bin/bash"]
