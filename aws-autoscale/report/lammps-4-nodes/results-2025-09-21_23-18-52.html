# Base Image: Ubuntu 22.04 for ARM with Flux, OpenMPI, and AWS EFA support
# This image is pre-configured for multi-node HPC workloads on AWS.
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set environment variable to ensure builds are non-interactive
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies required for LAMMPS
# Includes compilers, cmake, git, networking tools, and fftw library.
# ca-certificates are installed as requested.
# The apt cache is cleaned to keep the image layer small.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    libfftw3-dev \
    && apt-get clean && rm -rf /var/lib/apt/lists/*

# Create a temporary directory for the source code and build process
WORKDIR /build

# Clone the latest version of the LAMMPS source code from the official repository
RUN git clone https://github.com/lammps/lammps.git

# Configure, compile, and install LAMMPS
# The build is configured using CMake to leverage MPI for multi-node execution
# It specifically uses the mpicc and mpicxx wrappers from the base image.
# The REAXFF package is enabled as required for the 'lammps-reax' application.
# Common packages KSPACE and MOLECULE are also included.
RUN mkdir -p /build/lammps/build && cd /build/lammps/build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D CMAKE_C_COMPILER=mpicc \
        -D CMAKE_CXX_COMPILER=mpicxx \
        -D PKG_REAXFF=yes \
        -D PKG_KSPACE=yes \
        -D PKG_MOLECULE=yes && \
    # Compile using all available CPU cores for speed
    make -j$(nproc) && \
    # Copy the final executable to a system-wide location on the PATH
    cp ./lmp /usr/local/bin/

# Set the final runtime working directory
WORKDIR /workspace

# Copy the example files for the HNS ReaxFF benchmark into the WORKDIR
RUN cp /build/lammps/examples/reaxff/HNS/* /workspace/

# Clean up the build directory to minimize final image size
RUN rm -rf /build

# Verify that the 'lmp' executable is correctly installed and available on the PATH
RUN which lmp

# Set a default command to start an interactive shell
CMD ["/bin/bash"]
