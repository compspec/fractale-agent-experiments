dockerfile
# Base image with OpenMPI, Flux, and AWS EFA support for ARM
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, including build-essential, cmake, git, and certificates
# We clean up the apt cache in the same layer to reduce image size
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set a working directory for the application examples
WORKDIR /opt/lammps-reax

# Clone, build, and install LAMMPS in a single layer to optimize image size
# - Clone the latest version of the LAMMPS source code.
# - Create a build directory.
# - Configure the build with CMake, specifying MPI compilers and enabling the REAXFF package.
# - Compile the source code using all available processor cores.
# - Copy the main LAMMPS executable to a system path.
# - Copy the required example files into the WORKDIR.
# - Clean up the cloned source code to minimize the final image size.
RUN set -e && \
    cd /tmp && \
    git clone --depth 1 https://github.com/lammps/lammps.git lammps-src && \
    cd lammps-src && \
    mkdir build && \
    cd build && \
    cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D CMAKE_CXX_COMPILER=mpicxx \
      -D CMAKE_C_COMPILER=mpicc \
      -D PKG_REAXFF=yes \
      ../cmake && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/ && \
    cp /tmp/lammps-src/examples/reaxff/HNS/* /opt/lammps-reax/ && \
    rm -rf /tmp/lammps-src

# Verify that the lmp executable is in the system's PATH
RUN which lmp
