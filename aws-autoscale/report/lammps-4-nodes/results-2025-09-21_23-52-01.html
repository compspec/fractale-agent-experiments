# Use the specified ARM64 base image with OpenMPI, Flux, and EFA support for AWS
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set the working directory for the application
WORKDIR /app

# Install build dependencies, git for cloning, and certificates.
# Clean up apt cache to keep the image size down.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest branch of the LAMMPS source code, configure the build for MPI with the REAXFF package,
# compile, copy the executable to the system PATH, copy example files, and clean up the source directory.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    mkdir build && \
    cd build && \
    cmake -D PKG_REAXFF=on \
          -D BUILD_MPI=yes \
          -D CMAKE_C_COMPILER=mpicc \
          -D CMAKE_CXX_COMPILER=mpicxx \
          ../cmake && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/ && \
    cp /lammps/examples/reaxff/HNS/* /app/ && \
    cd / && \
    rm -rf /lammps

# Verify that the LAMMPS executable is in the PATH
RUN which lmp

# Set the default command to launch a bash shell
CMD ["/bin/bash"]
