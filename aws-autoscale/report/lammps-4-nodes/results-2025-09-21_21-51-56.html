# Base image with OpenMPI, Flux, and AWS EFA support for ARM64
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package installs
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, clone, build, and install LAMMPS
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        build-essential \
        ca-certificates \
        cmake \
        git && \
    # Clone the latest version of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git /tmp/lammps && \
    cd /tmp/lammps && \
    # Create a build directory
    mkdir build && cd build && \
    # Configure the build with CMake
    # - Enable MPI support
    # - Specify MPI C++ and C compilers
    # - Enable the REAXFF package as required
    # - Set install prefix to a system path location
    cmake \
      -D CMAKE_INSTALL_PREFIX=/usr/local \
      -D BUILD_MPI=yes \
      -D CMAKE_CXX_COMPILER=mpicxx \
      -D CMAKE_C_COMPILER=mpicc \
      -D PKG_REAXFF=ON \
      ../cmake && \
    # Build LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Install the lmp executable to /usr/local/bin
    make install && \
    # Verify that the 'lmp' executable is in the system PATH
    which lmp && \
    # Create a workspace directory for the application examples
    mkdir -p /opt/lammps-reax-workspace && \
    # Copy the required example files into the workspace
    # This is done from the cloned source before it's removed
    cp /tmp/lammps/examples/reaxff/HNS/* /opt/lammps-reax-workspace/ && \
    # Clean up to reduce image size
    cd / && \
    rm -rf /tmp/lammps && \
    rm -rf /var/lib/apt/lists/*

# Set the final working directory to the location of the example files
WORKDIR /opt/lammps-reax-workspace
