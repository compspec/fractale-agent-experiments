# Base image with OpenMPI, libfabric for AWS EFA, and Flux Framework on Ubuntu 22.04 ARM
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Prevents interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies for LAMMPS
# ca-certificates is included for secure network communications (e.g., git clone)
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set a working directory for the build process
WORKDIR /opt

# Clone the latest version of the LAMMPS source code
RUN git clone --depth 1 https://github.com/lammps/lammps.git

# Create a build directory and move into it
WORKDIR /opt/lammps/build

# Configure the LAMMPS build with CMake
# - Use mpicc and mpicxx as the C and CXX compilers for MPI compatibility
# - Enable the REAXFF package as required
# - Enable the KSPACE package for FFT calculations, common in HPC simulations
RUN cmake ../cmake \
    -D CMAKE_C_COMPILER=mpicc \
    -D CMAKE_CXX_COMPILER=mpicxx \
    -D PKG_REAXFF=yes \
    -D PKG_KSPACE=yes

# Compile LAMMPS using all available processor cores
RUN make -j$(nproc)

# Copy the compiled lmp executable to a system location on the PATH
RUN cp ./lmp /usr/local/bin/

# Set the final working directory for running simulations
WORKDIR /opt/lammps-reax

# Copy the example files for the reaxff/HNS case into the working directory
RUN cp /opt/lammps/examples/reaxff/HNS/* .

# Clean up the build source to reduce image size
RUN rm -rf /opt/lammps

# Verify that the 'lmp' executable is in the system PATH
RUN which lmp

# This corrected and scaled-up MiniCluster manifest is designed to run a larger
# LAMMPS simulation across multiple high-performance ARM-based AWS nodes.
# It incorporates performance optimizations for AWS EFA networking and targets
# specific instance types for consistent performance.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  # Name for the MiniCluster resource.
  name: lammps-reax-aws-cpu
  # Deploy to the default Kubernetes namespace.
  namespace: default
spec:
  # --- Job Scale and Placement ---
  # Number of pods (nodes) for the MiniCluster.
  size: 4
  # Maximum number of pods for bursting; set to the initial size.
  maxSize: 4
  # Number of MPI tasks to run per pod. This should match the vCPU count of the target instance.
  tasks: 64

  # Pod specification to control scheduling.
  pod:
    # Schedule pods only on nodes with the specified AWS instance type.
    nodeSelector:
      node.kubernetes.io/instance-type: m7g.16xlarge

  # --- Job Control and Logging ---
  # Keep pods running after the job finishes for inspection and debugging.
  cleanup: false
  # Configure strict logging to fail the job on any Flux framework error.
  logging:
    strict: true

  # --- Flux Framework Configuration ---
  flux:
    # Specify the architecture for the Flux broker, as required for ARM instances.
    arch: "arm"
    # Configuration for the Flux view container, used by the operator.
    container:
      # Use the ARM build of the Flux view container as requested.
      image: ghcr.io/converged-computing/flux-view-ubuntu:arm-jammy
      # Disable the Flux sidecar container. Flux will be injected into and run
      # directly within the main application container for better performance.
      disable: true

  # --- Application Container Definition ---
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
      # The command to run, scaled up for a larger, multi-node simulation.
      # The Flux Operator prepends the necessary `flux run` invocation.
      command: lmp -v x 8 -v y 8 -v z 4 -in ./in.reaxff.hns -nocite
      # Per instructions, this container is a worker in the MiniCluster, not just a job launcher.
      launcher: false
      # Ensure the latest version of the container is always pulled before running.
      imagePullPolicy: Always
      # The working directory where the input files are located inside the container.
      workingDir: /opt/lammps-reax

      # Environment variables required for MPI to use the high-speed AWS EFA interconnect.
      environment:
        FI_PROVIDER: efa
        FI_EFA_USE_DEVICE_RDMA: "1"
        OMPI_MCA_pml: cm
        OMPI_MCA_mtl: ofi

      # Resource requests and limits for AWS-specific hardware (EFA networking).
      # No CPU or memory limits are set, allowing the pod to be "Burstable".
      resources:
        limits:
          vpc.amazonaws.com/efa: "1"
        requests:
          vpc.amazonaws.com/efa: "1"
