# Dockerfile for Kripke HPC Application
# Target: AWS CPU Instance in Kubernetes

# Use Ubuntu 22.04 as the base image, a stable and modern choice for builds.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary build dependencies, MPI libraries, and certificates in a single layer.
# This includes:
# - build-essential, gfortran, cmake: Core tools for compiling C++/Fortran applications.
# - git: For cloning the application source code.
# - ca-certificates: For secure network communications (e.g., git clone).
# - openmpi-bin, libopenmpi-dev: The OpenMPI runtime and development libraries.
# - openssh-client: Often used by MPI process managers for communication between nodes.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    cmake \
    git \
    ca-certificates \
    wget \
    libopenmpi-dev \
    openmpi-bin \
    openssh-client \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings are crucial for stability and performance.
# 1. Allow OpenMPI to run as the root user.
# 2. Force MPI communication over the primary container network interface (typically eth0 in Kubernetes).
#    This prevents MPI from trying to use loopback or internal Docker networks.
# 3. Disable a shared-memory mechanism that can cause issues in some container runtimes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_orte_base_help_aggregate=0
ENV OMPI_MCA_plm_rsh_agent=ssh

# Set a temporary working directory for the source code download and build process.
WORKDIR /tmp

# Clone the Kripke source code from the main branch, build it with MPI enabled,
# and install it to a temporary location.
# A release build is used for performance.
RUN git clone --depth 1 --branch main https://github.com/LLNL/Kripke.git && \
    cd Kripke && \
    mkdir build && \
    cd build && \
    cmake \
        -DCMAKE_BUILD_TYPE=Release \
        -DENABLE_MPI=ON \
        -DCMAKE_INSTALL_PREFIX=/opt/kripke_install \
        .. && \
    make -j$(nproc) && \
    make install

# Copy the compiled Kripke executable to a standard system path location.
# This ensures 'kripke' can be called directly from the command line.
RUN cp /opt/kripke_install/bin/kripke.exe /usr/local/bin/kripke

# Clean up the build environment to reduce the final image size.
# This removes the source code, build directory, and temporary install files.
RUN rm -rf /tmp/Kripke /opt/kripke_install

# Set the final working directory for the container.
WORKDIR /

# Define the default command to execute when the container starts.
# Running 'kripke' without arguments will display its usage information.
CMD ["kripke"]

# Dockerfile for Kripke HPC Application
# Target: AWS CPU Instance in Kubernetes

# Use Ubuntu 22.04 as the base image, a stable and modern choice for builds.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary build dependencies, MPI libraries, and certificates in a single layer.
# This includes:
# - build-essential, gfortran, cmake: Core tools for compiling C++/Fortran applications.
# - git: For cloning the application source code.
# - ca-certificates: For secure network communications (e.g., git clone).
# - openmpi-bin, libopenmpi-dev: The OpenMPI runtime and development libraries.
# - openssh-client: Often used by MPI process managers for communication between nodes.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    cmake \
    git \
    ca-certificates \
    wget \
    libopenmpi-dev \
    openmpi-bin \
    openssh-client \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings are crucial for stability and performance.
# 1. Allow OpenMPI to run as the root user.
# 2. Force MPI communication over the primary container network interface (typically eth0 in Kubernetes).
#    This prevents MPI from trying to use loopback or internal Docker networks.
# 3. Disable a shared-memory mechanism that can cause issues in some container runtimes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_orte_base_help_aggregate=0
ENV OMPI_MCA_plm_rsh_agent=ssh

# Set a temporary working directory for the source code download and build process.
WORKDIR /tmp

# Clone the Kripke source code from the main branch, including its submodules,
# build it with MPI enabled, and install it to a temporary location.
# FIX: Added --recurse-submodules to fetch build system dependencies (BLT).
# FIX: Reverted branch from 'master' back to 'main' as originally requested.
# A release build is used for performance.
RUN git clone --depth 1 --recurse-submodules --branch main https://github.com/LLNL/Kripke.git && \
    cd Kripke && \
    mkdir build && \
    cd build && \
    cmake \
        -DCMAKE_BUILD_TYPE=Release \
        -DENABLE_MPI=ON \
        -DCMAKE_INSTALL_PREFIX=/opt/kripke_install \
        .. && \
    make -j$(nproc) && \
    make install

# Copy the compiled Kripke executable to a standard system path location.
# This ensures 'kripke' can be called directly from the command line.
# FIX: Corrected executable name from 'kripke.exe' to 'kripke'.
RUN cp /opt/kripke_install/bin/kripke /usr/local/bin/kripke

# Clean up the build environment to reduce the final image size.
# This removes the source code, build directory, and temporary install files.
RUN rm -rf /tmp/Kripke /opt/kripke_install

# Set the final working directory for the container.
WORKDIR /

# Define the default command to execute when the container starts.
# Running 'kripke' without arguments will display its usage information.
CMD ["kripke"]

# Use Ubuntu 22.04 as the base image, a stable and modern choice for builds.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.

# Install all necessary build dependencies, MPI libraries, and certificates in a single layer.
# This includes:
# - build-essential, gfortran, cmake: Core tools for compiling C++/Fortran applications.
# - git: For cloning the application source code.
# - ca-certificates: For secure network communications (e.g., git clone).
# - openmpi-bin, libopenmpi-dev: The OpenMPI runtime and development libraries.
# - openssh-client: Often used by MPI process managers for communication between nodes.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    gfortran \
    cmake \
    git \
    ca-certificates \
    wget \
    libopenmpi-dev \
    openmpi-bin \
    openssh-client \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings are crucial for stability and performance.
# 1. Allow OpenMPI to run as the root user.
# 2. Force MPI communication over the primary container network interface (typically eth0 in Kubernetes).
#    This prevents MPI from trying to use loopback or internal Docker networks.
# 3. Disable a shared-memory mechanism that can cause issues in some container runtimes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_orte_base_help_aggregate=0
ENV OMPI_MCA_plm_rsh_agent=ssh

# Set a temporary working directory for the source code download and build process.
WORKDIR /tmp

# Clone the Kripke source code from the main branch, including its submodules,
# build it with MPI enabled, and install it directly to a system path.
# A release build is used for performance.
# FIX: Corrected git branch to 'main' as per instructions.
# FIX: Changed CMAKE_INSTALL_PREFIX to /usr/local to install directly to the PATH.
RUN git clone --depth 1 --recurse-submodules --branch main https://github.com/LLNL/Kripke.git && \
    cd Kripke && \
    mkdir build && \
    cd build && \
    cmake \
        -DCMAKE_BUILD_TYPE=Release \
        -DENABLE_MPI=ON \
        -DCMAKE_INSTALL_PREFIX=/usr/local \
        .. && \
    make -j$(nproc) && \
    make install

# Clean up the build environment to reduce the final image size.
# The executable is now in /usr/local/bin, so we only need to remove the source code.
# FIX: Removed redundant 'cp' and temporary install directory cleanup steps.
RUN rm -rf /tmp/Kripke

# Set the final working directory for the container.
WORKDIR /

# Running 'kripke' without arguments will display its usage information.
CMD ["kripke"]
