# Use a standard Ubuntu base image suitable for CPU builds
FROM ubuntu:22.04

# Set non-interactive frontend to avoid prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, and certificates
# This includes git and cmake to fetch and build the source code,
# OpenMPI for parallel execution, and ca-certificates for secure connections.
# We clean up apt cache in the same layer to reduce image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized/cloud environments like Kubernetes
# - Allow MPI processes to be launched by the root user.
# - Explicitly use the TCP network interface (eth0 is standard in K8s pods).
# - Disable transports like InfiniBand (openib) that are not typically available.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_pml=ob1

# Clone the Kripke repository, build from source with MPI enabled,
# install the executable, and clean up the source code to minimize image size.
RUN git clone --depth 1 -b main https://github.com/LLNL/Kripke.git /kripke && \
    cd /kripke && \
    mkdir build && \
    cd build && \
    cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D ENABLE_MPI=ON \
      -D ENABLE_OPENMP=ON \
      .. && \
    make -j$(nproc) && \
    # Copy the final executable to a system-wide location on the PATH
    cp bin/kripke.exe /usr/local/bin/kripke && \
    # Remove the source code directory after the build is complete
    cd / && \
    rm -rf /kripke

# Set the default entrypoint to the Kripke application
ENTRYPOINT ["mpirun", "kripke"]

# Set a default command to show help, can be overridden at runtime
CMD ["--help"]
