# Base Image: Ubuntu 22.04
# A stable and widely used base image for building applications.
FROM ubuntu:22.04

# Set a non-interactive frontend for package managers to prevent prompts during build.
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments, allowing root execution which is common in containers.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install build dependencies, clone, compile, and install Kripke in a single RUN layer to optimize image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    g++ \
    gfortran \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    ca-certificates && \
    #
    # Clone the Kripke repository from the main branch.
    #
    git clone https://github.com/LLNL/Kripke.git /opt/kripke-src && \
    #
    # Create a build directory and configure the build with CMake.
    # We enable MPI support as required and specify a Release build for performance.
    #
    mkdir /opt/kripke-src/build && \
    cd /opt/kripke-src/build && \
    cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D ENABLE_MPI=ON \
        .. && \
    #
    # Compile the application using all available processor cores.
    #
    make -j$(nproc) && \
    #
    # Copy the compiled executable to a system location on the PATH and give it a standard name.
    # This ensures 'kripke' can be called directly from any location.
    #
    mv /opt/kripke-src/build/bin/kripke.exe /usr/local/bin/kripke && \
    #
    # Clean up the build source and package manager cache to reduce final image size.
    #
    rm -rf /opt/kripke-src && \
    rm -rf /var/lib/apt/lists/*

# Set a default working directory for the container.
WORKDIR /

# Set a default command to run when the container starts.
# Displaying the help message is a good default to verify the application is installed correctly.
CMD ["kripke", "--help"]
