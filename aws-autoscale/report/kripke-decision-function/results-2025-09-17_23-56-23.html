# Base image: Ubuntu 22.04
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, git, and certificates
# This single RUN command helps to reduce the number of layers in the final image.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    g++ \
    git \
    libopenmpi-dev \
    openmpi-bin && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments
# These settings ensure MPI runs over standard TCP/IP networks (like in Kubernetes)
# and allows execution as the root user.
ENV OMPI_MCA_pml=ob1
ENV OMPI_MCA_btl=self,tcp
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Set a working directory
WORKDIR /app

# Clone the Kripke repository from the main branch
RUN git clone --depth 1 -b main https://github.com/LLNL/Kripke.git /app/Kripke

# Build Kripke with MPI support
WORKDIR /app/Kripke/build
RUN cmake .. -DENABLE_MPI=ON && \
    make -j$(nproc)

# Install the kripke executable to a system location on the PATH
# Using 'mv' to place the binary, adhering to the no COPY/ADD instruction.
RUN mv /app/Kripke/build/bin/kripke.exe /usr/local/bin/kripke

# Clean up the source code to reduce image size
RUN rm -rf /app/Kripke

# Set the working directory for running the application
WORKDIR /

# Define the default command to run when the container starts
CMD ["kripke"]
