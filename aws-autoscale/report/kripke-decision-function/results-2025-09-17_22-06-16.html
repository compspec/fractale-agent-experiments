# Use a standard Ubuntu base image suitable for CPU workloads
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, OpenMPI, and certificates
# This single RUN command helps to reduce layer count
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin && \
    # Clean up apt cache to reduce image size
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings force TCP communication over the pod's network interface (typically eth0)
# and disable shared-memory mechanisms that are problematic across nodes.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_orte_allowed_exit_without_sync=1
ENV OMPI_MCA_rmaps_base_mapping_policy=slot

# Clone, build, and install the Kripke application
# This is done in a single RUN command to keep the image lean by removing build artifacts
RUN git clone https://github.com/LLNL/Kripke.git /opt/kripke && \
    cd /opt/kripke && \
    # The default branch for clone is the main/master branch
    mkdir build && \
    cd build && \
    # Configure the build with MPI enabled
    cmake .. -DENABLE_MPI=ON && \
    # Build using all available processor cores
    make -j$(nproc) && \
    # Copy the executable to a location on the system PATH
    cp bin/kripke.exe /usr/local/bin/kripke && \
    # Clean up the source code to reduce final image size
    cd / && \
    rm -rf /opt/kripke

# Set a default working directory for the container
WORKDIR /data

# Define a default command to display Kripke's help message
CMD ["kripke", "--help"]
