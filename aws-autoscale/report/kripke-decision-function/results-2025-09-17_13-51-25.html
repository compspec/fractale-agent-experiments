# Base image: Ubuntu 22.04 LTS (Jammy Jellyfish)
# A stable and widely used base for builds.
FROM ubuntu:22.04

# Prevent interactive dialogues during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build dependencies, git, cmake, OpenMPI, and certificates.
# OpenSSH server/client are included for MPI compatibility in multi-node Kubernetes deployments.
# Using --no-install-recommends to keep the image size smaller.
# System certificates are installed via ca-certificates.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    openssh-client \
    openssh-server && \
    # Clean up apt cache to reduce layer size
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Clone, build, and install the Kripke application from source.
# This entire process is done in a single RUN command to optimize layer caching
# and to adhere to the constraint of not using COPY or ADD.
RUN mkdir /build_src && \
    cd /build_src && \
    # Clone the main branch of the Kripke repository from GitHub
    git clone https://github.com/LLNL/Kripke.git && \
    cd Kripke && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake.
    # - CMAKE_BUILD_TYPE=Release: Optimize for performance.
    # - ENABLE_MPI=ON: Enable MPI support as required.
    # - CMAKE_CXX_COMPILER=mpic++: Use the MPI C++ compiler wrapper.
    cmake \
        -DCMAKE_BUILD_TYPE=Release \
        -DENABLE_MPI=ON \
        -DCMAKE_CXX_COMPILER=mpic++ \
        .. && \
    # Compile the source code using all available processor cores.
    make -j$(nproc) && \
    # Copy the final executable to a directory on the system's PATH.
    # This ensures 'kripke' can be called directly from any shell.
    cp ./bin/kripke.exe /usr/local/bin/kripke && \
    # Clean up the source and build directories to minimize final image size.
    cd / && \
    rm -rf /build_src

# Set a default working directory.
WORKDIR /

# Set the default command to execute when the container starts.
# This will display Kripke's help message, which is a good way
# to verify that the application was built and installed correctly.
CMD ["kripke", "--help"]

    # Install CMake to /usr/local, which is on the system PATH.
