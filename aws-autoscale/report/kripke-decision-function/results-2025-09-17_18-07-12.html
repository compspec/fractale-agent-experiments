# Base image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, and certificates
# This single RUN command helps to reduce layer count.
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    && rm -rf /var/lib/apt/lists/*

# Configure Open MPI for container environments
# - Allow running as root, which is common in containers.
# - Exclude the loopback and docker interfaces to prevent MPI from trying to use them for communication.
# - Set a specific mechanism for intra-node communication that is more stable in some container runtimes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0

# Clone, build, and install Kripke
# The source code is cloned directly from the main branch.
# The executable is copied to /usr/local/bin to be in the system's PATH.
# The build directory and source code are removed to keep the final image clean.
RUN git clone https://github.com/LLNL/Kripke.git /opt/kripke \
    && cd /opt/kripke \
    && mkdir build \
    && cd build \
    && cmake -DENABLE_MPI=ON .. \
    && make -j$(nproc) \
    && cp ./bin/kripke /usr/local/bin/kripke \
    && cd / \
    && rm -rf /opt/kripke

# Set a default command to show Kripke's help message
CMD ["kripke", "--help"]
