# Dockerfile for lammps-reax
# Target: AWS CPU instance in Kubernetes

# Use a standard Ubuntu base image
FROM ubuntu:22.04

# Set non-interactive frontend to avoid prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, required libraries, and certificates
# This single RUN command helps in reducing the number of layers in the image.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    wget \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI to allow running as root, which is common in containers,
# and to work with container memory sharing mechanisms.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_rmaps_base_oversubscribe=1

# Clone the latest branch of LAMMPS source code, build it with the REAXFF package,
# and install the executable to a system path.
RUN git clone https://github.com/lammps/lammps.git /opt/lammps && \
    cd /opt/lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/

# Set the default working directory for the container
WORKDIR /data

# Copy the example files for the HNS simulation into the working directory
# and then clean up the source code to reduce final image size.
RUN cp /opt/lammps/examples/reaxff/HNS/* . && \
    rm -rf /opt/lammps

# Define the entrypoint for the container. The lammps executable is now on the PATH.
ENTRYPOINT ["lmp"]

# Set a default command to print LAMMPS help, can be overridden at runtime.
CMD ["-h"]
