# Use a standard Ubuntu LTS version as the base image
# This provides a stable and well-supported environment
FROM ubuntu:22.04

# Set environment variable to prevent interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments like Kubernetes
# OMPI_MCA_btl_vader_single_copy_mechanism=none: Disables a shared-memory optimization that can cause issues in containers.
# OMPI_MCA_rmaps_base_oversubscribe=yes: Allows MPI processes to be bound to the same core, which is common in oversubscribed Kubernetes nodes.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_rmaps_base_oversubscribe=yes

# Set the default working directory for subsequent commands
WORKDIR /app

# Combine installation, build, and cleanup into a single RUN layer to reduce image size
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    libopenmpi-dev \
    openmpi-bin \
    fftw3-dev \
    libgomp1 \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/* \
    # Clone the latest branch of the LAMMPS source code from GitHub
    && git clone https://github.com/lammps/lammps.git /lammps \
    # As per instructions, copy example files for the reaxff/HNS benchmark into the WORKDIR
    && cp /lammps/examples/reaxff/HNS/* . \
    # Create a build directory and navigate into it
    && cd /lammps && mkdir build && cd build \
    # Configure the LAMMPS build using CMake
    # -D PKG_REAXFF=on: Enables the ReaxFF package required for the simulation
    && cmake -D PKG_REAXFF=on ../cmake \
    # Compile the source code using all available processor cores
    && make -j$(nproc) \
    # Copy the compiled LAMMPS executable 'lmp' to a system path
    && cp lmp /usr/local/bin/ \
    # Clean up the source code directory to minimize final image size
    && rm -rf /lammps

# Set a default command to run the LAMMPS executable with a typical input file for the copied example
CMD ["lmp", "-in", "in.hns"]
