# Base image with OpenMPI, Flux, and AWS EFA support for ARM64
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, clone, build, and install LAMMPS
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    git \
    gfortran && \
    # Create a source directory
    mkdir -p /opt/src && \
    cd /opt/src && \
    # Clone the stable branch of LAMMPS for a production build
    git clone --depth 1 -b stable https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake
    # - PKG_REAXFF=on: Enable the ReaxFF package which requires a Fortran compiler
    # - BUILD_MPI=on: Enable MPI support
    # - CMAKE_BUILD_TYPE=Release: Build with optimizations
    # - CMAKE_C_COMPILER=mpicc / CMAKE_CXX_COMPILER=mpicxx: Use MPI compiler wrappers for multi-node compatibility
    cmake ../cmake \
    -D PKG_REAXFF=on \
    -D BUILD_MPI=on \
    -D CMAKE_BUILD_TYPE=Release \
    -D CMAKE_C_COMPILER=mpicc \
    -D CMAKE_CXX_COMPILER=mpicxx && \
    # Compile LAMMPS using all available cores
    make -j$(nproc) && \
    # Copy the lammps executable to a location on the system PATH
    cp src/lmp /usr/local/bin/ && \
    # Create the working directory for the application
    mkdir -p /opt/lammps-run && \
    # Copy the example files into the new working directory
    cp /opt/src/lammps/examples/reaxff/HNS/* /opt/lammps-run/ && \
    # Clean up the build directory and package cache to reduce image size
    cd / && \
    rm -rf /opt/src && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/* && \
    # Verify that the lammps executable is in the PATH
    which lmp

# Set the default working directory for the container
WORKDIR /opt/lammps-run

# Define the default command to run when the container starts
CMD ["/bin/bash"]
