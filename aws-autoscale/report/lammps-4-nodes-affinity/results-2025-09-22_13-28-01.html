# Base image with OpenMPI, Flux, and AWS EFA support for ARM
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Prevents interactive prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies for LAMMPS
# build-essential: C/C++ compilers and make
# cmake: Build system generator for LAMMPS
# git: To clone the LAMMPS source code
# wget: General utility, sometimes needed for fetching dependencies
# libfftw3-dev: Required for Fast Fourier Transform support in LAMMPS (e.g., for KSPACE package)
# ca-certificates: For secure network connections (e.g., git clone)
RUN apt-get update && \
    apt-get install -y \
    build-essential \
    cmake \
    git \
    wget \
    libfftw3-dev \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest development branch of the LAMMPS source code
RUN git clone --depth 1 -b develop https://github.com/lammps/lammps.git /opt/lammps

# Configure and build LAMMPS using CMake
# - We create a build directory as is standard practice with CMake
# - CMAKE_C_COMPILER and CMAKE_CXX_COMPILER are set to the MPI wrappers provided by the base image to ensure MPI support.
# - PKG_REAXFF=ON: Enables the ReaxFF package as requested.
# - PKG_KSPACE=ON: Enables long-range electrostatics (e.g., PPPM), a common requirement.
# - PKG_MOLECULE=ON: Enables molecular dynamics simulations.
# - PKG_RIGID=ON: Enables simulations with rigid bodies.
# - FFT=FFTW3: Links against the FFTW3 library we installed.
# - The build is parallelized using make -j$(nproc) for speed.
RUN cd /opt/lammps && \
    mkdir build && \
    cd build && \
    cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D CMAKE_C_COMPILER=mpicc \
      -D CMAKE_CXX_COMPILER=mpicxx \
      -D PKG_REAXFF=ON \
      -D PKG_KSPACE=ON \
      -D PKG_MOLECULE=ON \
      -D PKG_RIGID=ON \
      -D FFT=FFTW3 \
      ../cmake && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable to a system-wide location on the PATH
RUN cp /opt/lammps/build/lmp /usr/local/bin/

# Verify that the lmp executable is in the PATH
RUN which lmp

# Set a default working directory for running simulations
WORKDIR /workspace

# Copy the HNS ReaxFF example files into the working directory to be readily available for test runs
RUN cp -r /opt/lammps/examples/reaxff/HNS/* .

# Set the default command to run a shell
CMD ["/bin/bash"]
