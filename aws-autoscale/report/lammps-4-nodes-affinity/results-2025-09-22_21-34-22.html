dockerfile
# Base image with OpenMPI, Flux, and EFA support for AWS on ARM64
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, clone LAMMPS, build with REAXFF package, install, and clean up
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    wget && \
    # Create a temporary build directory
    mkdir -p /tmp/build && \
    cd /tmp/build && \
    # Clone the latest version of the LAMMPS source code
    git clone https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && cd build && \
    # Configure the build with CMake
    # - BUILD_MPI=yes: Enable MPI support for multi-node execution
    # - PKG_REAXFF=yes: Include the ReaxFF package as requested
    # - CMAKE_C_COMPILER/CMAKE_CXX_COMPILER: Use the MPI wrappers from the base image
    # - CMAKE_INSTALL_PREFIX: Set the installation path for binaries
    cmake -D CMAKE_INSTALL_PREFIX=/usr/local \
          -D BUILD_MPI=yes \
          -D PKG_REAXFF=yes \
          -D CMAKE_C_COMPILER=mpicc \
          -D CMAKE_CXX_COMPILER=mpicxx \
          ../cmake && \
    # Compile using all available processor cores
    make -j$(nproc) && \
    # Install the 'lmp' executable to /usr/local/bin, which is on the system PATH
    make install && \
    # Create the final working directory for the application
    mkdir -p /opt/lammps-reax && \
    # Copy the required example files to the working directory
    cp /tmp/build/lammps/examples/reaxff/HNS/* /opt/lammps-reax/ && \
    # Clean up the build directory and package cache to reduce image size
    cd / && \
    rm -rf /tmp/build && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Verify that the LAMMPS executable is in the system PATH
RUN which lmp

# Set the default working directory for the container
WORKDIR /opt/lammps-reax

# Set a default command to start a shell
CMD ["/bin/bash"]

# This MiniCluster manifest is designed for running an HPC application (LAMMPS)
# on a single AWS ARM-based CPU node with EFA networking, managed by the Flux Operator.
apiVersion: flux-framework.org/v1alpha2
kind: MiniCluster
metadata:
  name: lammps-reax-aws-cpu
  namespace: default
spec:
  # The number of pods in the MiniCluster, set to 1 for single-node testing.
  size: 1
  # The total number of MPI tasks to run. Corresponds to the new command parameters (2*2*2=8).
  tasks: 8

  # Do not retry the job on failure.
  backoffLimit: 1

  # Keep pods after the job finishes for debugging and log inspection.
  cleanup: false

  # Enable strict mode for logging, which provides more detailed output.
  logging:
    strict: true

  # Pod-specific configurations are nested under the 'pod' field.
  pod:
    # Set the restart policy to Never, which is typical for batch jobs.
    restartPolicy: Never
    # Add a nodeSelector to ensure pods are scheduled on specific AWS instance types.
    nodeSelector:
      node.kubernetes.io/instance-type: m7g.16xlarge

  # Flux Framework specific configurations
  flux:
    # Specify the architecture for ARM instances.
    arch: arm
    # Add Flux option flags to control CPU affinity for each task.
    optionFlags: -o cpu-affinity=per-task
    # Flux container settings
    container:
      # Use the ARM-specific image for the Flux view.
      image: ghcr.io/converged-computing/flux-view-ubuntu:arm-jammy
      # Since Flux is expected to be included in the main application container,
      # we disable the automatic injection of the Flux sidecar container.
      disable: true

  # Define the main application container
  containers:
    - image: ghcr.io/converged-computing/fractale-agent-experiments:lammps-reax
      imagePullPolicy: Always

      # The command to be executed by Flux for a smaller, single-node test case.
      command: lmp -v x 2 -v y 2 -v z 2 -in ./in.reaxff.hns -nocite

      # This is the main application container, not a launcher pod.
      launcher: false

      # Resource requests and limits required for AWS Elastic Fabric Adapter (EFA).
      # This ensures the pod is scheduled on a node with an available EFA device.
      resources:
        limits:
          vpc.amazonaws.com/efa: "1"
        requests:
          vpc.amazonaws.com/efa: "1"
