dockerfile
# Base image contains OpenMPI, Flux, and AWS EFA drivers for ARM architecture
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Set working directory for the application files and build process
WORKDIR /opt/lammps-reax

# Install dependencies, clone, build, and install LAMMPS with ReaxFF in a single layer
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    ca-certificates \
    cmake \
    fftw3-dev \
    git \
    && \
    # Clone the latest version of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git /tmp/lammps-src \
    && cd /tmp/lammps-src \
    && mkdir build && cd build \
    && \
    # Configure the build with CMake
    # - Use MPI wrappers as the C and CXX compilers for multi-node compatibility
    # - Enable the REAXFF package required for the simulation
    cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=on \
        -D CMAKE_C_COMPILER=mpicc \
        -D CMAKE_CXX_COMPILER=mpicxx \
        -D PKG_REAXFF=on \
        ../cmake \
    && \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) \
    && \
    # Copy the main LAMMPS executable to a system-wide location on the PATH
    cp lmp /usr/local/bin/ \
    && \
    # Return to the designated WORKDIR
    cd /opt/lammps-reax \
    && \
    # Copy the specific example files into the WORKDIR
    cp /tmp/lammps-src/examples/reaxff/HNS/* . \
    && \
    # Clean up the build directory and source code to reduce image size
    rm -rf /tmp/lammps-src \
    && \
    # Clean up apt cache
    rm -rf /var/lib/apt/lists/* \
    && \
    # Verify that the 'lmp' executable is correctly installed and available on the PATH
    which lmp

# Set the entrypoint to the LAMMPS executable
ENTRYPOINT ["lmp"]

# Default command can be overridden at runtime, e.g., with mpirun arguments
CMD ["-in", "in.hns"]
