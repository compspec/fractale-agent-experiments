# Base image with Flux, OpenMPI, and AWS EFA support for ARM
FROM ghcr.io/converged-computing/flux-openmpi:ubuntu2204-efa-arm

# Set non-interactive frontend for package management
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, certificates, and clean up
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Set a temporary directory for building the application
WORKDIR /build

# Clone the latest version of LAMMPS, build with REAXFF package, and install
# The C and CXX compilers are explicitly set to the MPI wrappers provided by the base image
# to ensure it is built for a multi-node MPI environment.
RUN git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    cmake \
        -D CMAKE_C_COMPILER=mpicc \
        -D CMAKE_CXX_COMPILER=mpicxx \
        -D PKG_REAXFF=ON \
        -D BUILD_MPI=yes \
        -D CMAKE_INSTALL_PREFIX=/usr/local \
        ../cmake && \
    make -j$(nproc) && \
    make install

# Set the final working directory for the container
WORKDIR /app

# Copy the example files for the reaxff HNS benchmark into the working directory
RUN cp /build/lammps/examples/reaxff/HNS/* .

# Clean up the build directory to reduce final image size
RUN rm -rf /build

# Verify that the LAMMPS executable is in the system PATH
RUN which lmp
