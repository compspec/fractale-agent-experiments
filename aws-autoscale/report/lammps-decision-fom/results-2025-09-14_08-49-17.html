# Use Ubuntu 22.04 LTS as the base image for a stable and well-supported environment.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install build tools, dependencies, and required libraries.
# - build-essential, cmake, git: Core tools for compiling LAMMPS from source.
# - openmpi-bin, libopenmpi-dev: OpenMPI for parallel processing, essential for HPC.
# - libgomp1: Required for OpenMP support.
# - libfftw3-dev: FFTW library to accelerate long-range electrostatics calculations (KSPACE).
# - ca-certificates: Required for secure network communication (e.g., for git).
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    libfftw3-dev \
    ca-certificates && \
    # Clean up apt cache to reduce image size.
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest stable branch of the LAMMPS source code.
# A shallow clone (--depth 1) is used to minimize the download size.
RUN git clone -b stable --depth 1 https://github.com/lammps/lammps.git /lammps

# Create a build directory and switch to it, following cmake best practices.
WORKDIR /lammps/build

# Configure the LAMMPS build using cmake.
# - CMAKE_BUILD_TYPE=Release: Enables optimizations for production performance.
# - BUILD_MPI=yes: Enables MPI support.
# - PKG_REAXFF=ON: Enables the ReaxFF package as required for the 'lammps-reax' application.
# - PKG_KSPACE=ON: Enables KSPACE for long-range electrostatics (e.g., PPPM).
# - PKG_MOLECULE=ON: A core package required by many force fields.
RUN cmake ../cmake \
    -D CMAKE_BUILD_TYPE=Release \
    -D BUILD_MPI=yes \
    -D PKG_REAXFF=ON \
    -D PKG_KSPACE=ON \
    -D PKG_MOLECULE=ON

# Compile LAMMPS using the number of available processor cores to speed up the build.
RUN make -j$(nproc)

# Copy the compiled LAMMPS executable 'lmp' to a standard system location on the PATH.
# This makes 'lmp' directly invokable from any directory.
RUN cp /lammps/build/lmp /usr/local/bin/

# Set the final working directory for running simulations.
WORKDIR /app

# Copy the ReaxFF HNS (hexanitrostilbene) example files into the working directory.
RUN cp /lammps/examples/reaxff/HNS/* /app/

# Remove the LAMMPS source code directory to significantly reduce the final image size.
RUN rm -rf /lammps

# Configure OpenMPI for containerized environments like Kubernetes.
# These settings ensure MPI communicates over the container's TCP network interface (e.g., eth0)
# and avoids transports like InfiniBand or shared memory that are unsuitable for cross-pod communication.
ENV OMPI_MCA_btl=^openib
ENV OMPI_MCA_btl_tcp_if_include=eth0
ENV OMPI_MCA_orte_base_help_aggregate=0
