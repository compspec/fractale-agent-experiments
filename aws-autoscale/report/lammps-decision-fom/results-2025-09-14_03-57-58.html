# Base image: Ubuntu 22.04 LTS (Jammy Jellyfish)
# A stable and widely used base for building applications.
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, LAMMPS dependencies, clone, build, and clean up.
# This is all done in a single RUN layer to optimize image size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    libgomp1 \
    # Clone the latest branch of the LAMMPS source code.
    && git clone --depth 1 https://github.com/lammps/lammps.git /tmp/lammps \
    # Create a build directory and enter it.
    && cd /tmp/lammps \
    && mkdir build && cd build \
    # Configure the build with CMake.
    # We enable MPI support (BUILD_MPI) and the REAXFF package (PKG_REAXFF) for this build.
    && cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on \
    # Compile LAMMPS using all available processor cores.
    && make -j$(nproc) \
    # Copy the compiled LAMMPS executable to a system-wide location on the PATH.
    && cp lmp /usr/local/bin/lmp \
    # Create the application working directory.
    && mkdir -p /app \
    # Copy the HNS example files for the reaxff potential into the working directory.
    && cp /tmp/lammps/examples/reaxff/HNS/* /app/ \
    # Clean up the build environment to reduce the final image size.
    && rm -rf /tmp/lammps \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

# Configure OpenMPI for running in containerized environments like Kubernetes.
# These settings prevent MPI from trying to use shared memory mechanisms that
# are not available across different nodes or pods and exclude loopback/docker network interfaces.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_pml=ob1
ENV OMPI_MCA_btl=^openib

# Set the final working directory for the container.
WORKDIR /app

# Define a default command to be executed when the container starts.
# This can be overridden at runtime.
CMD ["/bin/bash"]
