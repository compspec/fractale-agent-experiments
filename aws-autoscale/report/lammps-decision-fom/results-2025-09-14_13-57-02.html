# Base Image: Ubuntu 22.04 LTS
# Provides a stable and widely supported foundation.
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies for building and running LAMMPS with MPI
# Includes build tools, cmake, git, OpenMPI, the required libgomp1, and ca-certificates.
# We clean up the apt cache in the same layer to reduce image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Clone, build, and install LAMMPS in a single layer to optimize image size
# This multistep process does the following:
# 1. Clones the latest branch of the LAMMPS source code from GitHub.
# 2. Creates a build directory and navigates into it.
# 3. Configures the build using CMake, specifically enabling the REAXFF package and MPI support.
# 4. Compiles the source code using all available processor cores.
# 5. Copies the compiled 'lmp' executable to a system location on the PATH.
# 6. Creates the final application work directory '/app'.
# 7. Copies the required example files from the cloned source into '/app'.
# 8. Removes the source code to minimize the final image size.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps && \
    cd /lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=on && \
    make -j$(nproc) && \
    cp ./lmp /usr/local/bin/lmp && \
    mkdir -p /app && \
    cp /lammps/examples/reaxff/HNS/* /app/ && \
    rm -rf /lammps

# Set the default working directory for the container
WORKDIR /app

# Configure OpenMPI for containerized environments
# These settings are crucial for running MPI applications as root within
# containers, preventing attempts to use inappropriate network interfaces
# like the Docker bridge or localhost for inter-process communication.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0

# Set a default command to start a shell, useful for interactive sessions
CMD ["/bin/bash"]
