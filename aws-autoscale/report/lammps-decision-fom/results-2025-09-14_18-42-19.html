# Use Ubuntu 22.04 LTS as the base image, a common choice for cloud environments.
FROM ubuntu:22.04

# Set the DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation.
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies for LAMMPS, and MPI.
# This includes ca-certificates for secure network connections (e.g., git clone)
# and libgomp1 as explicitly required.
# We clean up the apt cache in the same layer to reduce image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI to allow running as root, which is necessary in many
# container orchestrators like Kubernetes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Set the final working directory for the application.
WORKDIR /app

# This single RUN command performs all build steps to minimize layers.
# 1. Creates a temporary build directory.
# 2. Clones the latest LAMMPS source code from the default 'develop' branch.
# 3. Configures the build with CMake, enabling the required REAXFF and MPI packages.
# 4. Compiles the source code using all available processor cores.
# 5. Copies the 'lmp' executable to /usr/local/bin, making it available on the system's PATH.
# 6. Copies the required example files into the WORKDIR.
# 7. Cleans up the temporary build directory to minimize the final image size.
RUN mkdir -p /build && cd /build && \
    git clone --depth 1 https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=on \
        -D PKG_REAXFF=on \
        -D PKG_KIM=on && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/lmp && \
    cp /build/lammps/examples/reaxff/HNS/* /app/ && \
    rm -rf /build

# Set the LAMMPS executable as the default entrypoint.
# This allows running simulations by passing arguments directly to the container,
# e.g., `docker run <image_name> -in in.hns`.
ENTRYPOINT ["lmp"]
