# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to avoid prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Configure Open MPI for containerized environments
# Allow running as root and enable oversubscription for typical Kubernetes/Docker environments
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
ENV OMPI_MCA_rmaps_base_oversubscribe=1
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none

# Install build dependencies, MPI, required libraries, and certificates
# This is done in a single RUN layer to reduce image size
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    gfortran \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    ca-certificates && \
    # Clean up apt cache to reduce image size
    rm -rf /var/lib/apt/lists/*

# Clone, build, and install LAMMPS with the REAXFF package in a single layer to optimize image size
RUN set -e && \
    # Clone the latest stable branch of LAMMPS source code
    git clone --depth 1 -b stable https://github.com/lammps/lammps.git /tmp/lammps && \
    cd /tmp/lammps && \
    # Create a build directory
    mkdir build && cd build && \
    # Configure the build with the REAXFF package and MPI support for a CPU environment
    cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on \
        ../cmake && \
    # Build LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the compiled lmp executable to a system location on the PATH
    cp lmp /usr/local/bin/lmp && \
    # Create the workspace directory and copy the required example files
    mkdir -p /workspace && \
    cp /tmp/lammps/examples/reaxff/HNS/* /workspace/ && \
    # Clean up by removing the source code directory
    rm -rf /tmp/lammps

# Set the working directory for the container
WORKDIR /workspace

# Set a default command to start a bash session
CMD ["/bin/bash"]
