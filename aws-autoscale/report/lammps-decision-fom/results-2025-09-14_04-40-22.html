# Use a standard Ubuntu base image
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI libraries, FFTW, libgomp1, and certificates
# This is done in a single RUN layer to reduce image size.
# We do not use --purge and clean up the apt cache manually.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    libfftw3-dev \
    libgomp1 \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest branch of the LAMMPS source code from the official repository
# We use a temporary location for the build process.
RUN git clone https://github.com/lammps/lammps.git /tmp/lammps

# Create a build directory and enter it
WORKDIR /tmp/lammps/build

# Configure the LAMMPS build using CMake.
# We explicitly enable the REAXFF package and MPI support for CPU execution.
# Then, compile the source code using all available processor cores.
RUN cmake -D PKG_REAXFF=on -D BUILD_MPI=yes ../cmake && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable to a standard system path
# This ensures 'lmp' is available globally without modifying PATH.
RUN cp /tmp/lammps/build/lmp /usr/local/bin/lmp

# Create and set the final runtime working directory
WORKDIR /workspace

# Copy the specific example files required for 'lammps-reax' into the WORKDIR
RUN cp /tmp/lammps/examples/reaxff/HNS/* /workspace/

# Clean up the source code and build artifacts to reduce final image size
RUN rm -rf /tmp/lammps

# Configure OpenMPI for optimal performance and compatibility within containerized environments
# This prevents issues with shared memory mechanisms (vader) and disables hardware-specific
# transports like InfiniBand (openib) that are not typically available in containers.
# It also explicitly allows MPI to run as the root user.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none \
    OMPI_MCA_btl=^openib \
    OMPI_MCA_pml=ob1 \
    OMPI_ALLOW_RUN_AS_ROOT=1 \
    OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Provide a default command to start an interactive shell when the container runs
# This allows users to easily attach and execute their LAMMPS simulations.
CMD ["/bin/bash"]
