# Specifies the base image for the build. Ubuntu 22.04 is a modern, long-term support release.
FROM ubuntu:22.04

# Set environment variables to enable non-interactive installation of packages.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI implementation, and other required utilities.
# - build-essential: Compiler toolchain (gcc, g++, make).
# - cmake: Required for the LAMMPS build system.
# - git: To clone the LAMMPS source code from its repository.
# - openmpi-bin & libopenmpi-dev: Open MPI runtime and development files.
# - ca-certificates: To allow secure network connections (e.g., for git).
# - libgomp1: GNU OpenMP library, often a dependency for HPC applications.
# The apt cache is cleaned in the same layer to reduce final image size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI to allow execution as the root user. This is a common requirement
# for running MPI applications within container environments like Docker and Kubernetes.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# This single RUN command encapsulates the entire build process to optimize layer usage.
# 1. Clones the latest revision of the LAMMPS source code.
# 2. Creates a build directory and configures the build using CMake, enabling MPI, OpenMP, and the REAXFF package.
# 3. Compiles the source code using all available processor cores.
# 4. Copies the final 'lmp' executable to a system location on the PATH.
# 5. Creates the final workspace directory.
# 6. Copies the required example files from the source tree into the workspace.
# 7. Cleans up the cloned source code to minimize the final image size.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /tmp/lammps-src && \
    cd /tmp/lammps-src && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=on \
        -D BUILD_OMP=on \
        -D PKG_REAXFF=on && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/ && \
    mkdir -p /opt/workspace && \
    cp /tmp/lammps-src/examples/reaxff/HNS/* /opt/workspace/ && \
    rm -rf /tmp/lammps-src

# Set the final working directory for the container.
WORKDIR /opt/workspace

# Defines the default command to be executed when the container starts.
# Displaying the help message confirms that the 'lmp' executable is in the PATH and operational.
CMD ["lmp", "-h"]
