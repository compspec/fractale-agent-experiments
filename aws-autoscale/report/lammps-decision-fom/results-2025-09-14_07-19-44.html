# Base Image: Ubuntu 22.04 LTS for a stable and widely supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install dependencies required for building and running LAMMPS with MPI.
# This includes build-essential tools, cmake, git for source control,
# OpenMPI for parallel execution, gfortran for certain LAMMPS packages,
# libgomp1 as requested, and ca-certificates for secure network connections.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    gfortran \
    libopenmpi-dev \
    openmpi-bin \
    libgomp1 \
    ca-certificates && \
    rm -rf /var/lib/apt/lists/*

# Clone the latest version of the LAMMPS source code directly from the official repository.
# Using --depth 1 creates a shallow clone, reducing image size by not fetching the entire commit history.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /lammps

# Create a build directory and set it as the working directory.
WORKDIR /lammps/build

# Configure the LAMMPS build using CMake.
# -D PKG_REAXFF=on: Enables the REAXFF package required for the 'reax' potential.
# -D BUILD_MPI=on: Enables MPI support for parallel runs in containerized environments like Kubernetes.
RUN cmake \
    -D PKG_REAXFF=on \
    -D BUILD_MPI=on \
    ../cmake

# Compile LAMMPS using all available processor cores to speed up the build process.
RUN make -j$(nproc)

# Copy the compiled LAMMPS executable 'lmp' to a directory in the system's PATH.
# This makes the 'lmp' command globally available without needing to specify its full path.
RUN cp /lammps/build/lmp /usr/local/bin/

# Create a dedicated working directory for running simulations.
WORKDIR /workspace

# Copy the HNS reaxff example files from the cloned source into the working directory.
# This makes it easy to run a benchmark or test simulation upon starting the container.
RUN cp /lammps/examples/reaxff/HNS/* /workspace/

# Set the default command to launch a bash shell.
# This allows for interactive use and easy execution of custom run scripts in the pipeline.
CMD ["/bin/bash"]
