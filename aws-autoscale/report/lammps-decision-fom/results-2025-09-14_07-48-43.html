# Base Image: Ubuntu 22.04 LTS
# Using a specific LTS version for reproducibility and stability.
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to prevent prompts during build.
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI libraries, git, certificates, and OpenMP runtime.
# - build-essential & cmake: For compiling LAMMPS from source.
# - git: For cloning the LAMMPS repository.
# - ca-certificates: For secure git/wget operations.
# - openmpi-bin & libopenmpi-dev: For MPI support, crucial for HPC applications.
# - libgomp1: GNU OpenMP runtime library, often needed for parallel execution on CPUs.
# We clean up the apt cache in the same layer to reduce final image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Clone the LAMMPS source code into a temporary directory.
# Using --depth 1 for a shallow clone to save space and time.
# The 'stable' branch is chosen for a production-ready build.
RUN git clone --depth 1 --branch stable https://github.com/lammps/lammps.git /tmp/lammps

# Configure, build, and install LAMMPS.
# A separate build directory is the standard out-of-source build practice.
# -D PKG_REAXFF=on: Enables the ReaxFF package as required for 'lammps-reax'.
# -D BUILD_MPI=on: Compiles LAMMPS with MPI support for distributed computing.
RUN cd /tmp/lammps && \
    mkdir build && \
    cd build && \
    cmake \
      -D PKG_REAXFF=on \
      -D BUILD_MPI=on \
      ../cmake && \
    # Build using all available processor cores to speed up compilation.
    make -j$(nproc) && \
    # Copy the final executable to a system-wide location on the PATH.
    # This makes 'lmp' directly accessible from any directory.
    cp lmp /usr/local/bin/

# Set up the default working directory for the container.
WORKDIR /app

# Copy the example files for the HNS ReaxFF potential into the WORKDIR.
# This makes the container ready to run the HNS example simulation immediately.
RUN cp /tmp/lammps/examples/reaxff/HNS/* /app/

# Clean up the source code and build artifacts to minimize the final image size.
RUN rm -rf /tmp/lammps

# Set the default command. This provides a default behavior for the container.
# It can be easily overridden at runtime, e.g., 'docker run <image> lmp -in in.hns'.
# For MPI execution in a containerized environment, use 'mpirun', for example:
# mpirun -np 4 --allow-run-as-root lmp -in in.hns
CMD ["lmp", "--help"]
