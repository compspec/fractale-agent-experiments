# Base image: Ubuntu 22.04 LTS for a stable, widely supported environment
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, runtime libraries, and certificates.
# Then, clone, build, and install LAMMPS in a single RUN layer to optimize image size.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
        build-essential \
        cmake \
        git \
        ca-certificates \
        openmpi-bin \
        libopenmpi-dev \
        libgomp1 && \
    # Clone the latest version of the LAMMPS source code
    git clone --depth 1 https://github.com/lammps/lammps.git /usr/local/src/lammps && \
    # Create a build directory and configure the build with CMake
    # We enable the REAXFF package and ensure MPI support is built
    cd /usr/local/src/lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
        -D PKG_REAXFF=on \
        -D BUILD_MPI=on && \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the compiled LAMMPS executable to a directory on the system's PATH
    # This fulfills the requirement to make 'lmp' directly available
    cp lmp /usr/local/bin/ && \
    # Create the application working directory
    mkdir -p /app && \
    # Copy the specified example files into the application working directory
    cp /usr/local/src/lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up to reduce final image size
    # Remove the cloned source code directory
    cd / && rm -rf /usr/local/src/lammps && \
    # Remove build-time packages that are not needed for runtime
    apt-get purge -y --auto-remove build-essential cmake git libopenmpi-dev && \
    # Clear the apt cache
    rm -rf /var/lib/apt/lists/*

# Set environment variables for running Open MPI in a containerized environment.
# These settings are crucial for compatibility and performance in platforms like Kubernetes.
# Allow running MPI processes as the root user
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1
# Use TCP for inter-process communication, suitable for standard container networking
ENV OMPI_MCA_btl=tcp,self
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
# Specify the network interface to be used by MPI, typically 'eth0' in Kubernetes pods
ENV OMPI_MCA_btl_tcp_if_include=eth0

# Set the default working directory for the container
WORKDIR /app

# Define a default command to facilitate easy interaction with the container
CMD ["/bin/bash"]
