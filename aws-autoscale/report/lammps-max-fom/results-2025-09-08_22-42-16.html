# Base Image: Ubuntu 22.04 LTS for a stable, recent environment
FROM ubuntu:22.04

# Set non-interactive frontend for package management
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, certificates, and other required libraries
# This includes:
# - build-essential, cmake, gfortran, git: For compiling LAMMPS from source
# - openmpi-bin, libopenmpi-dev: Open MPI for parallel execution
# - ca-certificates, wget: For secure network operations
# - libgomp1: As required for OpenMP support
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    gfortran \
    git \
    wget \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Configure Open MPI for containerized environments
# This helps prevent MPI from using loopback or docker-internal network interfaces,
# ensuring it communicates over the primary container network (e.g., Kubernetes pod network).
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl_tcp_if_exclude=lo,docker0
ENV OMPI_MCA_pml=ob1
ENV OMPI_MCA_btl=^openib

# Set a temporary directory for the build process
WORKDIR /build

# Clone the latest branch of the LAMMPS repository, build it with the REAXFF package,
# and install the executable to a system-wide location on the PATH.
RUN git clone https://github.com/lammps/lammps.git && \
    cd lammps && \
    mkdir build && \
    cd build && \
    cmake ../cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=on && \
    make -j$(nproc) && \
    cp lmp /usr/local/bin/lmp

# Set the final working directory for the application
WORKDIR /app

# Copy the example input files from the cloned source into the working directory
RUN cp /build/lammps/examples/reaxff/HNS/* .

# Clean up the build directory to reduce final image size
RUN rm -rf /build

# Set the default command to start a bash session in the container
CMD ["/bin/bash"]
