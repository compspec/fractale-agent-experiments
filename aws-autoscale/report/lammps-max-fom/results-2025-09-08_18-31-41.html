# Base image: Ubuntu 22.04 LTS for a stable, well-supported environment
FROM ubuntu:22.04

# Set DEBIAN_FRONTEND to noninteractive to prevent prompts during package installation
ENV DEBIAN_FRONTEND=noninteractive

# Install all necessary dependencies in a single RUN layer to reduce image size.
# This includes:
# - ca-certificates: For secure network communication.
# - Build tools: git, cmake, build-essential, gfortran for compiling C++/Fortran code.
# - MPI libraries: openmpi-bin and libopenmpi-dev for parallel execution.
# - LAMMPS dependencies: libfftw3-dev for Fast Fourier Transforms.
# - Explicit requirement: libgomp1 for OpenMP support.
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    ca-certificates \
    git \
    cmake \
    make \
    build-essential \
    gfortran \
    libfftw3-dev \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    # Clean up apt cache to keep the image lean
    rm -rf /var/lib/apt/lists/*

# Define a working directory for the application files
WORKDIR /app

# Clone the latest branch of LAMMPS, build it with required packages,
# install the executable, copy example files, and clean up the source code.
# This is all done in a single RUN command to minimize layer count and image size.
RUN git clone --depth 1 https://github.com/lammps/lammps.git /tmp/lammps-src && \
    cd /tmp/lammps-src && \
    mkdir build && \
    cd build && \
    # Configure the build with CMake
    # - PKG_REAXFF=on: Enables the ReaxFF package.
    # - BUILD_MPI=on: Builds the MPI-enabled executable.
    # - CMAKE_BUILD_TYPE=Release: Optimizes the build for performance.
    cmake ../cmake \
      -D PKG_REAXFF=on \
      -D BUILD_MPI=on \
      -D CMAKE_BUILD_TYPE=Release && \
    # Compile using all available processor cores
    make -j$(nproc) && \
    # Copy the compiled LAMMPS executable to a system location on the PATH
    cp lmp /usr/local/bin/ && \
    # Copy the required example files into the WORKDIR
    cp /tmp/lammps-src/examples/reaxff/HNS/* /app/ && \
    # Clean up the source and build directories to reduce final image size
    rm -rf /tmp/lammps-src

# The final WORKDIR is set to /app, where the input files are located.
# Users can mount a volume here or build upon this image to add their own input scripts.
# To run, use a command like: `mpirun -np <cores> lmp -in <your_input_script.in>`
WORKDIR /app
