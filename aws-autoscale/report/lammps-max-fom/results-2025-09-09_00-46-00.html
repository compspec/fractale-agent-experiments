# Base Image: Ubuntu 22.04 LTS (Jammy Jellyfish)
# Provides a stable, modern OS environment.
FROM ubuntu:22.04

# Set non-interactive frontend for package installations to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install essential build tools, dependencies, MPI, and certificates
# - build-essential: Compiler suite (gcc, g++, make)
# - cmake: Build system generator for LAMMPS
# - git: For cloning the source code
# - ca-certificates: For secure network connections (e.g., git clone)
# - openmpi-bin, libopenmpi-dev: OpenMPI runtime and development libraries
# - libgomp1: GNU OpenMP runtime library, an explicit requirement
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    ca-certificates \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/*

# Set the default working directory for the application runtime
WORKDIR /app

# Clone the latest stable branch of LAMMPS, build it with the REAXFF package,
# install the executable, copy example files, and clean up source code.
# This is done in a single RUN layer to optimize the final image size.
RUN git clone --depth 1 -b stable https://github.com/lammps/lammps.git /usr/local/src/lammps && \
    cd /usr/local/src/lammps/build && \
    cmake ../cmake \
        -D PKG_REAXFF=on \
        -D BUILD_MPI=yes && \
    make -j$(nproc) && \
    # Copy the LAMMPS executable to a location on the system PATH
    cp lmp /usr/local/bin/lmp && \
    # Copy the required example files to the final working directory
    cp /usr/local/src/lammps/examples/reaxff/HNS/* /app/ && \
    # Clean up the source repository to reduce image size
    rm -rf /usr/local/src/lammps

# Set the final working directory for container execution
WORKDIR /app
