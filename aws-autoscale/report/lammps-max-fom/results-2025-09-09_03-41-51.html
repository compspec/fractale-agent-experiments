# Dockerfile for lammps-reax on a CPU instance
# Base Image: Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package installations
ENV DEBIAN_FRONTEND=noninteractive

# Configure OpenMPI for containerized environments
# This prevents attempts to use shared memory or high-speed interconnects
# that are typically not available in standard container runtimes.
# It also allows running as root, which is common in containers.
ENV OMPI_MCA_btl_vader_single_copy_mechanism=none
ENV OMPI_MCA_btl=^openib,vader
ENV OMPI_MCA_pml=ob1
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Install dependencies, clone, build LAMMPS with REAXFF, copy artifacts, and clean up in a single layer
RUN apt-get update && apt-get install -y \
    build-essential \
    cmake \
    git \
    ca-certificates \
    libfftw3-dev \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 \
    && rm -rf /var/lib/apt/lists/* \
    # Create a temporary directory for the build
    && mkdir -p /tmp/build \
    && cd /tmp/build \
    # Clone the latest stable branch of the LAMMPS repository
    && git clone --depth 1 -b stable https://github.com/lammps/lammps.git lammps \
    && cd lammps \
    && mkdir build \
    && cd build \
    # Configure the build with CMake, enabling the REAXFF package
    && cmake ../cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D BUILD_OMP=yes \
        -D PKG_REAXFF=yes \
    # Build LAMMPS using all available processor cores
    && make -j$(nproc) \
    # Copy the main LAMMPS executable to a system-wide location on the PATH
    && cp lmp /usr/local/bin/lmp \
    # Create the final working directory for the application
    && mkdir -p /app \
    # Copy the required example files into the working directory
    && cp /tmp/build/lammps/examples/reaxff/HNS/* /app/ \
    # Clean up the build environment to reduce image size
    && cd / \
    && rm -rf /tmp/build \
    && apt-get purge -y --auto-remove \
        build-essential \
        cmake \
        git \
        libfftw3-dev \
        libopenmpi-dev

# Set the final working directory for the container
WORKDIR /app

# Define a default command (optional, can be overridden)
CMD ["/bin/bash"]
