# Use a common base image like Ubuntu 22.04 LTS
FROM ubuntu:22.04

# Set non-interactive frontend for package installers to avoid prompts
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI libraries, git, required certificates, and libgomp1
# Combine into a single RUN layer to reduce image size
RUN apt-get update && \
    apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    gfortran \
    openmpi-bin \
    libopenmpi-dev \
    ca-certificates \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Configure OpenMPI environment variables for container compatibility
# This allows MPI to run as root and oversubscribe resources, which is common in Kubernetes
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Clone the latest branch of the LAMMPS source code from the official repository
# Using --depth 1 significantly reduces the clone size
RUN git clone --depth 1 https://github.com/lammps/lammps.git /opt/lammps-src

# Build LAMMPS from source using CMake
# Enable the REAXFF package as required by the 'lammps-reax' application
# This is a CPU-only build with MPI enabled
RUN cd /opt/lammps-src && \
    mkdir build && \
    cd build && \
    cmake \
        -D CMAKE_BUILD_TYPE=Release \
        -D BUILD_MPI=yes \
        -D PKG_REAXFF=on \
        ../cmake && \
    make -j$(nproc)

# Copy the compiled LAMMPS executable to a system-wide location on the PATH
# This makes the 'lmp' command available globally
RUN cp /opt/lammps-src/build/lmp /usr/local/bin/

# Set up the working directory for application runs
WORKDIR /app

# Copy all files from the ReaxFF HNS example directory into the WORKDIR
# These can be used as a starting point or for validation
RUN cp /opt/lammps-src/examples/reaxff/HNS/* /app/

# Clean up the system to reduce the final image size
# Remove build-only dependencies and the downloaded source code
RUN apt-get purge -y --auto-remove \
    build-essential \
    cmake \
    git \
    gfortran \
    libopenmpi-dev && \
    rm -rf /opt/lammps-src

# Set the default command to run the LAMMPS executable
# This will print the help message if no arguments are provided
CMD ["lmp"]
