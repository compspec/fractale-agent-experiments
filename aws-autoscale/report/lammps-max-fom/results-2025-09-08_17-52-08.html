# Use a standard Ubuntu LTS release as the base image
FROM ubuntu:22.04

# Set non-interactive frontend for package managers
ENV DEBIAN_FRONTEND=noninteractive

# Install build dependencies, MPI, LAMMPS prerequisites, and certificates.
# Clean up apt cache in the same layer to reduce image size.
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential \
    cmake \
    git \
    wget \
    ca-certificates \
    libfftw3-dev \
    openmpi-bin \
    libopenmpi-dev \
    libgomp1 && \
    rm -rf /var/lib/apt/lists/*

# Set OpenMPI environment variables to allow running as root, which is common in containers.
ENV OMPI_ALLOW_RUN_AS_ROOT=1
ENV OMPI_ALLOW_RUN_AS_ROOT_CONFIRM=1

# Create a temporary directory for the build process.
# This entire process is done in a single RUN command to minimize layer size by
# ensuring the source code and build artifacts are not stored in the final image.
RUN mkdir /tmp/build && cd /tmp/build && \
    # Clone the latest stable branch of the LAMMPS source code
    git clone --depth 1 --branch stable https://github.com/lammps/lammps.git . && \
    # Create a build directory
    mkdir build && cd build && \
    # Configure the build with CMake. Enable MPI and the REAXFF package.
    cmake \
      -D CMAKE_BUILD_TYPE=Release \
      -D BUILD_MPI=yes \
      -D PKG_REAXFF=yes \
      ../cmake && \
    # Compile LAMMPS using all available processor cores
    make -j$(nproc) && \
    # Copy the main LAMMPS executable to a location in the system's PATH
    cp lmp /usr/local/bin/lmp && \
    # Create the application work directory and copy the required example files
    mkdir -p /app && \
    cp /tmp/build/examples/reaxff/HNS/* /app/ && \
    # Clean up the build directory to keep the final image clean
    cd / && rm -rf /tmp/build

# Set the working directory for the container
WORKDIR /app

# Set the default entrypoint to the LAMMPS executable.
# Users can append input files and arguments at runtime (e.g., docker run image -in in.hns)
ENTRYPOINT ["lmp"]
